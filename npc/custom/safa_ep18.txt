	end;

OnNPCKillEvent:
	if (killedrid == 0) end;

	getmapxy(.@map$, .@x, .@y, BL_PC);
	// inarray statt einer Kette von Vergleichen: das Ereignis
	// feuert bei JEDEM Kill auf dem ganzen Server, also auch bei
	// jedem Poring. Was hier steht, laeuft millionenfach.
	if (inarray(.karte$[0], .@map$) < 0) end;

	if (rand(10000) >= 2000) end;
	makeitem 1000405, 1, .@map$, .@x, .@y;
	end;

OnInit:
	setarray .karte$[0], "ra_fild10","ra_fild11","gw_fild01","gw_fild02","oz_dun01","oz_dun02","ra_temin","ra_temple";
	end;
