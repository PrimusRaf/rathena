//===== SafaRO Script ========================================
//= Wildlife - die Rudel der Episode 22
//===== Erzeugt von ==========================================
//= tools/wildlife/bauen.py  --  NICHT von Hand pflegen.
//= Aenderungen gehoeren ins Skript, dann neu laufen lassen.
//===========================================================
//
//  WAS PASSIERT
//
//  Wer draussen ein Monster erlegt, hat eine kleine Chance,
//  dass an derselben Stelle die WILDLIFE-Fassung desselben
//  Monsters aufsteht: gleicher Sprite, gleicher Name mit dem
//  Vorsatz "Wildlife", stark erhoehte Lebenspunkte, aggressiv -
//  und ein Rudel aus vier weiteren Arten DERSELBEN KARTE im
//  Ruecken.
//
//  Wie hart genau, steht nicht fest, sondern richtet sich nach
//  dem Spieler, der das Rudel geweckt hat. Die Stellschrauben
//  dafuer stehen unten in OnInit und sind dort einzeln
//  begruendet.
//
//  Aus einem Poring wird ein Wildlife Poring, und mit ihm
//  stehen Wildlife Drops, Wildlife Poporing und was sonst noch
//  auf dieser Karte lebt. Das Rudel ist also ortstypisch und
//  nicht ausgewuerfelt.
//
//  WARUM DAS DEN SERVER SCHWERER MACHT
//
//  Nicht ueber Zahlen, sondern ueber Ueberraschung. Bisher war
//  jede Karte berechenbar: man kennt den Besatz, man kennt den
//  Schaden, man laesst @autopot laufen. Ein Rudel, das aus
//  einem gewoehnlichen Kill entsteht und sofort angreift,
//  unterbricht genau diese Routine - und zwar dort, wo der
//  Spieler ohnehin steht, statt in einem Gebiet, das er
//  betreten muss.
//
//  WARUM ES KEINE EIGENEN MONSTER-IDS GIBT
//
//  Der Client sucht das Sprite ueber die Monsternummer, in
//  npcidentity.lub und jobname.lub aus data.grf. Eine erfundene
//  Nummer bliebe unsichtbar - dieselbe Falle, die schon im Kopf
//  von db/import/mob_db.yml beim Golden Poring beschrieben ist.
//
//  Wildlife benutzt deshalb den amtlichen Monstertyp und wertet
//  ihn beim Erscheinen ueber setunitdata auf. Kein Eintrag in
//  mob_db, keiner in mob_avail, kein Client-Update - und es
//  wirkt von selbst auf jedes Monster, das rAthena je
//  hinzufuegt.
//
//  WAS AN DIESER DATEI ERZEUGT IST
//
//  Nur die Tabelle unten: Karte -> Arten. Sie laesst sich zur
//  Laufzeit nicht ermitteln, weil Spawns in rAthena NPC-Objekte
//  sind und keine abfragbaren Daten. Alles andere - Werte,
//  Rasse, Erfahrung, Ausschluss von MVPs - holt das Skript ueber
//  getmonsterinfo und bleibt dadurch aktuell.
//
//  MVPS UND MINIBOSSE SIND AUSGENOMMEN
//
//  Zweimal: beim Erzeugen der Tabelle (Class: Boss und
//  Modes.Mvp aus mob_db.yml) und noch einmal zur Laufzeit ueber
//  MD_MVP und MD_STATUSIMMUNE. Doppelt, weil die Tabelle nur die
//  RUDELMITGLIEDER filtert - der Ringleader entsteht aus dem
//  Monster, das der Spieler gerade erlegt hat, und das steht in
//  keiner Tabelle.
//
//===========================================================


-	script	safa_wildlife	-1,{
	end;


// ------------------------------------------------------------
//  Der Ausloeser
// ------------------------------------------------------------
//  Dieses Ereignis feuert bei JEDEM Kill auf dem ganzen Server.
//  Die Reihenfolge der Pruefungen ist deshalb kein Zufall,
//  sondern nach Kosten sortiert: erst Zahlenvergleiche auf
//  Variablen, die schon im Speicher liegen, dann der Wuerfel,
//  und erst ganz zuletzt die Datenbankzugriffe ueber
//  getmonsterinfo und die Kartensuche.
//
//  Bei fuenf Prozent laeuft alles ab dem Wuerfel nur noch bei
//  jedem zwanzigsten Kill - und selbst davon faellt das meiste
//  gleich wieder an der Spielersperre und am Rudeldeckel weg.
OnNPCKillEvent:
	if (killedrid == 0) end;
	if (callsub(S_Aktiv) >= .max) end;
	if (gettimetick(2) < @wild_cd) end;
	if (rand(10000) >= .chance) end;

	// getmonsterinfo liefert -1 bei unbekannter Klasse. Ohne
	// diese Pruefung ginge -1 durch und jeder Kill zaehlte -
	// derselbe Fehler, der bei den MVP-Muenzen vermieden wurde.
	.@mode = getmonsterinfo(killedrid, MOB_MODE);
	if (.@mode == -1) end;

	// MD_MVP sind die MVPs, MD_STATUSIMMUNE setzt rAthena bei
	// Class: Boss - also den Minibossen. Aus keinem von beiden
	// wird ein Wildlife.
	if (.@mode & MD_MVP) end;
	if (.@mode & MD_STATUSIMMUNE) end;

	if (getmonsterinfo(killedrid, MOB_LV) < .min_lv) end;

	// Champion-Monster als Vorlage ergaeben einen MENSCHEN: der
	// Client kennt diese von rAthena erfundenen Nummern nicht und
	// zeichnet ersatzweise ein Novizensprite. Aus der Tabelle sind
	// sie gefiltert - aber der Ausloeser hier ist das ERLEGTE
	// Monster, und Champions laufen ueberall herum
	// (npc/re/mobs/championmobs.txt).
	//
	// Die Spanne ist lueckenlos und aus db/re/mob_db.yml gezaehlt,
	// nicht geschaetzt: 311 Monster mit AegisName C1_ bis C5_,
	// Nummern 2603 bis 2913.
	if (killedrid >= .champ_von && killedrid <= .champ_bis) end;

	getmapxy(.@map$, .@x, .@y, BL_PC);
	.@i = inarray(.karte$[0], .@map$);
	if (.@i < 0) end;

	// Die Sperre gilt dem SPIELER, nicht dem Server. Sonst
	// koennte einer mit hoher Tempo-Ausruestung alle Rudel des
	// Servers auf sich ziehen.
	@wild_cd = gettimetick(2) + .cd;

	callsub S_Rudel, .@map$, .@x, .@y, killedrid, .@i;
	end;


// ------------------------------------------------------------
//  @wild - nachsehen, ob das hier ueberhaupt greift
// ------------------------------------------------------------
//  Ohne diesen Befehl bleibt nur: Monster kloppen und hoffen.
//  Und wenn dann nichts kommt, weiss man nicht, woran es lag -
//  an der Karte, am Deckel, an der Sperre, am Wuerfel oder
//  daran, dass die Datei gar nicht geladen ist.
//
//    @wild         Status: was gilt hier gerade
//    @wild los     ein Rudel erzwingen, ohne Wuerfel und Sperre
//    @wild aus     alle Rudel entfernen, ohne Belohnung
//    @wild karte   welche Arten diese Karte beisteuert
//
//  "los" geht bewusst am Deckel und an der Sperre vorbei. Der
//  Zweck ist zu sehen, ob die KETTE funktioniert - Tabelle,
//  Spawn, Werteaufwertung, Helfer, Ansage, Tod, Belohnung. Ob
//  die Wahrscheinlichkeit richtig eingestellt ist, ist eine
//  andere Frage und steht im Status.
OnGM:
	.@was$ = "";
	if (.@atcmd_numparameters > 0)
		.@was$ = .@atcmd_parameters$[0];

	getmapxy(.@map$, .@x, .@y, BL_PC);
	.@i = inarray(.karte$[0], .@map$);

	// --- aufraeumen ------------------------------------------
	if (.@was$ == "aus") {
		for (.@k = 0; .@k < .max; .@k++) {
			if (!$@wild_gid[.@k]) continue;
			killmonster $@wild_karte$[.@k], "safa_wildlife::OnWildTot";
			killmonster $@wild_karte$[.@k], "safa_wildlife::OnWildHelfer";
			$@wild_gid[.@k] = 0;
			$@wild_karte$[.@k] = "";
			$@wild_alter[.@k] = 0;
		}
		dispbottom "[Wildlife] Alle Rudel entfernt.";
		end;
	}

	// --- Arten dieser Karte ----------------------------------
	if (.@was$ == "karte") {
		if (.@i < 0) {
			dispbottom "[Wildlife] " + .@map$
			           + " steht NICHT in der Tabelle. Hier kommt nie eines.";
			end;
		}
		dispbottom "[Wildlife] " + .@map$ + " steuert "
		           + .anz[.@i] + " Arten bei:";
		for (.@k = 0; .@k < .anz[.@i]; .@k++)
			dispbottom "   " + .art[.von[.@i] + .@k] + "  "
			           + getmonsterinfo(.art[.von[.@i] + .@k], MOB_NAME);
		end;
	}

	// --- erzwingen -------------------------------------------
	if (.@was$ == "los") {
		if (.@i < 0) {
			dispbottom "[Wildlife] " + .@map$
			           + " steht nicht in der Tabelle - hier geht es nicht.";
			end;
		}
		// Vorlage aus dem Bestand dieser Karte. Beim normalen
		// Weg ist es das gerade erlegte Monster; hier gibt es
		// keines, also wird eines der ortsueblichen gezogen.
		.@basis = .art[.von[.@i] + rand(.anz[.@i])];

		// S_Rudel steigt aus, wenn kein Platz frei ist. Das
		// merkt man sonst nicht - es passiert schlicht nichts.
		.@vorher = callsub(S_Aktiv);
		// Letztes Argument 1: S_Rudel berichtet, was es tut.
		callsub S_Rudel, .@map$, .@x, .@y, .@basis, .@i, 1;

		if (callsub(S_Aktiv) == .@vorher) {
			dispbottom "[Wildlife] Nichts passiert. Alle " + .max
			           + " Plaetze belegt - erst \"@wild aus\".";
			end;
		}
		dispbottom "[Wildlife] Erzwungen: "
		           + getmonsterinfo(.@basis, MOB_NAME)
		           + ". Aktive Rudel jetzt " + callsub(S_Aktiv) + ".";
		end;
	}

	// --- Status ----------------------------------------------
	dispbottom "[Wildlife] Skript ist geladen.";
	dispbottom "  Karte " + .@map$ + ": "
	           + (.@i < 0 ? "NICHT in der Tabelle - hier kommt nie eines"
	                      : "in der Tabelle, " + .anz[.@i] + " Arten");
	dispbottom "  Tabelle: " + getarraysize(.karte$) + " Karten, "
	           + getarraysize(.art) + " Arteneintraege";
	// Das doppelte Prozentzeichen ist kein Tippfehler: der
	// Rumpf dieser Datei geht durch Pythons %-Formatierung, und
	// ein einzelnes % waere dort eine Formatangabe. Im
	// erzeugten Skript steht wieder eines.
	dispbottom "  Chance: " + .chance + " von 10000 ("
	           + (.chance / 100) + "," + (.chance % 100)
	           + " Prozent) je Kill";
	.@a = callsub(S_Aktiv);
	dispbottom "  Rudel aktiv: " + .@a + " von " + .max
	           + ((.@a >= .max) ? "  <-- DECKEL ERREICHT, nichts Neues" : "");

	.@rest = @wild_cd - gettimetick(2);
	dispbottom "  Deine Sperre: "
	           + ((.@rest > 0) ? .@rest + " Sekunden <-- laeuft noch"
	                           : "frei");
	dispbottom "  Mindestlevel der Vorlage: " + .min_lv
	           + ", Rudel loest sich nach " + .leben + " Minuten auf";

	for (.@k = 0; .@k < .max; .@k++) {
		if (!$@wild_gid[.@k]) continue;
		dispbottom "   Platz " + .@k + ": Nr " + $@wild_gid[.@k] + " auf "
		           + $@wild_karte$[.@k] + ", " + $@wild_alter[.@k] + " Min"
		           + (unitexists($@wild_gid[.@k]) ? "  lebt" : "  <-- Tier weg, zaehlt nicht mehr mit");
	}
	end;


// ------------------------------------------------------------
//  Wieviele Rudel stehen gerade wirklich
// ------------------------------------------------------------
//  GEZAEHLT, NICHT MITGESCHRIEBEN - und das ist der Punkt.
//
//  Vorher lief hier ein Zaehler $@wild_aktiv mit: eins hoch beim
//  Aufstellen, eins runter beim Tod, eins runter im Aufraeumtakt.
//  Drei Stellen, die alle stimmen mussten.
//
//  Stimmte eine davon einmal nicht - ein Anfuehrer, den kein
//  Spieler erlegt hat, ein @killmonster von aussen, ein
//  Serverzustand zwischen zwei Reloads -, dann stand der Zaehler
//  zu hoch. Und weil er zu hoch stand, sperrte die allererste
//  Pruefung im Ausloeser jeden weiteren Spawn. Fuer immer.
//
//  Das Fiese daran: $@-Variablen ueberleben @reloadscript. Ein
//  einmal verklemmter Zaehler blieb also verklemmt, egal wie oft
//  man die Datei neu lud - nur ein Serverneustart half. Von
//  aussen sah es aus, als griffe das Skript gar nicht.
//
//  Jetzt gibt es keinen Zaehler mehr. Gezaehlt wird, was
//  tatsaechlich noch lebt. Das kann nicht auseinanderlaufen.
S_Aktiv:
	.@n = 0;
	for (.@z = 0; .@z < .max; .@z++)
		if ($@wild_gid[.@z] && unitexists($@wild_gid[.@z]))
			.@n++;
	return .@n;


// ------------------------------------------------------------
//  Das Rudel aufstellen
// ------------------------------------------------------------
S_Rudel:
	.@map$   = getarg(0);
	.@x      = getarg(1);
	.@y      = getarg(2);
	.@basis  = getarg(3);
	.@i      = getarg(4);
	.@laut   = getarg(5, 0);   // 1 = Schritt fuer Schritt berichten

	// --- Erst den Platz, dann das Tier --------------------------
	//  Die Reihenfolge ist wichtig. Umgekehrt stuende bei einem
	//  vollen Buch bereits ein Rudel auf der Karte, das in keiner
	//  Liste gefuehrt wird - der Aufraeumtimer kennt es nicht und
	//  es bliebe bis zum Neustart stehen.
	//
	//  WARUM EINE SCHLEIFE UND KEIN inarray
	//
	//  Weil rAthena ein Feldelement LOESCHT, sobald man ihm
	//  seinen Standardwert zuweist - Skriptfelder sind intern
	//  eine Zuordnung Index->Wert, kein Block. Ein freier Platz
	//  (Wert 0) ist danach nicht "0", sondern gar nicht da, und
	//  getarraysize endet beim letzten belegten Platz.
	//
	//  Steht also nur auf Platz 0 ein Rudel und die uebrigen
	//  fuenf sind frei, hat $@wild_gid die Laenge 1. inarray
	//  suchte dann NUR in diesem einen Platz, faende die 0 nicht
	//  und lieferte -1 - und ein Rueckfall auf Platz 0 haette das
	//  dort eingetragene Rudel aus der Liste geworfen. Es waere
	//  fuer immer stehengeblieben.
	//
	//  Die Schleife liest ueber das Feldende hinaus, und das ist
	//  hier genau richtig: ein nicht vorhandenes Element liefert
	//  0, also "frei".
	.@s = -1;
	for (.@k = 0; .@k < .max; .@k++) {
		if ($@wild_gid[.@k] == 0) {
			.@s = .@k;
			break;
		}
	}
	if (.@s < 0) return;

	.@name$ = "Wildlife " + getmonsterinfo(.@basis, MOB_NAME);

	if (.@laut)
		dispbottom "  Vorlage " + .@basis + " \"" + .@name$
		           + "\", Rasse " + getmonsterinfo(.@basis, MOB_RACE)
		           + ", Stufe " + getmonsterinfo(.@basis, MOB_LV);

	monster .@map$, .@x, .@y, .@name$, .@basis, 1,
	        "safa_wildlife::OnWildTot";
	.@gid = $@mobid[0];
	if (!.@gid) {
		if (.@laut)
			dispbottom "  FEHLER: monster hat keine Spielnummer geliefert.";
		return;
	}
	// getunitdata GIBT NICHTS ZURUECK - es fuellt ein Feld.
	// doc/script_commands.txt:8885: getunitdata <GID>,<arrayname>.
	// Der Wert steht danach unter dem Index der UMOB-Konstante.
	//
	// Das war der Fehler, an dem das ganze System haengengeblieben
	// ist: jeder Aufruf der Form getunitdata(gid, UMOB_X) hat im
	// Log "Error in argument! Please give a variable to store
	// values in" gemeldet, nichts geliefert - und das Skript an
	// dieser Stelle abgebrochen. Deshalb blieb das Tier bei seinen
	// Grundwerten, blieb harmlos und bekam nie Helfer.
	getunitdata .@gid, .@d;

	if (.@laut)
		dispbottom "  Gespawnt, Spielnummer " + .@gid
		           + ", Modus vorher " + .@d[UMOB_MODE];

	// --- Kampfwerte am Ausloeser ausrichten ------------------
	//  WARUM UEBERHAUPT AUSRICHTEN
	//
	//  Ein fester Zahlensatz kann auf diesem Server nicht
	//  funktionieren. Zwischen Stufe 5 und Stufe 500 liegen bei
	//  Ausweichen, Ruestung und Lebenspunkten drei Groessen-
	//  ordnungen. Was den einen umbringt, merkt der andere nicht.
	//  Also wird jedes Rudel auf den zugeschnitten, der es
	//  geweckt hat.
	//
	//  DIE TREFFERFORMEL IM RENEWAL
	//
	//  battle.cpp:3291 setzt die Grundrate auf 0 - NICHT auf 80,
	//  das ist Pre-Renewal. battle.cpp:3310 addiert dann
	//  hit - flee, battle.cpp:3344 kappt auf min_hitrate 5 bis
	//  max_hitrate 100. Es gilt also schlicht:
	//
	//      Trefferchance% = MOB_HIT - SPIELER_FLEE
	//
	//  Ein Mob-HIT ist von Haus aus Level + DEX + 150
	//  (status.cpp:2640), ein Spieler-FLEE Level + AGI + LUK/5
	//  + 100 + Ausruestung (status.cpp:2644). Auf Stufe 500 steht
	//  also 270 gegen 900. Die Tiere trafen zu 5 Prozent, und
	//  daran haette auch der zehnfache Angriffswert nichts
	//  geaendert.
	//
	//  DASS DAS HIER UEBERHAUPT WIRKT
	//
	//  setunitdata UMOB_HIT schreibt den ENDWERT, nicht einen
	//  Grundwert. script.cpp:19535 ruft danach nur SCB_BATTLE,
	//  und SCB_BATTLE schliesst SCB_BASE aus (status.hpp:3304) -
	//  status_calc_misc laeuft also nicht erneut und rechnet
	//  Level + DEX + 150 nicht noch einmal obendrauf.
	.@ziel = getcharid(3);

	//  Ohne angehaengten Spieler gibt readparam 0 zurueck, und
	//  daraus wuerde ein Tier mit Angriff 0. Dann lieber die
	//  Ruecklage aus OnInit.
	if (.@ziel) {
		.@p_flee = readparam(bFlee);
		.@p_hit  = readparam(bHit);
		.@p_def1 = readparam(bDef);
		.@p_def2 = readparam(bDef2);
		.@p_hp   = readparam(bMaxHP);
	} else {
		.@p_flee = .not_flee;
		.@p_hit  = .not_hit;
		.@p_def1 = 0;
		.@p_def2 = 0;
		.@p_hp   = .not_hp;
	}

	// TREFFEN: so viel Prozent der Schlaege sollen ankommen.
	.@k_hit = .@p_flee + .treffer_mob;

	// AUSWEICHEN: hier haengt der Soeldner.
	//
	//  status_calc_mercenary_ rechnet mit dem DB-Level des
	//  Soeldners, nicht mit deinem (status.cpp:5037). Der
	//  staerkste Eintrag in db/re/mercenary_db.yml steht auf
	//  Level 87 mit DEX 87, sein HIT ist also bei rund 324
	//  gedeckelt - egal, wie weit sein Herr ist. Ein Spieler auf
	//  Stufe 500 kommt auf ueber 900. Zwischen diesen beiden
	//  Zahlen liegt ein Fenster, in dem der Soeldner haeufig
	//  danebenhaut und der Spieler trifft.
	//
	//  Der feste Wert allein waere aber eine Falle fuer niedrige
	//  Stufen: ein Spieler auf Stufe 50 hat selbst nur HIT 268
	//  und traefe dann zu 18 Prozent. Deshalb die Deckelung an
	//  seinem eigenen Trefferwert - er soll immer .treffer_spieler
	//  Prozent landen, der Soeldner nimmt, was uebrig bleibt.
	.@k_flee = .flee_wert;
	.@pmax = .@p_hit - .treffer_spieler;
	if (.@k_flee > .@pmax) .@k_flee = .@pmax;
	if (.@k_flee < 0)      .@k_flee = 0;

	// SCHADEN: Zielgroesse ist ein Anteil der Lebenspunkte des
	// Ausloesers PRO TREFFER - VOR seiner Ruestung.
	//
	//  WAS HIER FRUEHER STAND UND WARUM ES WEG IST
	//
	//  Die Renewal-Formel lautet (battle.cpp:4899):
	//
	//      Schaden = Angriff * (4000 + DEF) / (4000 + 10*DEF) - DEF2
	//
	//  Der erste Anlauf hat sie nach Angriff AUFGELOEST, damit nach
	//  Abzug von DEF und DEF2 genau .schaden_pct uebrig bleibt:
	//
	//      .@k_atk = (.@will + .@p_def2) * (4000 + 10*.@p_def1)
	//                / (4000 + .@p_def1)
	//
	//  Das war sauber gerechnet und trotzdem falsch. Es rechnet die
	//  Ruestung des Spielers HERAUS. Gegen Wildlife waren DEF, VIT
	//  und jedes Ruestungsteil damit exakt wirkungslos - der
	//  Tankbuild und der Papierbuild verloren denselben Anteil ihrer
	//  Lebenspunkte je Treffer. Ein Gegner, gegen den Ausruestung
	//  nichts aendert, ist kein Gegner, sondern eine Uhr.
	//
	//  Jetzt steht der Angriffswert schlicht auf dem Zielschaden,
	//  und die Formel oben laeuft ganz normal darueber. Ein Spieler
	//  mit DEF 300 nimmt davon 4300/7000, also 61 Prozent; mit
	//  DEF 900 sind es 4900/13000, also 38 Prozent. Genau dieser
	//  Unterschied hat vorher gefehlt.
	//
	//  .schaden_pct ist deshalb jetzt der Wert VOR der Ruestung und
	//  entsprechend hoeher angesetzt als die 3, die vorher danach
	//  uebrig blieben.
	//
	//  Weiterhin nicht eingerechnet: Kartenabzuege, Element und
	//  Rassenresistenzen - die kommen danach und ziehen den Wert
	//  zusaetzlich herunter. Ebenso batk (STR + Level,
	//  status.cpp:2484), das oben draufkommt; bei dreistelligen
	//  Werten faellt es neben zehntausenden Angriff nicht ins
	//  Gewicht.
	.@k_atk = .@p_hp * .schaden_pct / 100;

	//  HARTE GRENZE 65535.
	//
	//  status.hpp:3365 - struct weapon_atk fuehrt atk und atk2 als
	//  uint16. Ein groesserer Wert laeuft ueber und wird winzig;
	//  aus 70000 wuerden 4464. Genau so ein Ueberlauf faellt im
	//  Spiel als "das Vieh kitzelt nur" auf und ist von aussen
	//  nicht zu sehen. Deshalb wird hier gekappt und im
	//  ausfuehrlichen Bericht vermerkt.
	.@gekappt = 0;
	if (.@k_atk > 65535) {
		.@k_atk = 65535;
		.@gekappt = 1;
	}
	if (.@k_atk < 1) .@k_atk = 1;

	if (.@laut) {
		dispbottom "  Spieler: FLEE " + .@p_flee + ", HIT " + .@p_hit
		           + ", DEF " + .@p_def1 + "/" + .@p_def2
		           + ", MaxHP " + .@p_hp;
		dispbottom "  Daraus: Mob-HIT " + .@k_hit
		           + " (Trefferrate " + .treffer_mob + "%)"
		           + ", Mob-FLEE " + .@k_flee
		           + ", Angriff " + .@k_atk
		           + (.@gekappt ? "  <-- auf 65535 gekappt, uint16" : "");
	}

	// --- Werte anheben ---------------------------------------
	//  Der HP-Faktor ist absichtlich degressiv. Vierzigfach ist
	//  bei einem Poring (50 HP) eine Ansage und bei einem
	//  Monster mit fuenf Millionen HP eine Wand, die niemand
	//  umwirft, bevor der Aufraeumtimer greift. Ein Rudel, das
	//  man nicht toeten KANN, ist keine Schwierigkeit, sondern
	//  eine Sperre.
	//
	//  Der Faktor allein reicht aber nicht: ein Poring kommt so
	//  auf 2000 Lebenspunkte, und das raeumt ein Soeldner im
	//  Vorbeigehen weg. Deshalb zusaetzlich eine Untergrenze,
	//  die an den Lebenspunkten des Ausloesers haengt - wer viel
	//  aushaelt, trifft auf etwas, das ebenfalls viel aushaelt.
	//  Genommen wird der groessere der beiden Werte.
	.@hp = .@d[UMOB_MAXHP];
	.@f = .hp_faktor;
	if (.@hp > 1000000)     .@f = .hp_faktor / 5;
	else if (.@hp > 100000) .@f = .hp_faktor / 2;
	.@hp = .@hp * .@f;

	.@boden = .@p_hp * .hp_spieler;
	if (.@hp < .@boden) .@hp = .@boden;
	if (.@hp > .hp_deckel) .@hp = .hp_deckel;

	setunitdata .@gid, UMOB_MAXHP, .@hp;
	setunitdata .@gid, UMOB_HP,    .@hp;
	if (.@laut)
		dispbottom "  HP gesetzt auf " + .@hp
		           + " (Vorlage " + .@d[UMOB_MAXHP]
		           + ", Untergrenze " + .@boden + ")";

	setunitdata .@gid, UMOB_ATKMIN, .@k_atk * 80 / 100;
	setunitdata .@gid, UMOB_ATKMAX, .@k_atk;
	setunitdata .@gid, UMOB_HIT,    .@k_hit;
	setunitdata .@gid, UMOB_FLEE,   .@k_flee;
	setunitdata .@gid, UMOB_DEF,    .@d[UMOB_DEF] + .def_plus;
	setunitdata .@gid, UMOB_MDEF,   .@d[UMOB_MDEF] + .def_plus;
	setunitdata .@gid, UMOB_LEVEL,  .@d[UMOB_LEVEL] + .lv_plus;

	// --- Schlagtempo begrenzen -------------------------------
	//  Das Rudel fuehlte sich an, als haetten die Tiere 199 ASPD.
	//  Sie haben es nicht - dieses Skript fasst das Tempo gar nicht
	//  an, jedes Tier behaelt das seiner Vorlage. Sechs davon
	//  gleichzeitig ergeben aber dasselbe Bild, und bei einer
	//  schnellen Vorlage wird daraus ein Trommelfeuer.
	//
	//  AMOTION ist die Dauer der Schlaganimation, ADELAY der
	//  Abstand zwischen zwei Schlaegen (status.cpp, mob_db-Felder
	//  AttackMotion und AttackDelay). Beide sind ueber setunitdata
	//  erreichbar (script.cpp:19508 und 19509).
	//
	//  Gesetzt wird nur eine UNTERGRENZE. Ein von Haus aus traeges
	//  Tier soll traege bleiben - schneller machen wollen wir hier
	//  niemanden.
	if (.@d[UMOB_ADELAY] < .adelay_min)
		setunitdata .@gid, UMOB_ADELAY, .adelay_min;
	if (.@d[UMOB_AMOTION] < .amotion_min)
		setunitdata .@gid, UMOB_AMOTION, .amotion_min;

	// --- Angriffsverhalten -----------------------------------
	//  MD_AGGRESSIVE allein reicht NICHT, und das war der Fehler
	//  im ersten Anlauf: das Tier stand angriffslustig da und tat
	//  nichts.
	//
	//  Der Grund steht in mmo.hpp:249 - MD_CANATTACK ist ein
	//  EIGENES Flag (0x80). Ein Poring traegt von Haus aus nur
	//  MD_CANMOVE; er kann schlicht nicht zuschlagen. MD_AGGRESSIVE
	//  macht daraus ein Tier, das angreifen WILL und es nicht KANN.
	//
	//  Deshalb das ganze Buendel:
	//    MD_CANMOVE            laufen
	//    MD_CANATTACK          ueberhaupt zuschlagen koennen
	//    MD_AGGRESSIVE         von sich aus angreifen
	//    MD_ANGRY              auch verfolgen, nicht nur zuschlagen
	//    MD_CHANGECHASE        Ziel im Verfolgen wechseln
	//    MD_CHANGETARGETMELEE  auf den reagieren, der zuschlaegt
	//
	//  ODER statt setzen: was das Tier von Haus aus mitbringt -
	//  Looter, Detector, Elementarimmunitaeten - soll bleiben.
	setunitdata .@gid, UMOB_MODE,
	            .@d[UMOB_MODE]
	            | MD_CANMOVE | MD_CANATTACK | MD_AGGRESSIVE
	            | MD_ANGRY | MD_CHANGECHASE | MD_CHANGETARGETMELEE;

	// --- Und dann der Befehl -----------------------------------
	//  Die Mode-Flags sagen dem Tier, dass es angreifen DARF.
	//  Sie sagen ihm nicht, dass es JETZT anfangen soll - das
	//  entscheidet die Monster-KI im naechsten Takt, und ob sie
	//  einen Spieler ueberhaupt bemerkt, haengt an Sichtweite,
	//  Reizschwelle und daran, in welcher KI-Liste das Tier
	//  gerade steckt.
	//
	//  unitattack ueberspringt das alles: es setzt das Ziel
	//  direkt. Der zweite Parameter 1 heisst "dranbleiben", nicht
	//  einmal zuschlagen.
	//
	//  Ausloeser ist immer ein Spieler - entweder der, der etwas
	//  erlegt hat, oder der GM mit @wild los. Ohne angehaengten
	//  Spieler waere getcharid(3) null; die Pruefung faengt das
	//  ab, statt einen Fehler ins Log zu schreiben. .@ziel steht
	//  schon oben, wo die Kampfwerte geholt wurden.
	if (.@ziel)
		unitattack .@gid, .@ziel, 1;

	if (.@laut) {
		getunitdata .@gid, .@n;
		dispbottom "  Modus jetzt " + .@n[UMOB_MODE]
		           + "  CANATTACK=" + ((.@n[UMOB_MODE] & MD_CANATTACK) ? 1 : 0)
		           + " AGGRESSIVE=" + ((.@n[UMOB_MODE] & MD_AGGRESSIVE) ? 1 : 0)
		           + " CANMOVE=" + ((.@n[UMOB_MODE] & MD_CANMOVE) ? 1 : 0);
		//  Rueckgelesen, nicht nur berechnet: nur so faellt ein
		//  Ueberlauf oder ein von der Engine verworfener Wert auf.
		dispbottom "  Zurueckgelesen: HIT " + .@n[UMOB_HIT]
		           + ", FLEE " + .@n[UMOB_FLEE]
		           + "   (gesetzt: " + .@k_hit + " / " + .@k_flee + ")";
		dispbottom "  Angriff " + .@n[UMOB_ATKMIN] + " bis "
		           + .@n[UMOB_ATKMAX] + ", HP " + .@n[UMOB_MAXHP]
		           + ", Stufe " + .@n[UMOB_LEVEL];
		//  Zurueckgelesen, weil setunitdata ADELAY/AMOTION nur
		//  base_status setzt und SCB_BATTLE nachrechnen laesst -
		//  ob der Wert wirklich stehenbleibt, sagt nur diese Zeile.
		dispbottom "  Tempo: ADELAY " + .@n[UMOB_ADELAY]
		           + ", AMOTION " + .@n[UMOB_AMOTION]
		           + "   (Vorlage " + .@d[UMOB_ADELAY]
		           + "/" + .@d[UMOB_AMOTION]
		           + ", Untergrenze " + .adelay_min
		           + "/" + .amotion_min + ")";
		//  Was das Rudel je Angriffsrunde ausmacht - die Zahl, um
		//  die es beim Ausbalancieren wirklich geht.
		dispbottom "  Rudelschaden je Runde: " + .schaden_pct
		           + " + " + .helfer + "x"
		           + (.schaden_pct * .helfer_atk / 100) + " = "
		           + (.schaden_pct
		              + .helfer * .schaden_pct * .helfer_atk / 100)
		           + "% deiner MaxHP, VOR Ruestung";
	}

	// --- Das Rudel -------------------------------------------
	//  Die Helfer bekommen denselben Vorsatz. Das ist Absicht:
	//  wer "Wildlife" liest, weiss, dass dieses Tier nicht zum
	//  festen Besatz gehoert, gleich wieder verschwindet und
	//  haerter zuschlaegt als sein Namensvetter daneben.
	.@a = .von[.@i];
	.@n = .anz[.@i];
	if (.@laut)
		dispbottom "  Rudelquelle: Index " + .@i + ", .von=" + .@a
		           + ", .anz=" + .@n + ", .helfer=" + .helfer;
	for (.@k = 0; .@k < .helfer; .@k++) {
		.@art = .art[.@a + rand(.@n)];
		if (.@laut)
			dispbottom "   Helfer " + (.@k + 1) + ": Art " + .@art;

		monster .@map$, .@x, .@y,
		        "Wildlife " + getmonsterinfo(.@art, MOB_NAME),
		        .@art, 1, "safa_wildlife::OnWildHelfer";
		.@sg = $@mobid[0];
		if (!.@sg) continue;

		getunitdata .@sg, .@dh;

		//  Die Helfer bekommen dasselbe Trefferbild wie der
		//  Anfuehrer. Sonst waeren fuenf von sechs Tieren im
		//  Rudel harmlos, und genau so hat es sich vorher
		//  angefuehlt: ein Gegner, der ab und zu etwas macht,
		//  und Statisten drumherum.
		//
		//  Nur der Schaden ist geteilt - .helfer_atk Prozent
		//  vom Anfuehrer. Sechs Tiere mit vollem Schaden waeren
		//  das Sechsfache der eingestellten Zielgroesse.
		.@shp = .@dh[UMOB_MAXHP] * .helfer_hp;
		.@sboden = .@p_hp * .hp_spieler / 100 * .helfer_hp_pct;
		if (.@shp < .@sboden) .@shp = .@sboden;
		if (.@shp > .hp_deckel) .@shp = .hp_deckel;

		setunitdata .@sg, UMOB_MAXHP, .@shp;
		setunitdata .@sg, UMOB_HP,    .@shp;
		setunitdata .@sg, UMOB_ATKMIN,
		            .@k_atk * .helfer_atk / 100 * 80 / 100;
		setunitdata .@sg, UMOB_ATKMAX, .@k_atk * .helfer_atk / 100;
		setunitdata .@sg, UMOB_HIT,    .@k_hit;
		setunitdata .@sg, UMOB_FLEE,   .@k_flee;
		setunitdata .@sg, UMOB_DEF,    .@dh[UMOB_DEF] + .def_plus;
		setunitdata .@sg, UMOB_MDEF,   .@dh[UMOB_MDEF] + .def_plus;
		setunitdata .@sg, UMOB_LEVEL,  .@dh[UMOB_LEVEL] + .lv_plus;
		if (.@dh[UMOB_ADELAY] < .adelay_min)
			setunitdata .@sg, UMOB_ADELAY, .adelay_min;
		if (.@dh[UMOB_AMOTION] < .amotion_min)
			setunitdata .@sg, UMOB_AMOTION, .amotion_min;
		setunitdata .@sg, UMOB_MODE,
		            .@dh[UMOB_MODE]
		            | MD_CANMOVE | MD_CANATTACK | MD_AGGRESSIVE
		            | MD_ANGRY | MD_CHANGECHASE | MD_CHANGETARGETMELEE;
		if (.@ziel)
			unitattack .@sg, .@ziel, 1;
		.@gebaut++;
	}
	if (.@laut) {
		dispbottom "  Helfer gestellt: " + .@gebaut + " von " + .helfer;
		dispbottom "  Angriffsbefehl auf Spielernummer " + .@ziel
		           + (.@ziel ? "" : "  <-- kein Spieler angehaengt!");
	}

	// --- Buchfuehrung ----------------------------------------
	//  $@ ist serverweit und NICHT dauerhaft - genau richtig:
	//  ueber 'monster' gesetzte Tiere ueberleben einen Neustart
	//  ohnehin nicht, und eine dauerhafte Liste zeigte danach
	//  auf Nummern, die es nicht mehr gibt.
	//
	//  Der Platz .@s steht schon fest, siehe oben.
	$@wild_gid[.@s]    = .@gid;
	$@wild_karte$[.@s] = .@map$;
	$@wild_alter[.@s]  = 0;

	mapannounce .@map$,
	    "The ground shifts. A " + .@name$ + " has come up, and it is not alone.",
	    bc_map, "0xCC6600";
	return;


// ------------------------------------------------------------
//  Der Ringleader faellt
// ------------------------------------------------------------
OnWildTot:
	.@basis = killedrid;
	getmapxy(.@map$, .@x, .@y, BL_PC);

	// killedgid ist die Spielnummer des gefallenen Tieres, nicht
	// seine Klasse. Nur damit laesst sich der richtige Platz in
	// der Liste freigeben - killedrid waere bei zwei Rudeln
	// derselben Art mehrdeutig.
	.@s = inarray($@wild_gid[0], killedgid);
	if (.@s >= 0) {
		$@wild_gid[.@s]    = 0;
		$@wild_karte$[.@s] = "";
		$@wild_alter[.@s]  = 0;
	}

	// Ohne den Anfuehrer loest sich das Rudel auf. killmonster
	// mit Label loest das Todesereignis NICHT aus - die Helfer
	// verschwinden also, ohne noch einmal zu belohnen.
	killmonster .@map$, "safa_wildlife::OnWildHelfer";

	// --- Belohnung -------------------------------------------
	//  Der Faktor ist woertlich "so viele gewoehnliche Kills".
	//  getexp laeuft ueber quest_exp_rate, und die steht bei uns
	//  auf denselben 2500 wie base_exp_rate
	//  (conf/import/battle_conf.txt) - beide Seiten sind also
	//  gleich hochgerechnet und der Vergleich stimmt.
	//
	//  Fuenfzig war urspruenglich am Aufwand bemessen: vierzigfache
	//  Lebenspunkte plus fuenf Helfer waren rund fuenfundsechzig
	//  gewoehnliche Tiere an Arbeit.
	//
	//  Seit der Entschaerfung ist das Rudel kleiner - rund
	//  sechzehnfache Spieler-HP statt zweiundsechzigfache. Der
	//  Faktor steht trotzdem noch auf fuenfzig, und zwar bewusst:
	//  der Kampf soll sich lohnen, sonst weicht ihm jeder aus,
	//  statt sich darauf einzulassen. Wer das anders sieht, dreht
	//  .exp_faktor in OnInit - das ist die eine Zahl dafuer.
	getexp getmonsterinfo(.@basis, MOB_BASEEXP) * .exp_faktor,
	       getmonsterinfo(.@basis, MOB_JOBEXP) * .exp_faktor;

	// Monster koennen in rAthena kein Zeny fallen lassen - es
	// gibt schlicht kein Feld dafuer in mob_db. Deshalb hier.
	Zeny += getmonsterinfo(.@basis, MOB_LV) * .zeny_je_lv;

	// --- Und die Beute ---------------------------------------
	//  Trophaeen nur von Brute-Tieren, und das ist kein
	//  Versehen. Die Safari-Trophaee ist in
	//  npc/custom/safa_safari.txt an Brute ab Stufe 150
	//  gebunden, weil "Wild Safari" sonst ein leeres Wort waere.
	//  Wildlife haelt sich daran: ein Wildlife Zombie ist
	//  gefaehrlich, aber er ist kein Tier.
	//
	//  Alle uebrigen zahlen in Destiny Coins. Die sind auf
	//  diesem Server die zweite Waehrung und teuer genug
	//  (der Levelraum kostet 1.000), dass sich der Kampf auch
	//  ausserhalb der Safarigebiete lohnt.
	if (getmonsterinfo(.@basis, MOB_RACE) == RC_BRUTE)
		callfunc "F_SafaBeute", .mat, .troph;
	else
		callfunc "F_SafaBeute", .coin, .muenzen;

	// Fuer die Questreihe in npc/custom/safa_quests.txt. Der
	// Zaehler laeuft lebenslang weiter und wird von den Quests
	// nur mit einem Startwert verglichen - so kann eine Quest
	// nicht dadurch entwertet werden, dass jemand vor dem
	// Annehmen schon Rudel gelegt hat.
	wild_erlegt++;

	mapannounce .@map$,
	    "The pack breaks. " + strcharinfo(0) + " brought the leader down.",
	    bc_map, "0x66CC66";
	end;


// ------------------------------------------------------------
//  Ein Helfer faellt
// ------------------------------------------------------------
//  Er belohnt nichts eigenes - seine erhoehten Lebenspunkte
//  geben ueber die normale Erfahrung schon genug her. Der Zweck
//  dieses Labels ist ein anderer: Ohne ihn koennte der Tod eines
//  Helfers seinerseits ein Rudel ausloesen, und aus einem
//  Ereignis wuerde eine Kettenreaktion. Die Spielersperre
//  verhindert das.
OnWildHelfer:
	@wild_cd = gettimetick(2) + .cd;
	end;


// ------------------------------------------------------------
//  Aufraeumen
// ------------------------------------------------------------
//  Jede Minute. Zwei Aufgaben:
//
//  1. Plaetze freigeben, deren Tier es nicht mehr gibt. Das
//     faengt auch die Faelle ab, die kein Todesereignis
//     ausloesen - etwa @killall eines GM oder ein Rudel, das
//     ueber einen Kartenwechsel verschwindet. Ohne diese
//     Pruefung liefe der Zaehler voll und es gaebe nie wieder
//     ein Rudel.
//
//  2. Tiere entfernen, die zu lange stehen. Ein Rudel, das
//     niemand toetet, soll nicht bis zum naechsten Neustart auf
//     einer leeren Karte herumlaufen und einen der sechs
//     Plaetze belegen.
OnTimer60000:
	for (.@s = 0; .@s < .max; .@s++) {
		if (!$@wild_gid[.@s]) continue;

		if (!unitexists($@wild_gid[.@s])) {
			// Der Anfuehrer ist weg, ohne dass OnWildTot gelaufen
			// waere - etwa durch @killmonster, einen Kartenwechsel
			// oder weil ihn kein Spieler erlegt hat. Dann raeumt
			// das Todesereignis die Helfer NICHT ab, und das Rudel
			// bliebe ohne Kopf stehen. Also hier nachholen.
			killmonster $@wild_karte$[.@s], "safa_wildlife::OnWildHelfer";
			$@wild_gid[.@s]    = 0;
			$@wild_karte$[.@s] = "";
			$@wild_alter[.@s]  = 0;
			continue;
		}

		// Den Angriffsmodus jede Minute nachziehen. Wenn
		// irgendetwas im Server die Werte eines Monsters neu aus
		// der Datenbank rechnet, waere das Tier sonst still
		// wieder harmlos - und man saehe nur, dass es dasteht.
		getunitdata $@wild_gid[.@s], .@dt;
		setunitdata $@wild_gid[.@s], UMOB_MODE,
		            .@dt[UMOB_MODE]
		            | MD_CANMOVE | MD_CANATTACK | MD_AGGRESSIVE
		            | MD_ANGRY | MD_CHANGECHASE | MD_CHANGETARGETMELEE;

		$@wild_alter[.@s]++;
		if ($@wild_alter[.@s] < .leben) continue;

		.@m$ = $@wild_karte$[.@s];

		// killmonster statt unitkill: unitkill wuerde das
		// Todesereignis ausloesen, und das steht ohne
		// angehaengten Spieler da - getexp und strcharinfo
		// liefen dann ins Leere. Mit Label toetet killmonster
		// still.
		killmonster .@m$, "safa_wildlife::OnWildTot";
		killmonster .@m$, "safa_wildlife::OnWildHelfer";
		mapannounce .@m$,
		    "The pack moves on. Whatever it was looking for, it was not you.",
		    bc_map, "0x888888";

		$@wild_gid[.@s]    = 0;
		$@wild_karte$[.@s] = "";
		$@wild_alter[.@s]  = 0;
	}
	setnpctimer 0;
	end;


// ------------------------------------------------------------
//  Einstellungen und Tabelle
// ------------------------------------------------------------
OnInit:
	bindatcmd "wild", "safa_wildlife::OnGM", 99, 99;

	// --- Was sich lohnt zu drehen ----------------------------
	//
	//  ZUR CHANCE: SIE IST NICHT MEHR DER BEGRENZENDE WERT
	//
	//  Bei 500 von 10000 - fuenf Prozent je Kill - faellt der
	//  Wuerfel im Schnitt alle zwanzig Kills. Wer zwei Monster
	//  je Sekunde legt, haette damit alle zehn Sekunden ein
	//  Rudel.
	//
	//  Dass es nicht so kommt, liegt an den beiden Werten
	//  darunter: .cd sperrt jeden Spieler fuer eine Minute, und
	//  .max deckelt bei sechs Rudeln serverweit. Bei zehn bis
	//  zwoelf aktiven Leuten ist die Sechs damit praktisch immer
	//  ausgeschoepft.
	//
	//  Wer die gefuehlte Haeufigkeit aendern will, dreht deshalb
	//  ab hier an .cd und .max, nicht mehr an .chance.
	.chance     = 2000;  // von 10000 je Kill
	.champ_von  = 2603;   // Champion-Monster, siehe OnNPCKillEvent
	.champ_bis  = 2913;
	.max        = 6;     // gleichzeitige Rudel serverweit
	.cd         = 60;    // Sekunden Sperre je Spieler
	.min_lv     = 5;    // darunter kein Wildlife
	.leben      = 30;    // Minuten, dann loest sich das Rudel auf

	// --- Wie hart es zugeht ----------------------------------
	//  Diese sechs Zahlen sind die Schwierigkeit. Alles andere
	//  rechnet sich daraus und aus den Werten des Spielers, der
	//  das Rudel geweckt hat.
	//
	//  .schaden_pct  Anteil der Lebenspunkte des Ausloesers, auf den
	//                der Angriffswert des Anfuehrers gesetzt wird -
	//                VOR seiner Ruestung, vor Karten, Element und
	//                Rassenresistenz.
	//
	//                Was davon ankommt, haengt an DEF:
	//                  DEF 0    voller Wert
	//                  DEF 300  61 Prozent
	//                  DEF 900  38 Prozent
	//                Bei 6 landet ein Spieler mit DEF 300 also rund
	//                3,7 Prozent je Treffer.
	//
	//                DIE RUDELRECHNUNG IST DIE WICHTIGE
	//                Nicht der Anfuehrer allein zaehlt, sondern was
	//                je Angriffsrunde ankommt:
	//                    Anfuehrer + .helfer * .helfer_atk Prozent
	//                Vorher: 3 + 5*1,5 = 10,5 Prozent je Runde, und
	//                zwar unabhaengig von jeder Ruestung. Das war
	//                der Grund, warum ein Rudel niemanden stehen
	//                liess.
	//
	//  .treffer_mob  Trefferrate der Tiere gegen den Ausloeser in
	//                Prozent. Wird direkt in den HIT umgesetzt:
	//                HIT = Spieler-FLEE + dieser Wert. 95 laesst
	//                dem Ausweichen noch einen Rest, ohne dass
	//                das Rudel harmlos wird.
	//
	//                Hinweis: greifen mehr als drei Tiere
	//                denselben Spieler an, zieht rAthena dessen
	//                FLEE zusaetzlich um 10 Prozent je weiterem
	//                Angreifer ab (agi_penalty_type 1, count 3,
	//                num 10 in conf/battle/battle.conf). Ein
	//                volles Rudel trifft also noch etwas oefter
	//                als hier eingestellt. Das ist gewollt -
	//                umzingelt zu werden soll wehtun.
	//
	//  .treffer_spieler  Gegenstueck: so viel Prozent seiner
	//                eigenen Schlaege soll der Ausloeser landen.
	//                Deckelt das FLEE der Tiere nach oben.
	//
	//  .flee_wert    Ausweichen der Tiere, solange die Deckelung
	//                darueber es zulaesst. Hier haengt der
	//                Soeldner: sein HIT ist durch das DB-Level
	//                bei rund 324 gedeckelt (status.cpp:5037),
	//                bei 250 landet er also je nach Typ zwischen
	//                der 5-Prozent-Untergrenze und etwa 75
	//                Prozent. Er hilft noch, traegt den Kampf
	//                aber nicht mehr allein.
	//
	//  .hp_spieler   Untergrenze der Lebenspunkte des Anfuehrers,
	//                als Vielfaches der Lebenspunkte des
	//                Ausloesers. Verhindert den Fall "Poring mit
	//                2000 HP, vom Soeldner im Vorbeigehen
	//                erledigt".
	.schaden_pct      = 4;
	.treffer_mob      = 90;
	.treffer_spieler  = 95;
	.flee_wert        = 250;
	.hp_spieler       = 8;
	.hp_deckel        = 50000000;

	//  --- Schlagtempo, Untergrenze in Millisekunden -----------
	//  Siehe die Begruendung in S_Rudel. Kurz: das Skript macht
	//  kein Tier schneller, aber sechs gleichzeitig angreifende
