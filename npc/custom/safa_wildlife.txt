//===== SafaRO Script ========================================
//= Wildlife - die Rudel der Episode 22
//===== Erzeugt von ==========================================
//= tools/wildlife/bauen.py  --  NICHT von Hand pflegen.
//= Aenderungen gehoeren ins Skript, dann neu laufen lassen.
//===========================================================
//
//  WAS PASSIERT
//
//  Wer draussen ein Monster erlegt, hat eine kleine Chance,
//  dass an derselben Stelle die WILDLIFE-Fassung desselben
//  Monsters aufsteht: gleicher Sprite, gleicher Name mit dem
//  Vorsatz "Wildlife", vierzigfache Lebenspunkte, dreifacher
//  Angriff, aggressiv - und ein Rudel aus fuenf weiteren Arten
//  DERSELBEN KARTE im Ruecken.
//
//  Aus einem Poring wird ein Wildlife Poring, und mit ihm
//  stehen Wildlife Drops, Wildlife Poporing und was sonst noch
//  auf dieser Karte lebt. Das Rudel ist also ortstypisch und
//  nicht ausgewuerfelt.
//
//  WARUM DAS DEN SERVER SCHWERER MACHT
//
//  Nicht ueber Zahlen, sondern ueber Ueberraschung. Bisher war
//  jede Karte berechenbar: man kennt den Besatz, man kennt den
//  Schaden, man laesst @autopot laufen. Ein Rudel, das aus
//  einem gewoehnlichen Kill entsteht und sofort angreift,
//  unterbricht genau diese Routine - und zwar dort, wo der
//  Spieler ohnehin steht, statt in einem Gebiet, das er
//  betreten muss.
//
//  WARUM ES KEINE EIGENEN MONSTER-IDS GIBT
//
//  Der Client sucht das Sprite ueber die Monsternummer, in
//  npcidentity.lub und jobname.lub aus data.grf. Eine erfundene
//  Nummer bliebe unsichtbar - dieselbe Falle, die schon im Kopf
//  von db/import/mob_db.yml beim Golden Poring beschrieben ist.
//
//  Wildlife benutzt deshalb den amtlichen Monstertyp und wertet
//  ihn beim Erscheinen ueber setunitdata auf. Kein Eintrag in
//  mob_db, keiner in mob_avail, kein Client-Update - und es
//  wirkt von selbst auf jedes Monster, das rAthena je
//  hinzufuegt.
//
//  WAS AN DIESER DATEI ERZEUGT IST
//
//  Nur die Tabelle unten: Karte -> Arten. Sie laesst sich zur
//  Laufzeit nicht ermitteln, weil Spawns in rAthena NPC-Objekte
//  sind und keine abfragbaren Daten. Alles andere - Werte,
//  Rasse, Erfahrung, Ausschluss von MVPs - holt das Skript ueber
//  getmonsterinfo und bleibt dadurch aktuell.
//
//  MVPS UND MINIBOSSE SIND AUSGENOMMEN
//
//  Zweimal: beim Erzeugen der Tabelle (Class: Boss und
//  Modes.Mvp aus mob_db.yml) und noch einmal zur Laufzeit ueber
//  MD_MVP und MD_STATUSIMMUNE. Doppelt, weil die Tabelle nur die
//  RUDELMITGLIEDER filtert - der Ringleader entsteht aus dem
//  Monster, das der Spieler gerade erlegt hat, und das steht in
//  keiner Tabelle.
//
//===========================================================


-	script	safa_wildlife	-1,{
	end;


// ------------------------------------------------------------
//  Der Ausloeser
// ------------------------------------------------------------
//  Dieses Ereignis feuert bei JEDEM Kill auf dem ganzen Server.
//  Die Reihenfolge der Pruefungen ist deshalb kein Zufall,
//  sondern nach Kosten sortiert: erst Zahlenvergleiche auf
//  Variablen, die schon im Speicher liegen, dann der Wuerfel,
//  und erst ganz zuletzt die Datenbankzugriffe ueber
//  getmonsterinfo und die Kartensuche.
//
//  Bei fuenf Prozent laeuft alles ab dem Wuerfel nur noch bei
//  jedem zwanzigsten Kill - und selbst davon faellt das meiste
//  gleich wieder an der Spielersperre und am Rudeldeckel weg.
OnNPCKillEvent:
	if (killedrid == 0) end;
	if ($@wild_aktiv >= .max) end;
	if (gettimetick(2) < @wild_cd) end;
	if (rand(10000) >= .chance) end;

	// getmonsterinfo liefert -1 bei unbekannter Klasse. Ohne
	// diese Pruefung ginge -1 durch und jeder Kill zaehlte -
	// derselbe Fehler, der bei den MVP-Muenzen vermieden wurde.
	.@mode = getmonsterinfo(killedrid, MOB_MODE);
	if (.@mode == -1) end;

	// MD_MVP sind die MVPs, MD_STATUSIMMUNE setzt rAthena bei
	// Class: Boss - also den Minibossen. Aus keinem von beiden
	// wird ein Wildlife.
	if (.@mode & MD_MVP) end;
	if (.@mode & MD_STATUSIMMUNE) end;

	if (getmonsterinfo(killedrid, MOB_LV) < .min_lv) end;

	// Champion-Monster als Vorlage ergaeben einen MENSCHEN: der
	// Client kennt diese von rAthena erfundenen Nummern nicht und
	// zeichnet ersatzweise ein Novizensprite. Aus der Tabelle sind
	// sie gefiltert - aber der Ausloeser hier ist das ERLEGTE
	// Monster, und Champions laufen ueberall herum
	// (npc/re/mobs/championmobs.txt).
	//
	// Die Spanne ist lueckenlos und aus db/re/mob_db.yml gezaehlt,
	// nicht geschaetzt: 311 Monster mit AegisName C1_ bis C5_,
	// Nummern 2603 bis 2913.
	if (killedrid >= .champ_von && killedrid <= .champ_bis) end;

	getmapxy(.@map$, .@x, .@y, BL_PC);
	.@i = inarray(.karte$[0], .@map$);
	if (.@i < 0) end;

	// Die Sperre gilt dem SPIELER, nicht dem Server. Sonst
	// koennte einer mit hoher Tempo-Ausruestung alle Rudel des
	// Servers auf sich ziehen.
	@wild_cd = gettimetick(2) + .cd;

	callsub S_Rudel, .@map$, .@x, .@y, killedrid, .@i;
	end;


// ------------------------------------------------------------
//  @wild - nachsehen, ob das hier ueberhaupt greift
// ------------------------------------------------------------
//  Ohne diesen Befehl bleibt nur: Monster kloppen und hoffen.
//  Und wenn dann nichts kommt, weiss man nicht, woran es lag -
//  an der Karte, am Deckel, an der Sperre, am Wuerfel oder
//  daran, dass die Datei gar nicht geladen ist.
//
//    @wild         Status: was gilt hier gerade
//    @wild los     ein Rudel erzwingen, ohne Wuerfel und Sperre
//    @wild aus     alle Rudel entfernen, ohne Belohnung
//    @wild karte   welche Arten diese Karte beisteuert
//
//  "los" geht bewusst am Deckel und an der Sperre vorbei. Der
//  Zweck ist zu sehen, ob die KETTE funktioniert - Tabelle,
//  Spawn, Werteaufwertung, Helfer, Ansage, Tod, Belohnung. Ob
//  die Wahrscheinlichkeit richtig eingestellt ist, ist eine
//  andere Frage und steht im Status.
OnGM:
	.@was$ = "";
	if (.@atcmd_numparameters > 0)
		.@was$ = .@atcmd_parameters$[0];

	getmapxy(.@map$, .@x, .@y, BL_PC);
	.@i = inarray(.karte$[0], .@map$);

	// --- aufraeumen ------------------------------------------
	if (.@was$ == "aus") {
		for (.@k = 0; .@k < .max; .@k++) {
			if (!$@wild_gid[.@k]) continue;
			killmonster $@wild_karte$[.@k], "safa_wildlife::OnWildTot";
			killmonster $@wild_karte$[.@k], "safa_wildlife::OnWildHelfer";
			$@wild_gid[.@k] = 0;
			$@wild_karte$[.@k] = "";
			$@wild_alter[.@k] = 0;
		}
		$@wild_aktiv = 0;
		dispbottom "[Wildlife] Alle Rudel entfernt, Zaehler auf 0.";
		end;
	}

	// --- Arten dieser Karte ----------------------------------
	if (.@was$ == "karte") {
		if (.@i < 0) {
			dispbottom "[Wildlife] " + .@map$
			           + " steht NICHT in der Tabelle. Hier kommt nie eines.";
			end;
		}
		dispbottom "[Wildlife] " + .@map$ + " steuert "
		           + .anz[.@i] + " Arten bei:";
		for (.@k = 0; .@k < .anz[.@i]; .@k++)
			dispbottom "   " + .art[.von[.@i] + .@k] + "  "
			           + getmonsterinfo(.art[.von[.@i] + .@k], MOB_NAME);
		end;
	}

	// --- erzwingen -------------------------------------------
	if (.@was$ == "los") {
		if (.@i < 0) {
			dispbottom "[Wildlife] " + .@map$
			           + " steht nicht in der Tabelle - hier geht es nicht.";
			end;
		}
		// Vorlage aus dem Bestand dieser Karte. Beim normalen
		// Weg ist es das gerade erlegte Monster; hier gibt es
		// keines, also wird eines der ortsueblichen gezogen.
		.@basis = .art[.von[.@i] + rand(.anz[.@i])];

		// S_Rudel steigt aus, wenn kein Platz frei ist. Das
		// merkt man sonst nicht - es passiert schlicht nichts.
		.@vorher = $@wild_aktiv;
		callsub S_Rudel, .@map$, .@x, .@y, .@basis, .@i;

		if ($@wild_aktiv == .@vorher) {
			dispbottom "[Wildlife] Nichts passiert. Alle " + .max
			           + " Plaetze belegt - erst \"@wild aus\".";
			end;
		}
		dispbottom "[Wildlife] Erzwungen: "
		           + getmonsterinfo(.@basis, MOB_NAME)
		           + ". Aktive Rudel jetzt " + $@wild_aktiv + ".";
		end;
	}

	// --- Status ----------------------------------------------
	dispbottom "[Wildlife] Skript ist geladen.";
	dispbottom "  Karte " + .@map$ + ": "
	           + (.@i < 0 ? "NICHT in der Tabelle - hier kommt nie eines"
	                      : "in der Tabelle, " + .anz[.@i] + " Arten");
	dispbottom "  Tabelle: " + getarraysize(.karte$) + " Karten, "
	           + getarraysize(.art) + " Arteneintraege";
	// Das doppelte Prozentzeichen ist kein Tippfehler: der
	// Rumpf dieser Datei geht durch Pythons %-Formatierung, und
	// ein einzelnes % waere dort eine Formatangabe. Im
	// erzeugten Skript steht wieder eines.
	dispbottom "  Chance: " + .chance + " von 10000 ("
	           + (.chance / 100) + "," + (.chance % 100)
	           + " Prozent) je Kill";
	dispbottom "  Rudel aktiv: " + $@wild_aktiv + " von " + .max
	           + (($@wild_aktiv >= .max) ? "  <-- DECKEL ERREICHT, nichts Neues" : "");

	.@rest = @wild_cd - gettimetick(2);
	dispbottom "  Deine Sperre: "
	           + ((.@rest > 0) ? .@rest + " Sekunden <-- laeuft noch"
	                           : "frei");
	dispbottom "  Mindestlevel der Vorlage: " + .min_lv
	           + ", Rudel loest sich nach " + .leben + " Minuten auf";

	for (.@k = 0; .@k < .max; .@k++) {
		if (!$@wild_gid[.@k]) continue;
		dispbottom "   Platz " + .@k + ": " + $@wild_karte$[.@k]
		           + ", " + $@wild_alter[.@k] + " Minuten"
		           + (unitexists($@wild_gid[.@k]) ? "" : "  <-- Tier weg, Platz wird beim naechsten Takt frei");
	}
	end;


// ------------------------------------------------------------
//  Das Rudel aufstellen
// ------------------------------------------------------------
S_Rudel:
	.@map$   = getarg(0);
	.@x      = getarg(1);
	.@y      = getarg(2);
	.@basis  = getarg(3);
	.@i      = getarg(4);

	// --- Erst den Platz, dann das Tier --------------------------
	//  Die Reihenfolge ist wichtig. Umgekehrt stuende bei einem
	//  vollen Buch bereits ein Rudel auf der Karte, das in keiner
	//  Liste gefuehrt wird - der Aufraeumtimer kennt es nicht und
	//  es bliebe bis zum Neustart stehen.
	//
	//  WARUM EINE SCHLEIFE UND KEIN inarray
	//
	//  Weil rAthena ein Feldelement LOESCHT, sobald man ihm
	//  seinen Standardwert zuweist - Skriptfelder sind intern
	//  eine Zuordnung Index->Wert, kein Block. Ein freier Platz
	//  (Wert 0) ist danach nicht "0", sondern gar nicht da, und
	//  getarraysize endet beim letzten belegten Platz.
	//
	//  Steht also nur auf Platz 0 ein Rudel und die uebrigen
	//  fuenf sind frei, hat $@wild_gid die Laenge 1. inarray
	//  suchte dann NUR in diesem einen Platz, faende die 0 nicht
	//  und lieferte -1 - und ein Rueckfall auf Platz 0 haette das
	//  dort eingetragene Rudel aus der Liste geworfen. Es waere
	//  fuer immer stehengeblieben.
	//
	//  Die Schleife liest ueber das Feldende hinaus, und das ist
	//  hier genau richtig: ein nicht vorhandenes Element liefert
	//  0, also "frei".
	.@s = -1;
	for (.@k = 0; .@k < .max; .@k++) {
		if ($@wild_gid[.@k] == 0) {
			.@s = .@k;
			break;
		}
	}
	if (.@s < 0) return;

	.@name$ = "Wildlife " + getmonsterinfo(.@basis, MOB_NAME);

	monster .@map$, .@x, .@y, .@name$, .@basis, 1,
	        "safa_wildlife::OnWildTot";
	.@gid = $@mobid[0];
	if (!.@gid) return;

	// --- Werte anheben ---------------------------------------
	//  Der HP-Faktor ist absichtlich degressiv. Vierzigfach ist
	//  bei einem Poring (50 HP) eine Ansage und bei einem
	//  Monster mit fuenf Millionen HP eine Wand, die niemand
	//  umwirft, bevor der Aufraeumtimer greift. Ein Rudel, das
	//  man nicht toeten KANN, ist keine Schwierigkeit, sondern
	//  eine Sperre.
	.@hp = getunitdata(.@gid, UMOB_MAXHP);
	.@f = .hp_faktor;
	if (.@hp > 1000000)     .@f = .hp_faktor / 5;
	else if (.@hp > 100000) .@f = .hp_faktor / 2;
	.@hp = .@hp * .@f;

	setunitdata .@gid, UMOB_MAXHP, .@hp;
	setunitdata .@gid, UMOB_HP,    .@hp;
	setunitdata .@gid, UMOB_ATKMIN,
	            getunitdata(.@gid, UMOB_ATKMIN) * .atk_faktor;
	setunitdata .@gid, UMOB_ATKMAX,
	            getunitdata(.@gid, UMOB_ATKMAX) * .atk_faktor;
	setunitdata .@gid, UMOB_DEF,
	            getunitdata(.@gid, UMOB_DEF) + .def_plus;
	setunitdata .@gid, UMOB_MDEF,
	            getunitdata(.@gid, UMOB_MDEF) + .def_plus;
	setunitdata .@gid, UMOB_LEVEL,
	            getunitdata(.@gid, UMOB_LEVEL) + .lv_plus;

	// Aggressiv UND beweglich. Ohne MD_CANMOVE stuende ein
	// Wildlife Poring aggressiv, aber angewurzelt da - und waere
	// aus zwei Feldern Abstand kostenlos zu erschiessen.
	setunitdata .@gid, UMOB_MODE,
	            getunitdata(.@gid, UMOB_MODE) | MD_AGGRESSIVE | MD_CANMOVE;

	// --- Das Rudel -------------------------------------------
	//  Die Helfer bekommen denselben Vorsatz. Das ist Absicht:
	//  wer "Wildlife" liest, weiss, dass dieses Tier nicht zum
	//  festen Besatz gehoert, gleich wieder verschwindet und
	//  haerter zuschlaegt als sein Namensvetter daneben.
	.@a = .von[.@i];
	.@n = .anz[.@i];
	for (.@k = 0; .@k < .helfer; .@k++) {
		.@art = .art[.@a + rand(.@n)];

		monster .@map$, .@x, .@y,
		        "Wildlife " + getmonsterinfo(.@art, MOB_NAME),
		        .@art, 1, "safa_wildlife::OnWildHelfer";
		.@sg = $@mobid[0];
		if (!.@sg) continue;

		.@shp = getunitdata(.@sg, UMOB_MAXHP) * .helfer_hp;
		setunitdata .@sg, UMOB_MAXHP, .@shp;
		setunitdata .@sg, UMOB_HP,    .@shp;
		setunitdata .@sg, UMOB_MODE,
		            getunitdata(.@sg, UMOB_MODE) | MD_AGGRESSIVE | MD_CANMOVE;
	}

	// --- Buchfuehrung ----------------------------------------
	//  $@ ist serverweit und NICHT dauerhaft - genau richtig:
	//  ueber 'monster' gesetzte Tiere ueberleben einen Neustart
	//  ohnehin nicht, und eine dauerhafte Liste zeigte danach
	//  auf Nummern, die es nicht mehr gibt.
	//
	//  Der Platz .@s steht schon fest, siehe oben.
	$@wild_gid[.@s]    = .@gid;
	$@wild_karte$[.@s] = .@map$;
	$@wild_alter[.@s]  = 0;
	$@wild_aktiv++;

	mapannounce .@map$,
	    "The ground shifts. A " + .@name$ + " has come up, and it is not alone.",
	    bc_map, "0xCC6600";
	return;


// ------------------------------------------------------------
//  Der Ringleader faellt
// ------------------------------------------------------------
OnWildTot:
	.@basis = killedrid;
	getmapxy(.@map$, .@x, .@y, BL_PC);

	// killedgid ist die Spielnummer des gefallenen Tieres, nicht
	// seine Klasse. Nur damit laesst sich der richtige Platz in
	// der Liste freigeben - killedrid waere bei zwei Rudeln
	// derselben Art mehrdeutig.
	.@s = inarray($@wild_gid[0], killedgid);
	if (.@s >= 0) {
		$@wild_gid[.@s]    = 0;
		$@wild_karte$[.@s] = "";
		$@wild_alter[.@s]  = 0;
		if ($@wild_aktiv > 0) $@wild_aktiv--;
	}

	// Ohne den Anfuehrer loest sich das Rudel auf. killmonster
	// mit Label loest das Todesereignis NICHT aus - die Helfer
	// verschwinden also, ohne noch einmal zu belohnen.
	killmonster .@map$, "safa_wildlife::OnWildHelfer";

	// --- Belohnung -------------------------------------------
	//  Der Faktor ist woertlich "so viele gewoehnliche Kills".
	//  getexp laeuft ueber quest_exp_rate, und die steht bei uns
	//  auf denselben 2500 wie base_exp_rate
	//  (conf/import/battle_conf.txt) - beide Seiten sind also
	//  gleich hochgerechnet und der Vergleich stimmt.
	//
	//  Fuenfzig passt zum Aufwand: vierzigfache Lebenspunkte
	//  plus fuenf Helfer mit fuenffachen sind rund fuenfundsechzig
	//  gewoehnliche Tiere an Arbeit.
	getexp getmonsterinfo(.@basis, MOB_BASEEXP) * .exp_faktor,
	       getmonsterinfo(.@basis, MOB_JOBEXP) * .exp_faktor;

	// Monster koennen in rAthena kein Zeny fallen lassen - es
	// gibt schlicht kein Feld dafuer in mob_db. Deshalb hier.
	Zeny += getmonsterinfo(.@basis, MOB_LV) * .zeny_je_lv;

	// --- Und die Beute ---------------------------------------
	//  Trophaeen nur von Brute-Tieren, und das ist kein
	//  Versehen. Die Safari-Trophaee ist in
	//  npc/custom/safa_safari.txt an Brute ab Stufe 150
	//  gebunden, weil "Wild Safari" sonst ein leeres Wort waere.
	//  Wildlife haelt sich daran: ein Wildlife Zombie ist
	//  gefaehrlich, aber er ist kein Tier.
	//
	//  Alle uebrigen zahlen in Destiny Coins. Die sind auf
	//  diesem Server die zweite Waehrung und teuer genug
	//  (der Levelraum kostet 1.000), dass sich der Kampf auch
	//  ausserhalb der Safarigebiete lohnt.
	if (getmonsterinfo(.@basis, MOB_RACE) == RC_BRUTE)
		callfunc "F_SafaBeute", .mat, .troph;
	else
		callfunc "F_SafaBeute", .coin, .muenzen;

	// Fuer die Questreihe in npc/custom/safa_quests.txt. Der
	// Zaehler laeuft lebenslang weiter und wird von den Quests
	// nur mit einem Startwert verglichen - so kann eine Quest
	// nicht dadurch entwertet werden, dass jemand vor dem
	// Annehmen schon Rudel gelegt hat.
	wild_erlegt++;

	mapannounce .@map$,
	    "The pack breaks. " + strcharinfo(0) + " brought the leader down.",
	    bc_map, "0x66CC66";
	end;


// ------------------------------------------------------------
//  Ein Helfer faellt
// ------------------------------------------------------------
//  Er belohnt nichts eigenes - seine erhoehten Lebenspunkte
//  geben ueber die normale Erfahrung schon genug her. Der Zweck
//  dieses Labels ist ein anderer: Ohne ihn koennte der Tod eines
//  Helfers seinerseits ein Rudel ausloesen, und aus einem
//  Ereignis wuerde eine Kettenreaktion. Die Spielersperre
//  verhindert das.
OnWildHelfer:
	@wild_cd = gettimetick(2) + .cd;
	end;


// ------------------------------------------------------------
//  Aufraeumen
// ------------------------------------------------------------
//  Jede Minute. Zwei Aufgaben:
//
//  1. Plaetze freigeben, deren Tier es nicht mehr gibt. Das
//     faengt auch die Faelle ab, die kein Todesereignis
//     ausloesen - etwa @killall eines GM oder ein Rudel, das
//     ueber einen Kartenwechsel verschwindet. Ohne diese
//     Pruefung liefe der Zaehler voll und es gaebe nie wieder
//     ein Rudel.
//
//  2. Tiere entfernen, die zu lange stehen. Ein Rudel, das
//     niemand toetet, soll nicht bis zum naechsten Neustart auf
//     einer leeren Karte herumlaufen und einen der sechs
//     Plaetze belegen.
OnTimer60000:
	for (.@s = 0; .@s < .max; .@s++) {
		if (!$@wild_gid[.@s]) continue;

		if (!unitexists($@wild_gid[.@s])) {
			// Der Anfuehrer ist weg, ohne dass OnWildTot gelaufen
			// waere - etwa durch @killmonster, einen Kartenwechsel
			// oder weil ihn kein Spieler erlegt hat. Dann raeumt
			// das Todesereignis die Helfer NICHT ab, und das Rudel
			// bliebe ohne Kopf stehen. Also hier nachholen.
			killmonster $@wild_karte$[.@s], "safa_wildlife::OnWildHelfer";
			$@wild_gid[.@s]    = 0;
			$@wild_karte$[.@s] = "";
			$@wild_alter[.@s]  = 0;
			if ($@wild_aktiv > 0) $@wild_aktiv--;
			continue;
		}

		$@wild_alter[.@s]++;
		if ($@wild_alter[.@s] < .leben) continue;

		.@m$ = $@wild_karte$[.@s];

		// killmonster statt unitkill: unitkill wuerde das
		// Todesereignis ausloesen, und das steht ohne
		// angehaengten Spieler da - getexp und strcharinfo
		// liefen dann ins Leere. Mit Label toetet killmonster
		// still.
		killmonster .@m$, "safa_wildlife::OnWildTot";
		killmonster .@m$, "safa_wildlife::OnWildHelfer";
		mapannounce .@m$,
		    "The pack moves on. Whatever it was looking for, it was not you.",
		    bc_map, "0x888888";

		$@wild_gid[.@s]    = 0;
		$@wild_karte$[.@s] = "";
		$@wild_alter[.@s]  = 0;
		if ($@wild_aktiv > 0) $@wild_aktiv--;
	}
	setnpctimer 0;
	end;


// ------------------------------------------------------------
//  Einstellungen und Tabelle
// ------------------------------------------------------------
OnInit:
	bindatcmd "wild", "safa_wildlife::OnGM", 99, 99;

	// --- Was sich lohnt zu drehen ----------------------------
	//
	//  ZUR CHANCE: SIE IST NICHT MEHR DER BEGRENZENDE WERT
	//
	//  Bei 500 von 10000 - fuenf Prozent je Kill - faellt der
	//  Wuerfel im Schnitt alle zwanzig Kills. Wer zwei Monster
	//  je Sekunde legt, haette damit alle zehn Sekunden ein
	//  Rudel.
	//
