//===== SafaRO Script =====================================
//= Isekai-System: Tiers, Union-Punkte, Todes-Debuff
//===== Beschreibung: ========================================
//= Neun Tiers von F bis SSS.
//=   F, E    allgemeine Boni, E ist die staerkere Fassung
//=   D, C    die sechs 1st Classes
//=   B, A    die zwoelf 2nd Classes - jeder Zweig einzeln
//=   S, SS   die zwoelf 3rd Classes
//=   SSS     die zwoelf 4th Classes
//=
//= Knight und Crusader haben getrennte Geschenke, ebenso Wizard
//= und Sage, Priest und Monk, Blacksmith und Alchemist und so
//= weiter. Jedes bringt den Waffenschaden des typischen
//= Waffentyps, die Signature-Skills der Klasse und Statuswerte.
//=
//= UNION-PUNKTE werden nicht gespeichert, sondern live gezaehlt:
//= 100 je existierendem Charakter des Accounts auf Level 500.
//=
//= Geschenk 510 (Seeker) wirkt zusaetzlich in src/map/status.cpp.
//=
//= IDs sind Tier*100 + Position. Die 100er-Schritte sind noetig,
//= weil ein Tier zwoelf Geschenke hat - mit Tier*10 waeren sich
//= Tier B Position 11 und Tier A Position 1 in die Quere gekommen.
//============================================================

// ------------------------------------------------------------
//  Union-Punkte
// ------------------------------------------------------------
//  Wird bei jedem Aufruf frisch gezaehlt statt gespeichert.
//  Damit verschwinden die Punkte eines Charakters in dem Moment,
//  in dem er geloescht wird - Hochleveln, loeschen, Punkte
//  behalten funktioniert also nicht.
// ------------------------------------------------------------
function	script	F_SafaPoints	{
	// Testwert eines GM. Nur ueber den Threshold Warden setzbar,
	// dort hinter einer Gruppenpruefung. 0 = aus, dann zaehlt
	// wieder die Datenbank.
	if (#isekai_debug_points > 0)
		return #isekai_debug_points;

	.@rows = query_sql("SELECT COUNT(*) FROM `char` WHERE `account_id` = " + getcharid(3) + " AND `base_level` >= 500", .@cnt);
	if (.@rows < 1)
		return 0;
	return .@cnt * 100;
}

function	script	F_SafaHighestTier	{
	.@p = callfunc("F_SafaPoints");
	.@best = 0;
	for (.@i = 0; .@i < 9; .@i++) {
		if (.@p >= callfunc("F_SafaTierReq", .@i))
			.@best = .@i;
	}
	return .@best;
}

//
//  bonus_script rechnet die Sekunden in Millisekunden um. Serverseitig
//  passen davon knapp 4,3 Milliarden in ein uint32 - der Buff selbst
//  koennte also fast 50 Tage laufen. Zum Client geht die Restzeit aber
//  ueber clif_status_change, und dort landet sie in einem Feld, das der
//  Client VORZEICHENBEHAFTET liest.
//
//  Alles ueber 2.147.483.647 ms kippt damit ins Negative. 4294967 s
//  wurden auf der Leitung zu -296 ms, der Client hielt den Buff fuer
//  laengst abgelaufen und hat das Symbol gar nicht erst gezeichnet -
//  die Boni wirkten, sichtbar war nichts.
//
//  Deshalb: unter 2147483 Sekunden bleiben. Die tatsaechliche Laufzeit
//  spielt ohnehin keine Rolle, weil stuendlich und bei jedem Login neu
//  gesetzt wird (siehe OnTimer weiter unten). haveTimeLimit = 0 in
//  stateiconinfo.lub blendet den Countdown im Client aus.
function	script	F_SafaRefresh	{
	bonus_script_clear 1;

	// Selbstheilung bei veralteten Nummern.
	// Die Geschenk-IDs haben sich waehrend der Entwicklung zweimal
	// geaendert. Ein Charakter, der noch eine alte Nummer traegt,
	// bekaeme von F_SafaGift einen leeren String zurueck - und
	// bonus_script lehnt leere Skripte ab ("No Script string, No
	// Duration"). Der Buff waere also stillschweigend weg, ohne dass
	// im Spiel irgendetwas darauf hinweist.
	if (isekai_gift > 0 && callfunc("F_SafaGift", isekai_gift) == "") {
		dispbottom "[System] Your gift no longer exists in this world. Speak to the Threshold Warden again.";
		set isekai_gift, 0;
	}

	if (isekai_gift > 0)
		bonus_script callfunc("F_SafaGift", isekai_gift), 2000000, 512, 0, EFST_KINGS_GRACE;

	.@pen = death_count * 5;
	if (.@pen > 90)
		.@pen = 90;

	if (.@pen > 0) {
		bonus_script "bonus bAtkRate,-" + .@pen + "; " +
		             "bonus bMatkRate,-" + .@pen + "; " +
		             "bonus bMaxHPrate,-" + .@pen + "; " +
		             "bonus bMaxSPrate,-" + .@pen + "; " +
		             "bonus bDefRate,-" + .@pen + "; " +
		             "bonus bMdefRate,-" + .@pen + "; " +
		             "bonus bFleeRate,-" + .@pen + "; " +
		             "bonus bHitRate,-" + .@pen + ";", 2000000, 512, 0, EFST_ALL_STAT_DOWN;
	}
	return;
}


// ------------------------------------------------------------
//  Ereignisse
// ------------------------------------------------------------
-	script	#safa_core	-1,{
	end;

OnInit:
	initnpctimer;
	end;

// Stuendlich alle eingeloggten Spieler auffrischen. addrid(0) haengt
// jeden Online-Charakter an dieses Skript, F_SafaRefresh setzt
// Geschenk und Debuff neu. Damit kann der Buff auch bei einer
// Dauersitzung nicht ablaufen.
OnTimer3600000:
	addrid(0);
	callfunc "F_SafaRefresh";
	end;

OnTimer3600100:
	initnpctimer;
	end;

OnPCLoginEvent:
	callfunc "F_SafaRefresh";
	if (death_count > 0) {
		.@pen = death_count * 5;
		if (.@pen > 90) .@pen = 90;
		dispbottom "[System] Deaths: " + death_count + ". All combat values reduced by " + .@pen + "%.";
	}
	if (isekai_gift == 0)
		dispbottom "[System] You carry no gift. The Threshold Warden binds it.";
	else
		dispbottom "[System] Gift: " + callfunc("F_SafaGiftName", isekai_gift) + " (Tier " + callfunc("F_SafaTierName", callfunc("F_SafaGiftTier", isekai_gift)) + ")";
	dispbottom "[Union] " + callfunc("F_SafaPoints") + " points - tier " + callfunc("F_SafaTierName", callfunc("F_SafaHighestTier")) + " unlocked.";
	end;

// Nur eine Meldung. Die Punkte selbst werden nicht gutgeschrieben,
// sondern immer live gezaehlt.
OnPCBaseLvUpEvent:
	if (BaseLevel < 500) end;
	if (isekai_maxed) end;
	set isekai_maxed, 1;
	specialeffect2 EF_ANGEL2;
	announce strcharinfo(0) + " has reached level 500. The Union grows stronger.", bc_all;
	dispbottom "[Union] This soul now carries 100 points for your line - as long as it exists.";
	end;

OnPCDieEvent:
	.@map$ = strcharinfo(3);

	if (getmapflag(.@map$, mf_pvp) || getmapflag(.@map$, mf_gvg) ||
	    getmapflag(.@map$, mf_gvg_castle) || getmapflag(.@map$, mf_gvg_dungeon) ||
	    getmapflag(.@map$, mf_battleground))
		end;

	set death_count, death_count + 1;
	callfunc "F_SafaRefresh";

	.@pen = death_count * 5;
	if (.@pen > 90) .@pen = 90;
	dispbottom "[System] Death " + death_count + ". All combat values now reduced by " + .@pen + "%.";
	end;
}
