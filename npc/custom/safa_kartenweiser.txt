//===== SafaRO Script ========================================
//= Der Kartenweise (Card Sage)
//===== Beschreibung: ========================================
//= Verdichtet drei gleiche Monstercards zu einer staerkeren
//= Fassung. Dialog ENGLISCH (Hausregel fuer NPCs), Kommentare
//= deutsch. Regeln:
//=
//=     3x Grundkarte  -> Stufe 2 (ein + am Namen), Wirkung x2
//=     3x Stufe 2     -> Stufe 3 (++),             Wirkung x4
//=     MVP-Karten     -> x3 je Stufe (Stufe 3 also x9)
//=     Skill-Karten   -> Skillstufe +1 je Stufe (nicht x2 -
//=                       skill SETZT die Stufe, sie summiert
//=                       sich nicht; Deckel ist das MaxLevel)
//=     Setzer-Boni    -> Elementruestung: je Stufe SubEle 15%
//=                       gegen das Konter-Element; sonstige
//=                       (Sicht, Unzerbrechlich ...): MaxHP
//=                       und MaxSP +3% je Stufe
//=
//= Die verstaerkten Items, die Zuordnungs-Arrays und alle
//= Client-Eintraege erzeugt tools/kartenweiser/bauen.py.
//= Die Arrays liest safa_kartenweiser_daten.txt beim Start:
//=
//=     $@KW_BASIS[i]    Karte, die verlangt wird (3 Stueck)
//=     $@KW_ZIEL[i]     Karte, die entsteht (1 Stueck)
//=     $@KW_GEBUEHR[i]  Zeny-Gebuehr des Tauschs
//=     $@KW_STUFE[i]    entstehende Stufe (2 oder 3)
//=     $@KW_MVP[i]      1 = MVP-Karte
//=
//= Die Gebuehren stehen deshalb NICHT hier, sondern im
//= Werkzeug (GEBUEHR in bauen.py) - sie sind je Tauschweg
//= vorberechnet, damit dieser NPC keine eigene Preistabelle
//= pflegen muss.
//=
//= WARUM KEIN ZUFALL: wie beim Card Remover (safa_dienste) -
//= die Kosten sind die Kosten-Seite, und die ist deutlich.
//= Drei Karten plus Gebuehr sind kalkulierbar; ein Wuerfel
//= wuerde nicht Fehlentscheidungen bestrafen, sondern Pech.
//===== Ort: =================================================
//= lhz_in02 264,283 - BEWUSST NICHT im Veins-Quartier. Der
//= Kartenweise ist einer der Geheim-NPCs: er steht im
//= Obergeschoss der Lighthalzen-Herberge und wird nirgends
//= angekuendigt. Wer ihn findet, hat ihn. Naechster Nachbar
//= ist der Lab Staff auf 265,273; die Zelle ist frei.
//============================================================

lhz_in02,264,283,4	script	Card Sage#safa	4_M_SAGE_C,{
	disable_items;
	mes "[Card Sage]";
	mes "Three identical cards carry the same soul. I can condense them into a single one - stronger than the three were together.";
	next;
	switch (select("Condense cards:How does it work?:Never mind")) {
	case 2:
		mes "[Card Sage]";
		mes "Bring me ^0000CCthree identical monster cards^000000 and my fee, and you receive one card of the next tier:";
		mes " ";
		mes "^0000CCTier 2^000000 (a + on its name): twice the effect";
		mes "^0000CCTier 3^000000 (++): doubled again - fourfold";
		next;
		mes "[Card Sage]";
		mes "^CC0000MVP cards^000000 are woven more densely: they ^CC0000triple^000000 with every tier.";
		mes " ";
		mes "Cards that grant a ^0000CCskill^000000 raise it by ^0000CC+1 level^000000 per condensation instead - a skill level is set, it does not add up.";
		next;
		mes "[Card Sage]";
		mes "An ^0000CCarmor element^000000 cannot be doubled either - there is no more element than the element.";
		mes "Instead I cushion the ^0000CCweakness^000000 of the elemental armor: per tier, ^0000CC15%^000000 less damage from the counter element. Angeling, for instance, softens Shadow.";
		mes " ";
		mes "Other unique effects (sight, unbreakable and the like) I compensate with ^0000CCMaxHP and MaxSP +3%^000000 per tier.";
		next;
		mes "[Card Sage]";
		mes "The fee depends on tier and rarity. And two warnings:";
		mes " ";
		mes "^CC0000The WHOLE card condenses - its drawbacks included.^000000 Condense a card with a flaw, and the flaw multiplies with it.";
		mes " ";
		mes "A condensed card is an ^0000CCitem of its own^000000 - set effects with other cards (combos) only know the base card.";
		close;
	case 3:
		close;
	}

	// ---- Inventar gegen die Zuordnung halten -------------------
	getinventorylist;
	set .@n, 0;
	for (set .@i, 0; .@i < @inventorylist_count; set .@i, .@i + 1) {
		if (@inventorylist_amount[.@i] < 3) continue;
		if (@inventorylist_bound[.@i]) continue;
		set .@k, inarray($@KW_BASIS, @inventorylist_id[.@i]);
		if (.@k < 0) continue;
		set .@weg[.@n], .@k;
		set .@n, .@n + 1;
	}
	if (!.@n) {
		mes "[Card Sage]";
		mes "You carry no three identical monster cards.";
		mes "Return when three souls of the same name crowd your bag.";
		close;
	}

	// ---- Auswahl -----------------------------------------------
	set .@m$, "";
	for (set .@i, 0; .@i < .@n; set .@i, .@i + 1) {
		set .@k, .@weg[.@i];
		set .@m$, .@m$ + "3x " + getitemname($@KW_BASIS[.@k])
			+ " > " + getitemname($@KW_ZIEL[.@k])
			+ ($@KW_MVP[.@k] ? " [MVP]" : "") + ":";
	}
	mes "[Card Sage]";
	mes "These condensations I can offer you:";
	next;
	set .@k, .@weg[select(.@m$) - 1];

	// ---- Bestaetigung ------------------------------------------
	mes "[Card Sage]";
	mes "^0000CC3x " + getitemname($@KW_BASIS[.@k]) + "^000000";
	mes "will become";
	mes "^0000CC1x " + getitemname($@KW_ZIEL[.@k]) + "^000000";
	mes " ";
	mes "Fee: ^CC0000" + callfunc("F_InsertComma", $@KW_GEBUEHR[.@k]) + " Zeny^000000";
	if ($@KW_MVP[.@k]) {
		mes "^CC0000An MVP card - threefold condensation.^000000";
	}
	next;
	if (select("Condense:Rather not") == 2) {
		mes "[Card Sage]";
		mes "Wise restraint is an answer, too.";
		close;
	}

	// ---- Pruefen und tauschen ----------------------------------
	// Erst jetzt und noch einmal: zwischen Menue und Klick kann
	// nichts verkauft werden (disable_items), aber Vorsicht
	// kostet nichts.
	if (countitem($@KW_BASIS[.@k]) < 3) {
		mes "[Card Sage]";
		mes "You no longer carry three " + getitemname($@KW_BASIS[.@k]) + ".";
		close;
	}
	if (Zeny < $@KW_GEBUEHR[.@k]) {
		mes "[Card Sage]";
		mes "You lack the fee of " + callfunc("F_InsertComma", $@KW_GEBUEHR[.@k]) + " Zeny.";
		close;
	}
	set Zeny, Zeny - $@KW_GEBUEHR[.@k];
	delitem $@KW_BASIS[.@k], 3;
	getitem $@KW_ZIEL[.@k], 1;
	specialeffect2 EF_ABSORBSPIRITS;
	mes "[Card Sage]";
	mes "Three souls, one will.";
	mes "Take it - " + getitemname($@KW_ZIEL[.@k]) + ".";
	close;
}
