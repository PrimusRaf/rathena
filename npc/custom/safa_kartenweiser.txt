//===== SafaRO Script ========================================
//= Der Kartenweise (Card Sage)
//===== Beschreibung: ========================================
//= Verdichtet drei gleiche Monstercards zu einer staerkeren
//= Fassung. Regeln:
//=
//=     3x Grundkarte  -> Stufe 2 (Name-Stern),  Wirkung x2
//=     3x Stufe 2     -> Stufe 3 (zwei Sterne), Wirkung x4
//=     MVP-Karten     -> x3 je Stufe (Stufe 3 also x9)
//=     Skill-Karten   -> Skillstufe +1 je Stufe (nicht x2 -
//=                       skill SETZT die Stufe, sie summiert
//=                       sich nicht; Deckel ist das MaxLevel)
//=
//= Die verstaerkten Items, die Zuordnungs-Arrays und alle
//= Client-Eintraege erzeugt tools/kartenweiser/bauen.py.
//= Die Arrays liest safa_kartenweiser_daten.txt beim Start:
//=
//=     $@KW_BASIS[i]    Karte, die verlangt wird (3 Stueck)
//=     $@KW_ZIEL[i]     Karte, die entsteht (1 Stueck)
//=     $@KW_GEBUEHR[i]  Zeny-Gebuehr des Tauschs
//=     $@KW_STUFE[i]    entstehende Stufe (2 oder 3)
//=     $@KW_MVP[i]      1 = MVP-Karte
//=
//= Die Gebuehren stehen deshalb NICHT hier, sondern im
//= Werkzeug (GEBUEHR in bauen.py) - sie sind je Tauschweg
//= vorberechnet, damit dieser NPC keine eigene Preistabelle
//= pflegen muss.
//=
//= WARUM KEIN ZUFALL: wie beim Card Remover (safa_dienste) -
//= die Kosten sind die Kosten-Seite, und die ist deutlich.
//= Drei Karten plus Gebuehr sind kalkulierbar; ein Wuerfel
//= wuerde nicht Fehlentscheidungen bestrafen, sondern Pech.
//===== Ort: =================================================
//= lhz_in02 264,283 - BEWUSST NICHT im Veins-Quartier. Der
//= Kartenweise ist einer der Geheim-NPCs: er steht im
//= Obergeschoss der Lighthalzen-Herberge und wird nirgends
//= angekuendigt. Wer ihn findet, hat ihn. Naechster Nachbar
//= ist der Lab Staff auf 265,273; die Zelle ist frei.
//============================================================

lhz_in02,264,283,4	script	Kartenweiser#safa	4_M_SAGE_C,{
	disable_items;
	mes "[Kartenweiser]";
	mes "Drei gleiche Karten tragen dieselbe Seele. Ich kann sie zu einer einzigen verdichten - staerker, als die drei es zusammen waren.";
	next;
	switch (select("Karten verdichten:Wie funktioniert das?:Schon gut")) {
	case 2:
		mes "[Kartenweiser]";
		mes "Bring mir ^0000CCdrei gleiche Monstercards^000000 und die Gebuehr, und du erhaeltst eine Karte der naechsten Stufe:";
		mes " ";
		mes "^0000CCStufe 2^000000 (ein + am Namen): doppelte Wirkung";
		mes "^0000CCStufe 3^000000: nochmals verdoppelt - vierfach";
		next;
		mes "[Kartenweiser]";
		mes "^CC0000MVP-Karten^000000 sind dichter gewoben: sie ver^CC0000drei^000000fachen sich je Stufe.";
		mes " ";
		mes "Karten, die einen ^0000CCSkill^000000 verleihen, heben ihn stattdessen um ^0000CC+1 Stufe^000000 je Verdichtung - eine Skillstufe wird gesetzt, sie summiert sich nicht.";
		next;
		mes "[Kartenweiser]";
		mes "Auch ein ^0000CCRuestungselement^000000 laesst sich nicht verdoppeln - mehr Element als Element gibt es nicht.";
		mes "Stattdessen federe ich die ^0000CCSchwaeche^000000 der Elementruestung ab: je Stufe ^0000CC15%^000000 weniger Schaden vom Konter-Element. Angeling etwa mildert Schatten.";
		mes " ";
		mes "Andere einmalige Wirkungen (Sicht, Unzerbrechlich und dergleichen) gleiche ich mit ^0000CCMaxHP und MaxSP +3%^000000 je Stufe aus.";
		next;
		mes "[Kartenweiser]";
		mes "Die Gebuehr richtet sich nach Stufe und Seltenheit. Und zwei Warnungen:";
		mes " ";
		mes "^CC0000Die GANZE Karte verdichtet sich - auch ihre Nachteile.^000000 Wer eine Karte mit Malus verdichtet, vervielfacht den Malus mit.";
		mes " ";
		mes "Die verdichtete Karte ist ein ^0000CCeigenes Item^000000 - Set-Wirkungen mit anderen Karten (Combos) kennen nur die Grundkarte.";
		close;
	case 3:
		close;
	}

	// ---- Inventar gegen die Zuordnung halten -------------------
	getinventorylist;
	set .@n, 0;
	for (set .@i, 0; .@i < @inventorylist_count; set .@i, .@i + 1) {
		if (@inventorylist_amount[.@i] < 3) continue;
		if (@inventorylist_bound[.@i]) continue;
		set .@k, inarray($@KW_BASIS, @inventorylist_id[.@i]);
		if (.@k < 0) continue;
		set .@weg[.@n], .@k;
		set .@n, .@n + 1;
	}
	if (!.@n) {
		mes "[Kartenweiser]";
		mes "Du traegst keine drei gleichen Monstercards bei dir.";
		mes "Komm wieder, wenn sich drei Seelen gleichen Namens in deiner Tasche draengen.";
		close;
	}

	// ---- Auswahl -----------------------------------------------
	set .@m$, "";
	for (set .@i, 0; .@i < .@n; set .@i, .@i + 1) {
		set .@k, .@weg[.@i];
		set .@m$, .@m$ + "3x " + getitemname($@KW_BASIS[.@k])
			+ " > " + getitemname($@KW_ZIEL[.@k])
			+ ($@KW_MVP[.@k] ? " [MVP]" : "") + ":";
	}
	mes "[Kartenweiser]";
	mes "Diese Verdichtungen kann ich dir anbieten:";
	next;
	set .@k, .@weg[select(.@m$) - 1];

	// ---- Bestaetigung ------------------------------------------
	mes "[Kartenweiser]";
	mes "^0000CC3x " + getitemname($@KW_BASIS[.@k]) + "^000000";
	mes "werden zu";
	mes "^0000CC1x " + getitemname($@KW_ZIEL[.@k]) + "^000000";
	mes " ";
	mes "Gebuehr: ^CC0000" + callfunc("F_InsertComma", $@KW_GEBUEHR[.@k]) + " Zeny^000000";
	if ($@KW_MVP[.@k]) {
		mes "^CC0000Eine MVP-Karte - dreifache Verdichtung.^000000";
	}
	next;
	if (select("Verdichten:Lieber nicht") == 2) {
		mes "[Kartenweiser]";
		mes "Eine weise Zurueckhaltung ist auch eine Antwort.";
		close;
	}

	// ---- Pruefen und tauschen ----------------------------------
	// Erst jetzt und noch einmal: zwischen Menue und Klick kann
	// nichts verkauft werden (disable_items), aber Vorsicht
	// kostet nichts.
	if (countitem($@KW_BASIS[.@k]) < 3) {
		mes "[Kartenweiser]";
		mes "Du hast keine drei " + getitemname($@KW_BASIS[.@k]) + " mehr bei dir.";
		close;
	}
	if (Zeny < $@KW_GEBUEHR[.@k]) {
		mes "[Kartenweiser]";
		mes "Dir fehlt die Gebuehr von " + callfunc("F_InsertComma", $@KW_GEBUEHR[.@k]) + " Zeny.";
		close;
	}
	set Zeny, Zeny - $@KW_GEBUEHR[.@k];
	delitem $@KW_BASIS[.@k], 3;
	getitem $@KW_ZIEL[.@k], 1;
	specialeffect2 EF_ABSORBSPIRITS;
	mes "[Kartenweiser]";
	mes "Drei Seelen, ein Wille.";
	mes "Nimm sie - " + getitemname($@KW_ZIEL[.@k]) + ".";
	close;
}
