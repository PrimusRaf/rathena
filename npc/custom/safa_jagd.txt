//= Die Jagd
//===== Erzeugt von ==========================================
//= tools/jagd/bauen.py  --  NICHT von Hand pflegen.
//===========================================================
//
//  Ein GM startet mit @jagd. Irgendwo in der Welt erscheint
//  ein Golden Poring mit EINEM Lebenspunkt. Alle 15 Minuten
//  springt er weiter, und jedes Mal geht ein Hinweis an alle.
//
//  Wer ihn findet, hat ihn. Ein Lebenspunkt heisst: Level
//  spielt keine Rolle, nur Suchen zaehlt.
//
//  BEFEHLE (nur GM)
//    @jagd          startet, oder sagt wo es steht
//    @jagd stop     bricht ab
//    @jagd hinweis  wiederholt den letzten Hinweis
//    @jagd wo       verraet dem GM die Karte
//
//  WARUM DER ZUSTAND IN $-VARIABLEN STEHT
//
//  $ ist serverweit UND dauerhaft. Die Jagd endet nicht von
//  selbst, sie kann also einen Neustart erleben. Mit
//  gewoehnlichen NPC-Variablen waere sie danach spurlos weg -
//  der Poring aber auch, und niemand haette je erfahren, was
//  aus ihm wurde. OnInit setzt sie deshalb fort.
//
//  WARUM DAS MONSTER 2248 IST
//
//  Der Golden Poring kommt auf unserem Server nirgends vor
//  (nachgesehen ueber alle Spawn-Dateien) und der Client kennt
//  sein Sprite. Eine frei erfundene Nummer kennt er nicht und
//  wuerde beim Erscheinen eine Fehlermeldung werfen. Seine
//  Werte - 1 HP, kein Angriff, keine Erfahrung - stehen in
//  db/import/mob_db.yml.
//===========================================================

-	script	safa_jagd	-1,{
	end;

// ------------------------------------------------------------
//  @jagd
// ------------------------------------------------------------
OnGM:
	.@was$ = "";
	if (.@atcmd_numparameters > 0)
		.@was$ = .@atcmd_parameters$[0];

	if (.@was$ == "stop") {
		if (!$jagd_laeuft) {
			dispbottom "[Jagd] Es laeuft gerade keine.";
			end;
		}
		callsub S_Aufraeumen;
		set $jagd_laeuft, 0;
		announce "The hunt is called off. It slipped away.", bc_all, 0xFFCC00;
		end;
	}

	if (.@was$ == "wo") {
		if (!$jagd_laeuft) {
			dispbottom "[Jagd] Es laeuft gerade keine.";
			end;
		}
		dispbottom "[Jagd] Karte: " + .karte$[$jagd_pos]
		           + "  (Sprung " + $jagd_spruenge + ")";
		end;
	}

	if (.@was$ == "hinweis") {
		if (!$jagd_laeuft) {
			dispbottom "[Jagd] Es laeuft gerade keine.";
			end;
		}
		announce $jagd_hinweis$, bc_all, 0x00CCFF;
		end;
	}

	if ($jagd_laeuft) {
		dispbottom "[Jagd] Laeuft bereits - Sprung " + $jagd_spruenge
		           + ". Mit @jagd stop abbrechen.";
		end;
	}

	set $jagd_laeuft, 1;
	set $jagd_spruenge, 0;
	announce "Something came through the Threshold that should not have. "
	         + "It is hiding. Find it.", bc_all, 0xFFCC00;
	callsub S_Setzen;
	initnpctimer;
	end;


// ------------------------------------------------------------
//  Sprung
// ------------------------------------------------------------
OnTimer900000:
	stopnpctimer;
	if (!$jagd_laeuft)
		end;
	callsub S_Aufraeumen;
	announce "It moved.", bc_all, 0xFFCC00;
	callsub S_Setzen;
	initnpctimer;
	end;


// ------------------------------------------------------------
//  Neues Versteck suchen, setzen, Hinweis geben
// ------------------------------------------------------------
S_Setzen:
	// Nicht zweimal hintereinander dieselbe Karte - das sieht nach
	// einem Fehler aus, auch wenn es keiner waere.
	.@neu = rand(.anzahl);
	if (.@neu == $jagd_pos)
		.@neu = (.@neu + 1) % .anzahl;
	set $jagd_pos, .@neu;
	set $jagd_spruenge, $jagd_spruenge + 1;

	monster .karte$[$jagd_pos], -1, -1, "Something That Slipped Through", 2248, 1,
	        "safa_jagd::OnKill";

	set $jagd_hinweis$, callsub(S_Hinweis, $jagd_pos);
	announce $jagd_hinweis$, bc_all, 0x00CCFF;
	return;


S_Aufraeumen:
	killmonster .karte$[$jagd_pos], "safa_jagd::OnKill";
	return;


// ------------------------------------------------------------
//  Ein Hinweis, nie der Kartenname
// ------------------------------------------------------------
//  Staedte haben kein Levelband und keine Monster - fuer sie
//  bleiben zwei der vier Arten uebrig. Deshalb die Fallunter-
//  scheidung, statt einfach aus vier zu wuerfeln und bei einer
//  Stadt einen leeren Satz zu erzeugen.
// ------------------------------------------------------------
S_Hinweis:
	.@i = getarg(0);

	if (.stadtkarte[.@i]) {
		// Beide Staedte-Hinweise nennen die Gegend. "Irgendeine
		// Stadt" waere bei 33 Staedten in einer Viertelstunde nicht
		// zu schaffen - und ein Hinweis, dem niemand nachgehen kann,
		// ist keiner.
		switch (rand(2)) {
		case 0:
			return "It has mixed in with people, somewhere in "
			       + .gegend$[.@i] + ".";
		default:
			return "There are walls around it, and a Kafra inside. "
			       + "Look in " + .gegend$[.@i] + ".";
		}
	}

	switch (rand(4)) {
	case 0:
		return "It is " + .weg[.@i] + " crossing"
		       + ((.weg[.@i] == 1) ? "" : "s") + " out from "
		       + .stadt$[.@i] + ".";
	case 1:
		return "Where it sits, hunters between level " + .lvmin[.@i]
		       + " and " + .lvmax[.@i] + " make their living.";
	case 2:
		return "It shares the ground with " + .mob$[.@i] + ".";
	default:
		return "Somewhere in " + .gegend$[.@i] + ", out in the open.";
	}


// ------------------------------------------------------------
//  Gefangen
// ------------------------------------------------------------
OnKill:
	if (!$jagd_laeuft)
		end;
	set $jagd_laeuft, 0;
	stopnpctimer;

	.@name$ = strcharinfo(0);
	announce .@name$ + " caught it, after " + $jagd_spruenge
	         + " hiding places. Everyone online shares the reward.",
	         bc_all, 0xFFCC00;

	// Der Faenger zuerst, solange nur er angehaengt ist.
	getitem 30001, 200;

	// Danach alle. addrid haengt die uebrigen an, ohne den Faenger
	// abzuhaengen - der Rest des Skripts laeuft je Spieler einmal,
	// er bekommt seine 20 also zusaetzlich.
	addrid(0);
	getitem 30001, 20;
	sc_start SC_EXPBOOST, 1800000, 50;
	dispbottom "[Jagd] " + .@name$ + " hat es gefangen. "
	           + "20 Muenzen und "
	           + "50% Erfahrung fuer die naechste halbe Stunde.";
	end;
