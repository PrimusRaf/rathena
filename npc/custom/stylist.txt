//= Changes your hair style, hair color, and cloth color.
//===== Additional Comments: =================================
//= 1.0 Initial script.
//= 1.1 Switched to 'getbattleflag', credits to Saithis. [Euphy]
//= 2.0 SafaRO: 256 Farben statt 8, Auswahl ueber Farbfamilien.
//= 2.1 Klammerung der ?: - siehe unten. Texte auf Englisch.
//============================================================
//= SafaRO, Episode 22: von prontera 170,180 nach veins 238,125
//= gezogen. Er ist der einzige der uebernommenen rAthena-Customs
//= ohne Duplikate in anderen Staedten - Warper, Healer, Job
//= Master und Card Remover stehen ohnehin in jeder Stadt und
//= brauchten deshalb keinen Umzug.
//=
//= 238,125 setzt die Dienstezeile nach Osten fort, hinter der
//= Kafra bei 230,125. Zelle geprueft: begehbar, erreichbar,
//= frei. Siehe npc/custom/safa_dienste.txt.
//============================================================
//= DIE 256 FARBEN
//=
//= Frueher liefen Kleidungs- und Haarfarbe von 0 bis 7, und das
//= Blaettern durch acht Stueck war die ganze Bedienung. Seit
//= client/farben.grf sind es 256, und mit derselben Bedienung
//= waere das unbenutzbar: wer Dunkelgruen sucht, klickt sich
//= sonst durch hundert Rot- und Gelbtoene dorthin.
//=
//= Deshalb zwei Ebenen. Die Farbnummern sind nicht willkuerlich
//= vergeben, sondern ein Raster (siehe tools/sprite/
//= paletten_erweitern.py):
//=
//=     0 -   7   was Gravity mitliefert
//=     8 - 223   12 Farbfamilien zu je 18 Toenen
//=   224 - 255   32 Neutraltoene, fast schwarz bis fast weiss
//=
//= Die 18 Toene einer Familie sind 6 Helligkeiten mal 3
//= Saettigungen, in dieser Reihenfolge. Genau darauf beruht die
//= Namensliste .tone$ weiter unten - wer das Raster im Python-
//= Skript aendert, muss sie hier mitaendern, sonst stehen die
//= Namen an den falschen Farben.
//============================================================
//= EIN AT-ZEICHEN ZUVIEL - der Fehler aus Fassung 2.0
//=
//= Dort stand beim Uebernehmen der Familiengrenzen
//=
//=     .@von = .von[.@w - 1];      richtig
//=     .@bis = .@bis[.@w - 1];     falsch
//=
//= Das zweite liest nicht aus dem NPC-Array .bis, sondern aus
//= dem lokalen .@bis - also aus sich selbst, und dort steht
//= nichts. Ergebnis: die Obergrenze war 0. Damit war
//= .@farbe < .@bis nie wahr, "Next" fiel jedes Mal auf den
//= Anfang der Familie zurueck, und wer Limette gewaehlt hatte,
//= sass fuer immer auf der 80.
//=
//= Das Menue daneben zeigte trotzdem "Next (81)" - es rechnet
//= mit derselben kaputten Grenze und kommt beim Zurueckfallen
//= zufaellig auf denselben Wert. Es sah also aus, als haette
//= das Skript recht und der Server nicht.
//=
//= Rechtschreibpruefung gibt es fuer .@ nicht: eine nie gesetzte
//= Skriptvariable ist keine Fehlermeldung, sie ist 0. Deshalb
//= stehen die beiden Zuweisungen jetzt direkt untereinander -
//= dort faellt ein fehlendes oder ueberzaehliges @ beim Lesen
//= auf.
//=
//= Die ?: sind zusaetzlich geklammert, wie script_commands.txt
//= es empfiehlt. Ursache waren sie nicht: parse_variable ruft
//= fuer die rechte Seite parse_subexpr(p, -1) auf, und damit
//= wird das ?: mitgenommen.
//============================================================

veins,238,125,4	script	Stylist#custom_stylist	122,{
	mes "[Stylist]";
	mes "What shall I change?";
	next;
	switch (select(" ~ Cloth color: ~ Hairstyle: ~ Hair color: ~ Nothing, thanks")) {
	case 1:
		set .@look, LOOK_CLOTHES_COLOR;
		set .@max, getbattleflag("max_cloth_color");
		set .@what$, "Cloth color";
		break;
	case 2:
		callsub S_Hair;
		end;
	case 3:
		set .@look, LOOK_HAIR_COLOR;
		set .@max, getbattleflag("max_hair_color");
		set .@what$, "Hair color";
		break;
	case 4:
		close;
	}

	set .@original, getlook(.@look);
	set .@color, .@original;

S_Family:
	mes "[Stylist]";
	mes "Which direction?";
	set .@menu$, "";
	set .@fams, getarraysize(.family$);
	for (set .@i, 0; .@i < .@fams; set .@i, .@i + 1)
		set .@menu$, .@menu$ + " ~ " + .family$[.@i] + " (^0055FF" + .from[.@i] + "-" + .to[.@i] + "^000000):";
	set .@menu$, .@menu$ + " ~ Surprise me: ~ Back to ^0055FF" + .@original + "^000000 and done: ~ Done";

	set .@pick, select(.@menu$);
	if (.@pick == .@fams + 1) {
		// Der Zufall greift ueber alle Farben, nicht nur ueber eine
		// Familie - wer wuerfeln will, will ueberrascht werden.
		set .@color, rand(0, .@max);
		set .@from, 0;
		set .@to, .@max;
	} else if (.@pick == .@fams + 2) {
		setlook .@look, .@original;
		mes "[Stylist]";
		mes "Just as you were.";
		close;
	} else if (.@pick == .@fams + 3) {
		mes "[Stylist]";
		mes "It suits you.";
		close;
	} else {
		set .@from, .from[.@pick - 1];
		set .@to, .to[.@pick - 1];
		if (.@to > .@max)
			set .@to, .@max;
		if (.@from > .@max) {
			mes "[Stylist]";
			mes "These colors are not unlocked on this";
			mes "server yet. Tell a GM about it.";
			close;
		}
		set .@color, .@from;
	}

	// Ab hier wird jede Auswahl sofort angelegt. Ein Spiegel, in
	// dem man erst nach dem Bezahlen sieht, was man bekommt, waere
	// kein Spiegel.
	while (1) {
		setlook .@look, .@color;
		message strcharinfo(0), .@what$ + " " + .@color + " of " + .@max + ".";

		set .@next, ((.@color < .@to) ? .@color + 1 : .@from);
		set .@prev, ((.@color > .@from) ? .@color - 1 : .@to);
		set .@hastones, (((.@to - .@from) == 17) ? 1 : 0);

		set .@menu$, " ~ Next (^0055FF" + .@next + "^000000)"
		           + ": ~ Previous (^0055FF" + .@prev + "^000000)";
		if (.@hastones)
			set .@menu$, .@menu$ + ": ~ Pick a shade";
		set .@menu$, .@menu$ + ": ~ Enter a number: ~ Another direction"
		           + ": ~ Back to ^0055FF" + .@original + "^000000 and done: ~ Done";

		set .@pick, select(.@menu$);
		// Ohne die Schattenliste fehlt ein Menuepunkt, und alles
		// dahinter rutscht um eins vor. Einmal zurueckschieben ist
		// billiger als zwei getrennte switch-Bloecke zu pflegen.
		if (!.@hastones && .@pick >= 3)
			set .@pick, .@pick + 1;

		switch (.@pick) {
		case 1:
			set .@color, .@next;
			break;
		case 2:
			set .@color, .@prev;
			break;
		case 3:
			set .@menu$, "";
			for (set .@i, 0; .@i < 18; set .@i, .@i + 1)
				set .@menu$, .@menu$ + " ~ " + .tone$[.@i] + ":";
			set .@shade, select(.@menu$ + " ~ Never mind");
			if (.@shade <= 18)
				set .@color, .@from + .@shade - 1;
			break;
		case 4:
			mes "[Stylist]";
			mes "A number from ^0055FF0^000000 to ^0055FF" + .@max + "^000000.";
			mes "^0055FF0^000000 rolls the dice.";
			next;
			input .@typed, 0, .@max;
			set .@color, ((.@typed) ? .@typed : rand(0, .@max));
			// Von Hand eingegeben heisst: der Familienrahmen gilt
			// nicht mehr, sonst springt das Blaettern gleich darauf
			// zurueck und die eingetippte Farbe ist wieder weg.
			set .@from, 0;
			set .@to, .@max;
			break;
		case 5:
			goto S_Family;
		case 6:
			setlook .@look, .@original;
			mes "[Stylist]";
			mes "Just as you were.";
			close;
		case 7:
			mes "[Stylist]";
			mes "It suits you.";
			close;
		}
	}

S_Hair:
	set .@max, getbattleflag("max_hair_style");
	set .@original, getlook(LOOK_HAIR);
	set .@style, 1;
	while (1) {
		setlook LOOK_HAIR, .@style;
		message strcharinfo(0), "Hairstyle " + .@style + " of " + .@max + ".";

		set .@next, ((.@style < .@max) ? .@style + 1 : 1);
		set .@prev, ((.@style > 1) ? .@style - 1 : .@max);

		switch (select(" ~ Next (^0055FF" + .@next + "^000000)"
		             + ": ~ Previous (^0055FF" + .@prev + "^000000)"
		             + ": ~ Enter a number"
		             + ": ~ Back to ^0055FF" + .@original + "^000000 and done"
		             + ": ~ Done")) {
		case 1:
			set .@style, .@next;
			break;
		case 2:
			set .@style, .@prev;
			break;
		case 3:
			mes "[Stylist]";
			mes "A number from ^0055FF1^000000 to ^0055FF" + .@max + "^000000.";
			mes "^0055FF0^000000 rolls the dice.";
			next;
			input .@typed, 0, .@max;
			set .@style, ((.@typed) ? .@typed : rand(1, .@max));
			break;
		case 4:
			setlook LOOK_HAIR, .@original;
			mes "[Stylist]";
			mes "Just as you were.";
			close;
		case 5:
			mes "[Stylist]";
			mes "It suits you.";
			close;
		}
	}

OnInit:
	// Die Grenzen stammen aus tools/sprite/paletten_erweitern.py.
	// Lass sie in einer Zeile mit den Namen stehen - wer sie
	// getrennt pflegt, verschiebt frueher oder spaeter eine davon.
	setarray .family$[0], "Classic", "Red", "Orange", "Gold", "Yellow",
	                      "Lime", "Green", "Teal", "Cyan", "Blue",
	                      "Violet", "Magenta", "Pink", "Neutral";
	setarray .from[0],       0,   8,  26,  44,  62,  80,  98, 116, 134, 152, 170, 188, 206, 224;
	setarray .to[0],         7,  25,  43,  61,  79,  97, 115, 133, 151, 169, 187, 205, 223, 255;

	// 6 Helligkeiten mal 3 Saettigungen, in genau der Reihenfolge,
	// in der das Python-Skript sie schreibt.
	setarray .tone$[0], "Very dark, muted",  "Very dark, clear",  "Very dark, vivid",
	                    "Dark, muted",       "Dark, clear",       "Dark, vivid",
	                    "Deep, muted",       "Deep, clear",       "Deep, vivid",
	                    "Medium, muted",     "Medium, clear",     "Medium, vivid",
	                    "Light, muted",      "Light, clear",      "Light, vivid",
	                    "Very light, muted", "Very light, clear", "Very light, vivid";
	end;
}
