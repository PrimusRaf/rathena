//= SafaRO, Episode 22: von prontera 170,180 nach veins 238,125
//= gezogen. Er ist der einzige der uebernommenen rAthena-Customs
//= ohne Duplikate in anderen Staedten - Warper, Healer, Job
//= Master und Card Remover stehen ohnehin in jeder Stadt und
//= brauchten deshalb keinen Umzug.
//=
//= 238,125 setzt die Dienstezeile nach Osten fort, hinter der
//= Kafra bei 230,125. Zelle geprueft: begehbar, erreichbar,
//= frei. Siehe npc/custom/safa_dienste.txt.
//============================================================
//= DIE 256 FARBEN
//=
//= Frueher liefen Kleidungs- und Haarfarbe von 0 bis 7, und das
//= Blaettern durch acht Stueck war die ganze Bedienung. Seit
//= client/farben.grf sind es 256, und mit derselben Bedienung
//= waere das unbenutzbar: wer Dunkelgruen sucht, klickt sich
//= sonst durch hundert Rot- und Gelbtoene dorthin.
//=
//= Deshalb zwei Ebenen. Die Farbnummern sind nicht willkuerlich
//= vergeben, sondern ein Raster (siehe tools/sprite/
//= paletten_erweitern.py):
//=
//=     0 -   7   was Gravity mitliefert
//=     8 - 223   12 Farbfamilien zu je 18 Toenen
//=   224 - 255   32 Neutraltoene, fast schwarz bis fast weiss
//=
//= Die 18 Toene einer Familie sind 6 Helligkeiten mal 3
//= Saettigungen, in dieser Reihenfolge. Genau darauf beruht die
//= Namensliste .ton$ weiter unten - wer das Raster im Python-
//= Skript aendert, muss sie hier mitaendern, sonst stehen die
//= Namen an den falschen Farben.
//============================================================

veins,238,125,4	script	Stylist#custom_stylist	122,{
	mes "[Stylistin]";
	mes "Was darf ich an dir aendern?";
	next;
	switch (select(" ~ Kleidungsfarbe: ~ Frisur: ~ Haarfarbe: ~ Nichts, danke")) {
	case 1:
		.@look = LOOK_CLOTHES_COLOR;
		.@max = getbattleflag("max_cloth_color");
		.@was$ = "Kleidungsfarbe";
		break;
	case 2:
		callsub S_Frisur;
		end;
	case 3:
		.@look = LOOK_HAIR_COLOR;
		.@max = getbattleflag("max_hair_color");
		.@was$ = "Haarfarbe";
		break;
	case 4:
		close;
	}

	.@urspr = getlook(.@look);
	.@farbe = .@urspr;

S_Familie:
	mes "[Stylistin]";
	mes "Welche Richtung?";
	.@menu$ = "";
	.@n = getarraysize(.familie$);
	for (.@i = 0; .@i < .@n; .@i++)
		.@menu$ = .@menu$ + " ~ " + .familie$[.@i] + " (^0055FF" + .von[.@i] + "-" + .bis[.@i] + "^000000):";
	.@menu$ = .@menu$ + " ~ Zufall: ~ Zurueck zu ^0055FF" + .@urspr + "^000000 und fertig: ~ Fertig";

	.@w = select(.@menu$);
	if (.@w == .@n + 1) {
		// Zufall ueber alles, nicht nur ueber eine Familie -
		// wer wuerfeln will, will ueberrascht werden.
		.@farbe = rand(0, .@max);
		.@von = 0;
		.@bis = .@max;
	} else if (.@w == .@n + 2) {
		setlook .@look, .@urspr;
		mes "Wie du warst.";
		close;
	} else if (.@w == .@n + 3) {
		mes "Steht dir.";
		close;
	} else {
		.@von = .von[.@w - 1];
		.@bis = .@bis[.@w - 1];
		if (.@bis > .@max)
			.@bis = .@max;
		if (.@von > .@max) {
			mes "Diese Farben sind auf dem Server noch nicht";
			mes "freigeschaltet. Sag einem GM Bescheid.";
			close;
		}
		.@farbe = .@von;
	}

	// Ab hier wird jede Auswahl sofort angelegt. Ein Spiegel, in
	// dem man erst nach dem Bezahlen sieht, was man bekommt, waere
	// kein Spiegel.
	while (1) {
		setlook .@look, .@farbe;
		message strcharinfo(0), .@was$ + " " + .@farbe + " von " + .@max + ".";

		.@menu$ = " ~ Weiter (^0055FF" + ((.@farbe < .@bis) ? .@farbe + 1 : .@von) + "^000000)"
		        + ": ~ Zurueck (^0055FF" + ((.@farbe > .@von) ? .@farbe - 1 : .@bis) + "^000000)";
		.@tonliste = (.@bis - .@von == 17);
		if (.@tonliste)
			.@menu$ = .@menu$ + ": ~ Ton aus der Liste";
		.@menu$ = .@menu$ + ": ~ Nummer eingeben: ~ Andere Richtung"
		        + ": ~ Zurueck zu ^0055FF" + .@urspr + "^000000 und fertig: ~ Fertig";

		.@w = select(.@menu$);
		if (!.@tonliste && .@w >= 3)
			.@w = .@w + 1;

		switch (.@w) {
		case 1:
			.@farbe = (.@farbe < .@bis) ? .@farbe + 1 : .@von;
			break;
		case 2:
			.@farbe = (.@farbe > .@von) ? .@farbe - 1 : .@bis;
			break;
		case 3:
			.@menu$ = "";
			for (.@i = 0; .@i < 18; .@i++)
				.@menu$ = .@menu$ + " ~ " + .ton$[.@i] + ":";
			.@ton = select(.@menu$ + " ~ Doch nicht");
			if (.@ton <= 18)
				.@farbe = .@von + .@ton - 1;
			break;
		case 4:
			mes "[Stylistin]";
			mes "Eine Zahl von ^0055FF0^000000 bis ^0055FF" + .@max + "^000000.";
			mes "Die ^0055FF0^000000 wuerfelt.";
			next;
			input .@eingabe, 0, .@max;
			.@farbe = (.@eingabe) ? .@eingabe : rand(0, .@max);
			// Von Hand eingegeben heisst: der Familienrahmen gilt
			// nicht mehr, sonst springt das Blaettern gleich darauf
			// zurueck und die eingetippte Farbe ist wieder weg.
			.@von = 0;
			.@bis = .@max;
			break;
		case 5:
			goto S_Familie;
		case 6:
			setlook .@look, .@urspr;
			mes "[Stylistin]";
			mes "Wie du warst.";
			close;
		case 7:
			mes "[Stylistin]";
			mes "Steht dir.";
			close;
		}
	}

S_Frisur:
	.@max = getbattleflag("max_hair_style");
	.@urspr = getlook(LOOK_HAIR);
	.@stil = 1;
	while (1) {
		setlook LOOK_HAIR, .@stil;
		message strcharinfo(0), "Frisur " + .@stil + " von " + .@max + ".";
		switch (select(" ~ Weiter (^0055FF" + ((.@stil != .@max) ? .@stil + 1 : 1) + "^000000)"
		             + ": ~ Zurueck (^0055FF" + ((.@stil != 1) ? .@stil - 1 : .@max) + "^000000)"
		             + ": ~ Nummer eingeben"
		             + ": ~ Zurueck zu ^0055FF" + .@urspr + "^000000 und fertig"
		             + ": ~ Fertig")) {
		case 1:
			.@stil = (.@stil != .@max) ? .@stil + 1 : 1;
			break;
		case 2:
			.@stil = (.@stil != 1) ? .@stil - 1 : .@max;
			break;
		case 3:
			mes "[Stylistin]";
			mes "Eine Zahl von ^0055FF1^000000 bis ^0055FF" + .@max + "^000000.";
			mes "Die ^0055FF0^000000 wuerfelt.";
			next;
			input .@eingabe, 0, .@max;
			.@stil = (.@eingabe) ? .@eingabe : rand(1, .@max);
			break;
		case 4:
			setlook LOOK_HAIR, .@urspr;
			mes "[Stylistin]";
			mes "Wie du warst.";
			close;
		case 5:
			mes "[Stylistin]";
			mes "Steht dir.";
			close;
		}
	}

OnInit:
	// Die Grenzen stammen aus tools/sprite/paletten_erweitern.py.
	// Lass sie in einer Zeile mit den Namen stehen - wer sie
	// getrennt pflegt, verschiebt frueher oder later eine davon.
	setarray .familie$[0], "Klassisch", "Rot", "Orange", "Gold", "Gelb",
	                       "Limette", "Gruen", "Tuerkis", "Cyan", "Blau",
	                       "Violett", "Magenta", "Pink", "Neutral";
	setarray .von[0],        0,   8,  26,  44,  62,  80,  98, 116, 134, 152, 170, 188, 206, 224;
	setarray .bis[0],        7,  25,  43,  61,  79,  97, 115, 133, 151, 169, 187, 205, 223, 255;

	// 6 Helligkeiten mal 3 Saettigungen, in genau der Reihenfolge,
	// in der das Python-Skript sie schreibt.
	setarray .ton$[0], "Sehr dunkel, staubig", "Sehr dunkel, klar", "Sehr dunkel, kraeftig",
	                   "Dunkel, staubig",      "Dunkel, klar",      "Dunkel, kraeftig",
	                   "Gedeckt, staubig",     "Gedeckt, klar",     "Gedeckt, kraeftig",
	                   "Mittel, staubig",      "Mittel, klar",      "Mittel, kraeftig",
	                   "Hell, staubig",        "Hell, klar",        "Hell, kraeftig",
	                   "Sehr hell, staubig",   "Sehr hell, klar",   "Sehr hell, kraeftig";
	end;
}
