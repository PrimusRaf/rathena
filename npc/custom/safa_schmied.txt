//
//  ZWEI DIENSTE, ZWEI GANZ VERSCHIEDENE MECHANIKEN
//
//  Sockel: die Slotzahl haengt an der Item-ID, nicht am
//  einzelnen Exemplar - auch der Client liest sie aus seiner
//  eigenen Tabelle. Ein Slot laesst sich deshalb nicht
//  "aufschalten". Der NPC tauscht das Stueck gegen dieselbe
//  Ausfuehrung mit mehr Slots, sofern es sie gibt. Die Paare
//  unten sind abgeleitet, nicht getippt.
//
//  Rund 3 Prozent der Ausruestung hat eine solche Variante.
//  Fuer den Rest lehnt er ab und sagt auch, warum.
//
//  Verzauberung: setrandomoption schreibt Optionen direkt auf
//  das getragene Teil. Kein Tausch, kein Fehlschlag - und es
//  geht auch auf Kostuem- und Schattenteilen, wo Karten
//  grundsaetzlich nicht hineinpassen (pc.cpp:5632).
//
//  WAS BEIM TAUSCH ERHALTEN BLEIBT
//
//  Der Verfeinerungsgrad wird uebernommen. Karten und bereits
//  gesetzte Optionen NICHT - deshalb nimmt der NPC solche
//  Stuecke gar nicht erst an, statt sie stillschweigend zu
//  verschlucken. Das offizielle Skript macht an dieser Stelle
//  das Gegenteil: es loescht und wuerfelt dazu noch.
//
//===========================================================

prontera,163,164,4	script	Master Artisan#safa	4_M_ALCHE_A,{
	mes "[Master Artisan]";
	mes "Armstrong sends me the ones who know where to go.";
	mes "I am for the ones who know what they want to wear.";
	next;

L_Menue:
	switch (select("Widen a socket - " + callfunc("F_InsertComma", .je_slot) + "z per slot",
	               "Enchant a piece - " + callfunc("F_InsertComma", .enchant) + "z",
	               "What can you not do?",
	               "Nothing today.")) {
	case 1: callsub S_Sockel;  break;
	case 2: callsub S_Zauber;  break;
	case 3: callsub S_Grenzen; break;
	case 4:
		mes "[Master Artisan]";
		mes "Come back when something of yours needs opening up.";
		close;
	}
	goto L_Menue;

// ------------------------------------------------------------
//  Sockel
// ------------------------------------------------------------
S_Sockel:
	// Nur normale Plaetze. Kostuem und Schatten stehen bewusst nicht
	// zur Wahl - dort passt ohnehin keine Karte hinein.
	setarray .@platz[0], EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW,
	                     EQI_ARMOR, EQI_HAND_R, EQI_HAND_L,
	                     EQI_GARMENT, EQI_SHOES, EQI_ACC_R, EQI_ACC_L;
	.@n = 0;
	.@liste$ = "";
	for (.@i = 0; .@i < getarraysize(.@platz); .@i++) {
		if (getequipisequiped(.@platz[.@i])) {
			.@wahl[.@n] = .@platz[.@i];
			.@liste$ = .@liste$ + getequipname(.@platz[.@i]) + ":";
			.@n++;
		}
	}
	if (.@n == 0) {
		mes "[Master Artisan]";
		mes "You are wearing nothing I can work on.";
		next;
		return;
	}

	mes "[Master Artisan]";
	mes "Which one?";
	next;
	.@k = select(.@liste$ + "Never mind") - 1;
	if (.@k >= .@n)
		return;
	.@teil = .@wahl[.@k];
	.@id = getequipid(.@teil);

	// Passendes Paar suchen.
	.@idx = -1;
	for (.@i = 0; .@i < getarraysize(.von); .@i++) {
		if (.von[.@i] == .@id) {
			.@idx = .@i;
			break;
		}
	}
	if (.@idx < 0) {
		mes "[Master Artisan]";
		mes "^FF6666That one has no wider version.^000000";
		mes " ";
		mes "I do not carve new holes - I trade the piece for the same piece with more of them. If the smiths never made such a version, there is nothing for me to trade with.";
		next;
		return;
	}

	// Karten und Optionen koennen wir nicht mitnehmen. Also gar nicht
	// erst anfangen, statt sie hinterher verschwinden zu lassen.
	if (getequipcardcnt(.@teil) > 0) {
		mes "[Master Artisan]";
		mes "^FF6666There are cards in it.^000000";
		mes "The trade would leave them behind, and I will not do that to you. Have them taken out first - the ^00CCFFSafari Specialist^000000 at ^00CCFFprontera 163, 174^000000 does it without risk.";
		next;
		return;
	}
	.@opt = 0;
	for (.@i = 0; .@i < 5; .@i++) {
		if (getequiprandomoption(.@teil, .@i, ROA_ID) > 0)
			.@opt++;
	}
	if (.@opt > 0) {
		mes "[Master Artisan]";
		mes "^FF6666That piece is already enchanted.^000000";
		mes "The enchantment sits on this exact piece, not on its name. Trade it away and the work is gone. Sockets first, enchantment second - that is the order.";
		next;
		return;
	}

	.@plus = .nach_slots[.@idx] - .vor_slots[.@idx];
	.@preis = .@plus * .je_slot;
	.@ref = getequiprefinerycnt(.@teil);

	mes "[Master Artisan]";
	mes "^00CCFF" + getequipname(.@teil) + "^000000";
	mes " ";
	mes "Slots: ^FFCC00" + .vor_slots[.@idx] + "^000000 -> ^FFCC00" + .nach_slots[.@idx] + "^000000";
	mes "Refine ^FFCC00+" + .@ref + "^000000 comes across with it.";
	mes " ";
	mes "^FFCC00" + callfunc("F_InsertComma", .@preis) + " Zeny^000000 for " + .@plus + " slot" + ((.@plus == 1) ? "" : "s") + ".";
	next;
	if (select("Do it:Leave it alone") == 2)
		return;
	if (Zeny < .@preis) {
		mes "[Master Artisan]";
		mes "Not with what you are carrying.";
		next;
		return;
	}

	// Erst vom Koerper nehmen, dann tauschen. delitem2 wuerde ein
	// getragenes Stueck zwar auch finden, aber der Client behielte
	// bis zum naechsten Statuslauf die alten Boni im Fenster.
	unequip .@teil;
	delitem2 .@id, 1, 1, .@ref, 0, 0, 0, 0, 0;
	getitem2 .nach[.@idx], 1, 1, .@ref, 0, 0, 0, 0, 0;
	Zeny -= .@preis;

	specialeffect2 EF_REPAIRWEAPON;
	mes "[Master Artisan]";
	mes "Done. ^00AA00" + .@plus + " more slot" + ((.@plus == 1) ? "" : "s") + "^000000, and the refine held.";
	mes "It is in your bag, not on your back - put it on yourself, I do not dress people.";
	next;
	return;

S_Zauber:

		if (getequipisequiped(.@platz[.@i])) {
			.@wahl[.@n] = .@platz[.@i];
			.@liste$ = .@liste$ + getequipname(.@platz[.@i]) + ":";
			.@n++;
		}
	}
	if (.@n == 0) {
		mes "[Master Artisan]";
		mes "Put something on first.";
		next;
		return;
	}

	mes "[Master Artisan]";
	mes "Which piece?";
	next;
	.@k = select(.@liste$ + "Never mind") - 1;
	if (.@k >= .@n)
		return;
	.@teil = .@wahl[.@k];

	// Freien Platz suchen. Fuenf sind das Hoechste, was ein Stueck
	// tragen kann (MAX_ITEM_RDM_OPT in mmo.hpp).
	.@frei = -1;
	for (.@i = 0; .@i < 5; .@i++) {
		if (getequiprandomoption(.@teil, .@i, ROA_ID) == 0) {
			.@frei = .@i;
			break;
		}
	}
	if (.@frei < 0) {
		mes "[Master Artisan]";
		mes "^FF6666Five enchantments already.^000000 That is all a thing will hold before it stops being itself.";
		next;
		return;
	}

	mes "[Master Artisan]";
	mes "^00CCFF" + getequipname(.@teil) + "^000000, slot " + (.@frei + 1) + " of 5.";
	mes " ";
	mes "Pick what it should carry. ^FFCC00" + callfunc("F_InsertComma", .enchant) + " Zeny^000000.";
	mes "^777777It always takes. How strongly is not mine to decide.^000000";
	next;
	.@o = select(.menue$) - 1;
	if (.@o >= getarraysize(.opt_id))
		return;

	mes "[Master Artisan]";
	mes "^00CCFF" + .opt_name$[.@o] + "^000000, somewhere between ^FFCC00"
	    + .opt_von[.@o] + "^000000 and ^FFCC00" + .opt_bis[.@o] + "^000000.";
	next;
	if (select("Pay " + callfunc("F_InsertComma", .enchant) + " Zeny:Not yet") == 2)
		return;
	if (Zeny < .enchant) {
		mes "[Master Artisan]";
		mes "Not with what you are carrying.";
		next;
		return;
	}

	// Zwischen Anzeige und Zahlung kann das Stueck vom Koerper
	// verschwunden sein - der Spieler kann inzwischen umgezogen haben.
	if (getequipisequiped(.@teil) == 0) {
		mes "[Master Artisan]";
		mes "You took it off. Put it back on and ask again.";
		next;
		return;
	}

	Zeny -= .enchant;
	.@wert = rand(.opt_von[.@o], .opt_bis[.@o]);
	setrandomoption(.@teil, .@frei, .opt_id[.@o], .@wert, 0);

	specialeffect2 EF_REFINEOK;
	mes "[Master Artisan]";
	mes "^00AA00" + .opt_name$[.@o] + " +" + .@wert + "^000000.";
	if (.@wert >= .opt_bis[.@o])
		mes "As high as it goes. You will not do better than that.";
	else if (.@wert <= .opt_von[.@o])
		mes "The low end. The stone decides, not me.";
	next;
	return;

//  Was nicht geht, und warum
// ------------------------------------------------------------

	next;

	mes " ";
	mes "For those, it is the only way to make them worth anything.";
	next;
	return;
