// ============================================================
//  SafaRO - Kartendrops fuer die Episoden-Materialien
// ============================================================
//
//  WAS DIESE DATEI ERSETZT
//
//  Vorgaenger war npc/custom/safa_ep18.txt. Die Datei ging von
//  einer falschen Annahme aus: sie hat behauptet, die Karten der
//  Episode 18 haetten keine Monster, und hat deshalb 370 Einheiten
//  der Monster 20920-20928 auf gw_fild, oz_dun, ra_fild10/11 und
//  die Rachel-Tempelkarten gesetzt.
//
//  Beides war falsch, nachgeprueft gegen ratemyserver.net und
//  gegen unsere eigenen Dateien:
//
//    1. gw_fild01/02 und oz_dun01/02 SIND bevoelkert. Die Spawns
//       stehen in npc/re/mobs/fields/gw_fild.txt und
//       npc/re/mobs/dungeons/oz_dun.txt, sind ueber
//       npc/re/scripts_monsters.conf geladen, und stimmen Zeile
//       fuer Zeile mit RMS ueberein (gw_fild01: Ashhopper 50,
//       Ashring 70, Grey Wolf 35 - identisch).
//
//    2. Die Monster 20920-20928 gehoeren nicht zu Episode 18.
//       Es sind die Chimaeren von amicitia1/2, wo sie bereits
//       stehen. RMS fuehrt amicitia unter Episode 17.1.
//
//  Die alte Datei hat also korrekt bevoelkerte Karten ein zweites
//  Mal bevoelkert, mit Monstern einer anderen Episode. Sie ist
//  ersatzlos entfallen; hier bleibt nur der Kartendrop.
//
//  WARUM DER DROP UEBER EIN KILL-EREIGNIS LAEUFT
//
//  Drops haengen bei rAthena am MONSTER, nicht an der Karte.
//  Traegt man den Splitter in mob_db ein, faellt er ueberall, wo
//  das Monster vorkommt. Ein echter Kartendrop braucht deshalb
//  OnNPCKillEvent mit einer Kartenpruefung - dasselbe Verfahren,
//  mit dem in safa_shops.txt die Destiny Coins bei MVPs fallen.
//
//  WARUM DAS UEBERHAUPT NOETIG IST
//
//  Nachgesehen in db/re/mob_db.yml: KEIN EINZIGES Monster auf dem
//  Server laesst eines dieser sechs Materialien fallen. Amtlich
//  kommen sie aus Instanzen und Quests, nicht von Feldmonstern.
//  Wer die Tauschlaeden benutzen will, hat ohne diese Datei nur
//  den Instanzweg.
//
//  WAS HIER FEHLT: EPISODE 19 UND 21
//
//  Episode 19 (Issgard, Schneeblueten) und die Feldkarten der
//  Episode 21 (jor_albe, jor_crk, jor_tmple1) sind auf dem Server
//  vollstaendig leer - keine einzige Spawn-Zeile. Das ist kein
//  Versehen von uns: RMS meldet fuer alle 18 Issgard-Karten
//  woertlich "No monster on this map", und markiert jor_tmple1
//  als "(placeholder)". Es gibt dort schlicht nichts zu toeten,
//  also kann es auch keinen Kartendrop geben.
//
// ============================================================


-	script	#safa_episoden	-1,{
	end;

OnNPCKillEvent:
	if (killedrid == 0) end;

	getmapxy(.@map$, .@x, .@y, BL_PC);

	// inarray statt einer Kette von Vergleichen: das Ereignis
	// feuert bei JEDEM Kill auf dem ganzen Server, also auch bei
	// jedem Poring. Was hier steht, laeuft millionenfach. Die
	// Kartenpruefung steht deshalb ganz vorn und bricht sofort ab.
	.@i = inarray(.karte$[0], .@map$);
	if (.@i < 0) end;

	if (rand(10000) >= .chance) end;

	// Zwei Episoden haben zwei Materialien. Statt beide zu geben
	// wird eines davon gezogen - sonst waere die Ausbeute dort
	// doppelt so hoch wie in Episode 18 und 20.
	.@ep = .ep[.@i];
	if (.@ep == 171)      setarray .@gabe[0], 25723, 25669;
	else if (.@ep == 172) setarray .@gabe[0], 1000103, 1000104;
	else if (.@ep == 18)  setarray .@gabe[0], 1000405;
	else end;

	// Ueber F_SafaBeute: Autoloot wirkt nur auf Monster-Drops
	// (mob.cpp:2592), nicht auf per Skript hingelegte Gegenstaende.
	callfunc "F_SafaBeute", .@gabe[rand(getarraysize(.@gabe))], 1;
	end;

OnInit:
	// 2000 von 10000, also 20 Prozent je Kill. Bewusst hoch: die
	// Tauschlaeden dieser Episoden verlangen dreistellige Mengen,
	// und wir wollen den Fortschritt oeffnen, nicht simulieren.
	.chance = 2000;

	// Karte -> Episode. Nur Karten, die nachweislich Monster
	// haben; geprueft gegen npc/re/mobs und gegen RMS.
	//
	//  17.1 Illusion            Cor Core, Mysterious Component
	//  17.2 Sage's Legacy       Varmeal Ticket, Magical Igni Stone
	//  18   Direction of Prayer Amethyst Fragment
	//
	//  EPISODE 20 STEHT BEWUSST NICHT HIER
	//
	//  for_dun01/02 war zwischenzeitlich drin, mit den Sacred Cat
	//  Whiskers. Das war falsch: die Tauschtabelle der Episode 20
	//  (npc/re/merchants/barters/garden_of_time.yml) verlangt gar
	//  keine Cat Whiskers, sondern Contaminated Magic, Temporal
	//  Crystals und die vier Jahreszeiten-Energien - und die
	//  fallen bei den for_dun-Monstern laengst von selbst.
	//
	//  Episode 20 ist also bereits vollstaendig; es fehlte nur ein
	//  NPC, der die Tabelle oeffnet. Der steht weiter unten.
	setarray .karte$[0],
		"sp_rudus","sp_rudus2","sp_rudus3","sp_rudus4","amicitia1","amicitia2",
		"ba_maison","ba_lost","ba_pw01","ba_pw02","ba_pw03","ba_2whs01","ba_2whs02","ba_bath","ba_lib",
		"gw_fild01","gw_fild02","oz_dun01","oz_dun02";
	setarray .ep[0],
		171,171,171,171,171,171,
		172,172,172,172,172,172,172,172,172,
		18,18,18,18;
	end;
}


// ============================================================
//  Die Tauschstellen an der Prontera-Achse
// ------------------------------------------------------------
//  Sechs NPCs, einer je Episode, an den vorgegebenen Feldern.
//
//  WARUM SIE NICHTS SELBST ANBIETEN
//
//  Die Tauschtabellen gibt es schon - rAthena pflegt sie in
//  npc/re/merchants/barters/*.yml. Ein eigener Laden waere eine
//  zweite, handgepflegte Kopie, die nach der ersten Aenderung
//  falsch dasteht. Deshalb rufen diese NPCs die amtlichen
//  Tabellen ueber callshop auf und pflegen selbst gar nichts.
//
//  ZWEI TABELLEN WAREN BISHER UNERREICHBAR
//
//  Nachgesehen, welcher NPC welche Tabelle oeffnet: fuer
//  barter_auto_equip_1, barter_auto_equip_2 und die fuenf
//  barter_t_garden_* gibt es auf dem ganzen Server KEINEN.
//  rAthena liefert die Tabellen mit, aber niemanden, der sie
//  aufmacht. Die zehn Aufwertungen der Episode 17.2 und die
//  gesamte Episode 20 waren damit tot. Hier sind sie erreichbar.
//
//  ZU DEN SPRITES
//
//  Alle sechs sind aus Originalen dieser Episoden uebernommen,
//  die ich im Quelltext gelesen habe - keine geratenen Namen.
//  Ein erfundenes Sprite laesst den NPC unsichtbar werden.
// ============================================================


// ------------------------------------------------------------
//  Episode 17.1 - Illusion
// ------------------------------------------------------------
//  Als Duplikat statt als eigenes Skript: das Original waehlt
//  den Laden nach Klasse aus (shop_ext171_JOB_RUNE_KNIGHT und so
//  weiter, 18 Stueck). Diese Zuordnung nachzubauen hiesse, 80
//  Zeilen rAthena-Logik zu kopieren, die beim naechsten neuen
//  Job veraltet. Das Duplikat teilt sich das Skript und bleibt
//  damit automatisch richtig.
//
//  Original: sp_cor,136,156 in
//  npc/re/merchants/enchan_illusion_17_1.txt
//
//  Nicht mit dabei ist der "Butterfly Merchant" von sp_cor, der
//  nur das Teleport-Fluegelchen verkauft - ein Gegenstand.
prontera,135,170,4	duplicate(Grace Operator#ext162)	Episode 17.1 Exchange#safa	4_M_REPAIR


// ------------------------------------------------------------
//  Episode 17.2 - Sage's Legacy
// ------------------------------------------------------------
//  MD_ASSISTANT sieht nach einem Fehler aus, ist aber keiner:
//  das ist kein NPC-Sprite aus src/map/npc.hpp, sondern ein
//  MONSTER aus db/re/mob_db.yml. rAthena laesst Monsternamen als
//  NPC-Sprite zu, und das amtliche Original - Spiera auf ba_in01
//  in npc/re/merchants/enchan_sage_legacy_17_2.txt - benutzt
//  genau diesen Namen. Bitte nicht gegen ein JT_-Sprite tauschen.
prontera,137,170,4	script	Episode 17.2 Exchange#safa	MD_ASSISTANT,{

	mes "[Episode 17.2 Exchange]";
	mes "Varmundt's mansion. I deal in";
	mes "^0000FFMagical Soapstone^000000 and";
	mes "^0000FFBarmil Tickets^000000.";
	next;

	switch (select(
		"Automatic Modules - 51 kinds, 300 soapstone each",
		"Automatic equipment - armor, wings, legs",
		"Automatic equipment - boosters and battle chips",
		"Mansion teleport wing",
		"Little Headdress",
		"Nothing today")) {

	case 1: close2; callshop "barter_EP17_2_EP", 1; end;

	// Diese beiden verlangen zusaetzlich das Illusionsteil, das
	// aufgewertet werden soll - Illusion_Armor_A wird zu
	// Auto_Armor_A. Ohne das Teil zeigt das Fenster den Eintrag
	// zwar an, der Kauf schlaegt aber fehl.
	case 2: close2; callshop "barter_auto_equip_1", 1; end;
	case 3: close2; callshop "barter_auto_equip_2", 1; end;

	case 4: close2; callshop "barter_Teleport_Ep17_02", 1; end;
	case 5: close2; callshop "barter_Ep_17_2_C_Admin_Acc", 1; end;
	}

	mes "[Episode 17.2 Exchange]";
	mes "The mansion keeps its hours.";
	close;
}
