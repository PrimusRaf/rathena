//===== SafaRO Script ========================================
//= The Wild - Enraged
//===== Von Hand gepflegt ====================================
//= Anders als safa_wildlife.txt wird diese Datei NICHT
//= erzeugt. Es gibt keine Tabelle darin - das Ereignis
//= braucht keine, weil es nichts aufstellt, sondern nur
//= vorhandene Monster umwertet.
//===========================================================
//
//  WAS PASSIERT
//
//  Waehrend das Ereignis laeuft, wird JEDES Monster auf dem
//  Server zehnmal so gefaehrlich: es haelt zehnmal so lange
//  durch und schlaegt zehnmal so hart zu. Der Name bekommt den
//  Vorsatz "Enraged", damit man auf einen Blick sieht, was
//  einen da anfaellt. Erfahrung und Beute steigen im selben
//  Faktor, sonst wuerde niemand rausgehen.
//
//  ES GIBT KEINE AUSNAHMEN.
//
//  Nicht "fast alle", nicht "alles ausser den Sonderfaellen":
//  jedes Monster, das lebt. MVPs, Minibosse, die Adds eines
//  MVP, Wildlife-Rudel, der Biomboss, Monster in laufenden
//  Instanzen, die Pflanzen und Kugeln, die Spieler selbst
//  aufstellen. Das ist der Unterschied zu Wildlife, das MVPs
//  und Minibosse ausnimmt.
//
//  Die erste Fassung dieser Datei liess drei Gruppen aus, weil
//  sie sich hinterher nicht exakt zuruecksetzen liessen. Das
//  ist bewusst aufgegeben worden - was es kostet, steht bei
//  S_Beruhigen, und es ist bezahlt, nicht uebersehen.
//
//  Gestartet wird ausschliesslich von Hand: @enraged an 60.
//  Kein Zeitplan, kein Wuerfel. Wer das aendern will, haengt
//  spaeter einen Timer an OnStart - die Mechanik darunter
//  bleibt dieselbe.
//
//
//  ZAEHIGKEIT: LEBENSPUNKTE, UND NUR NOTFALLS DAMAGETAKEN
//
//  Die erste Fassung machte das umgekehrt und lief ueber
//  DamageTaken - rechnerisch dasselbe, aber unsichtbar. Der
//  Poring behielt seine 50 auf der Leiste und starb bei
//  Spielern, die fuer Milliarden schlagen, weiterhin im ersten
//  Schlag. Ein Ereignis, von dem man nur den Namen sieht, ist
//  keines.
//
//  Jetzt bekommt jedes Monster seine Lebenspunkte mal .faktor,
//  sichtbar: aus 50 werden 500.
//
//  DIE GRENZE, DIE DAS NICHT UEBERALL ZULAESST
//
//  setunitdata liest seinen Wert als int32
//  (script.cpp: "int32 type = script_getnum(st, 3), value = 0;").
//  Ende der Fahnenstange ist damit 2.147.483.647. In
//  mob_db.yml stehen Lebenspunkte dagegen als uint32
//  (mob.cpp, asUInt32) und duerfen bis 4.294.967.295 gehen -
//  und genau dort stehen auf diesem Server die harten Brocken.
//
//  Ein "UMOB_MAXHP mal zehn" auf den Elephantmagon mit seinen
//  4,2 Milliarden waere also keine Verzehnfachung, sondern ein
//  Ueberlauf und damit eine HALBIERUNG. Das staerkste Monster
//  des Servers waere ausgerechnet im Haerte-Ereignis das
//  schwaechste - ohne Warntext, ohne Logeintrag.
//
//  Fuer die 151 Arten oberhalb von .hp_grenze bleibt es deshalb
//  bei DamageTaken: ein Prozentsatz auf den eingehenden Schaden
//  (max(dmg*DT/100, 1)), von der Groesse der Lebenspunkte
//  unabhaengig. DT 100 -> 10 heisst, das Monster haelt genau
//  zehnmal so lange. Bei einem Boss mit Milliarden erzaehlt die
//  Leiste ohnehin nichts mehr, dort faellt das Unsichtbare
//  nicht ins Gewicht.
//
//  Wer schon in mob_db.yml ein DamageTaken stehen hat - das
//  sind derzeit 217 Eintraege in db/re/mob_db.yml -, wird von
//  dort aus geteilt, nicht von 100 aus. Der Mindestwert ist 1;
//  darunter geht die Engine nicht, und der Mindestschaden
//  bleibt ohnehin immer 1. Unverwundbar wird hier keiner.
//
//
//  WARUM DER SCHADEN NICHT IMMER GANZE ZEHN SCHAFFT
//
//  setunitdata schreibt UMOB_ATKMIN und UMOB_ATKMAX in
//  status_data.rhw.atk/atk2, und die sind uint16 - Schluss bei
//  65535. Ein Monster mit 6553 Angriff oder weniger bekommt
//  seine vollen zehn; darueber kappt die Engine still.
//
//  Der hoechste Angriffswert in db/re/mob_db.yml ist 42597.
//  Dieses eine Monster kaeme also nur auf Faktor 1,5.
//
//  Das ist bewusst NICHT ueber die Angriffsgeschwindigkeit
//  ausgeglichen, obwohl das rechnerisch ginge. Zwei Gruende:
//  UMOB_ADELAY laesst sich hinterher nicht sauber
//  zuruecksetzen, weil getmonsterinfo den Wert gar nicht
//  herausgibt (es gibt kein MOB_ADELAY) - das Monster bliebe
//  also nach dem Ereignis schnell. Und zehnfache
//  Schlagfrequenz auf jedem Monster des Servers ist eine
//  Paketflut, die niemand gebraucht hat.
//
//  Wie oft die Grenze wirklich greift, sagt "@enraged status".
//  Wenn dort eine grosse Zahl steht, ist das eine Aussage ueber
//  mob_db.yml und keine ueber dieses Skript.
//
//
//  WORAN DAS SKRIPT ERKENNT, WAS ES SCHON HATTE
//
//  Am Vorsatz "Enraged" im Namen, und an sonst nichts. Das ist
//  keine Auswahl, sondern eine Sperre gegen die ZWEITE
//  Anwendung: ohne sie stiege der Angriff bei jedem Durchlauf
//  weiter, bis die uint16-Grenze alles einebnet.
//
//  Der Vorsatz taugt dafuer, weil er vorne steht. md->name ist
//  char[24]; bei einem langen Monsternamen schneidet die Engine
//  hinten ab - "Enraged Ancient Mimic" wird gekuerzt, aber
//  "Enraged " bleibt. Ein Vergleich auf den ganzen Namen waere
//  bei 639 der 2675 Arten fehlgeschlagen, und die haetten das
//  Ereignis nie wieder verlassen.
//
//  DAS ZURUECKSETZEN IST DAFUER UNGENAU
//
//  Hergestellt wird der DATENBANKZUSTAND, nicht der Zustand von
//  vorher. Fuer jedes gewoehnliche Monster ist das dasselbe.
//  Fuer die von anderen Skripten aufgewerteten nicht - die
//  stehen danach auf mob_db-Werten und heissen wieder wie in
//  mob_db.
//
//  Die Alternative waere ein Merkfeld ueber Spielnummern
//  gewesen. Bei zehntausenden lebenden Monstern und einem
//  Respawn im Sekundentakt waeren das hunderttausend Eintraege,
//  die einen @reloadscript ueberleben muessen und bei jedem
//  Absturz verwaisen - fuer eine Genauigkeit, die nur Monster
//  betrifft, die sich ohnehin binnen Minuten selbst aufloesen.
//  Die Begruendung im einzelnen steht bei S_Beruhigen.
//
//
//  WARUM IN HAEPPCHEN GEARBEITET WIRD
//
//  rAthena stellt die Monster ALLER Karten beim Start auf,
//  nicht erst, wenn jemand hinkommt. getunits(BL_MOB) liefert
//  deshalb je nach Ausbau fuenfstellig viele Spielnummern.
//  Alle in einem Durchgang umzuwerten waere ein spuerbarer
//  Hänger fuer jeden, der gerade spielt.
//
//  Stattdessen wird je Takt ein Stueck der Liste abgearbeitet
//  (.haeppchen). Ist die Liste durch, wird sie neu geholt -
//  damit sind auch die Monster erfasst, die inzwischen
//  nachgewachsen sind. Ein voller Durchlauf dauert dadurch ein
//  paar Sekunden; so lange kann ein frisch erschienenes
//  Monster noch normal sein. Das ist der bewusst gezahlte
//  Preis dafuer, dass niemand ruckelt.
//
//
//  WAS EIN ABSTURZ ANRICHTEN KOENNTE
//
//  Die Erfahrungs- und Beutesaetze sind Servereinstellungen,
//  keine Monsterwerte. Stuerzt der Server mitten im Ereignis
//  ab, stehen sie beim Hochfahren immer noch zehnfach - und
//  niemand merkt es, weil die Monster nach dem Neustart wieder
//  normal sind.
//
//  Deshalb liegen die urspruenglichen Saetze in DAUERHAFTEN
//  Variablen ($safa_enr_alt), zusammen mit dem Endzeitpunkt
//  ($safa_enr_bis). OnInit sieht beim Hochfahren nach: ist die
//  Zeit abgelaufen, werden die Saetze zurueckgesetzt; laeuft
//  sie noch, wird das Ereignis fortgesetzt.
//
//===========================================================


-	script	safa_enraged	-1,{
	end;


// ------------------------------------------------------------
//  @enraged - starten, beenden, nachsehen
// ------------------------------------------------------------
//    @enraged an [minuten]   startet, ohne Angabe .std_dauer
//    @enraged aus            beendet sofort und setzt zurueck
//    @enraged status         Was gilt gerade, und was nicht
//    @enraged probe          Was steht wirklich am Monster
//
//  "status" ist der eigentlich wichtige Befehl. Ein Ereignis,
//  das zehntausend Monster ueber mehrere Sekunden verteilt
//  umwertet, laesst sich sonst nicht beurteilen: man sieht ein
//  paar wuetende Poringe und weiss nicht, ob der Rest schon
//  dran war, ob die Saetze stehen und ob die Angriffsgrenze
//  zugeschlagen hat.
OnGM:
	.@was$ = "";
	if (.@atcmd_numparameters > 0)
		.@was$ = .@atcmd_parameters$[0];

	// --- starten ---------------------------------------------
	if (.@was$ == "an") {
		if ($safa_enr_bis > 0) {
			dispbottom "[Enraged] Laeuft bereits, noch "
			           + (($safa_enr_bis - gettimetick(2)) / 60)
			           + " Minuten. Erst \"@enraged aus\".";
			end;
		}
		.@min = .std_dauer;
		if (.@atcmd_numparameters > 1)
			.@min = atoi(.@atcmd_parameters$[1]);
		if (.@min < 1) .@min = 1;
		if (.@min > .max_dauer) .@min = .max_dauer;

		callsub S_Start, .@min;
		dispbottom "[Enraged] Gestartet fuer " + .@min + " Minuten.";
		dispbottom "  Der Durchlauf braucht ein paar Sekunden. "
		           + "\"@enraged status\" zeigt den Fortschritt.";
		end;
	}

	// --- beenden ---------------------------------------------
	if (.@was$ == "aus") {
		if ($safa_enr_bis == 0 && $@enr_modus == 0) {
			dispbottom "[Enraged] Laeuft gerade nicht.";
			end;
		}
		callsub S_Ende;
		dispbottom "[Enraged] Beendet. Das Beruhigen laeuft jetzt "
		           + "im Hintergrund weiter.";
		end;
	}

	// --- Nachmessen statt nachfuehlen ------------------------
	//  Der Grund, warum es diesen Befehl gibt: "man merkt
	//  nichts" ist keine Beobachtung, mit der sich arbeiten
	//  laesst. Auf einem Server, auf dem Spieler fuer
	//  Milliarden schlagen, ist ein Poring mit 50 oder mit 500
	//  Lebenspunkten derselbe eine Schlag - der Unterschied
	//  steht in den Zahlen, nicht im Kampfgefuehl.
	//
	//  Deshalb zeigt "probe" nebeneinander, was am Monster steht
	//  und was in mob_db steht. Stimmen die beiden ueberein,
	//  greift das Ereignis nicht. Weichen sie um .faktor ab,
	//  greift es - und dann ist die Frage eine ueber die
	//  Groessenordnung und nicht mehr ueber das Skript.
	if (.@was$ == "probe") {
		.@karte$ = strcharinfo(3);
		.@n = getmapunits(BL_MOB, .@karte$, .@m[0]);
		dispbottom "[Enraged] " + .@karte$ + ": " + .@n + " Monster.";
		.@z = 0;
		for (.@i = 0; .@i < .@n && .@z < 5; .@i++) {
			if (!unitexists(.@m[.@i])) continue;
			getunitdata .@m[.@i], .@d;
			.@k = .@d[UMOB_CLASS];
			dispbottom "  " + getunitname(.@m[.@i]);
			dispbottom "     jetzt : HP " + .@d[UMOB_HP] + "/"
			           + .@d[UMOB_MAXHP]
			           + "  ATK " + .@d[UMOB_ATKMIN] + "-"
			           + .@d[UMOB_ATKMAX]
			           + "  DmgTaken " + .@d[UMOB_DAMAGETAKEN];
			dispbottom "     mob_db: HP " + getmonsterinfo(.@k, MOB_MAXHP)
			           + "  ATK " + getmonsterinfo(.@k, MOB_ATKMIN) + "-"
			           + getmonsterinfo(.@k, MOB_ATKMAX);
			.@z++;
		}
		if (.@z == 0)
			dispbottom "  Hier steht gerade keines.";
		end;
	}

	// --- Status ----------------------------------------------
	dispbottom "[Enraged] Skript ist geladen.";

	if ($safa_enr_bis > 0) {
		.@rest = $safa_enr_bis - gettimetick(2);
		dispbottom "  Zustand: LAEUFT, noch " + (.@rest / 60) + " Min "
		           + (.@rest % 60) + " Sek";
	} else if ($@enr_modus == 2) {
		dispbottom "  Zustand: klingt ab - die Monster werden gerade "
		           + "wieder normal";
	} else {
		dispbottom "  Zustand: aus";
	}

	dispbottom "  Faktor: " + .faktor + "fach auf Zaehigkeit, Angriff, "
	           + "Erfahrung und Beute";

	// DIE ZAHL, DIE MAN WISSEN WILL.
	//
	// Zuerst und in einer eigenen Zeile, weil alles darunter nur
	// erklaert, wie sie zustande kommt. "5012 von 5012" heisst:
	// es sind alle. Steht dort dauerhaft etwas anderes, stimmt
	// etwas nicht - und dann helfen die Zeilen darunter weiter.
	if ($@enr_stand_alle > 0) {
		dispbottom "  ^0000CCWuetend: " + $@enr_stand_wach + " von "
		           + $@enr_stand_alle + " Monstern^000000"
		           // Eine kleine Luecke ist der Normalfall und kein
		           // Fehler: zwischen zwei Durchlaeufen wachsen
		           // Monster nach, und die sind eine Sekunde lang
		           // noch normal. Erst eine Luecke, die BLEIBT,
		           // ist eine.
		           + (($@enr_stand_wach >= $@enr_stand_alle)
		              ? "  - das sind alle"
		              : "  - der Rest ist gerade nachgewachsen")
		           + "  (gezaehlt vor "
		           + (gettimetick(2) - $@enr_stand_zeit) + " s)";
	} else {
		dispbottom "  Wuetend: noch nicht gezaehlt - der erste "
		           + "Durchlauf ist noch nicht durch.";
	}

	// Der Cursor. Er sagt NICHT, wie viele wuetend sind, sondern
	// wie weit das Haeppchen dieser Sekunde gekommen ist. Genau
	// diese Verwechslung hat schon einmal wie ein Fehler
	// ausgesehen, deshalb steht sie jetzt dabei.
	dispbottom "  Haeppchen gerade bei " + $@enr_pos + " von "
	           + getarraysize($@enr_liste)
	           + " - das ist der Cursor, nicht die Menge";
	dispbottom "  Neu umgewertet: " + $@enr_wach
	           + ", zurueckgesetzt: " + $@enr_ruhig;

	// Der Herzschlag. Ein Takt je Sekunde; zweimal "status" kurz
	// hintereinander muss eine groessere Zahl zeigen. Tut es das
	// nicht, steht der Timer - und dann ist jede andere Zahl hier
	// eingefroren und sagt gar nichts mehr.
	dispbottom "  Takte seit dem Start: " + $@enr_takte
	           + "  (zweimal nachsehen - die Zahl MUSS steigen)";

	// Die Zahl gehoert hierher, weil sie die einzige Antwort auf
	// "warum ist das Monster vor mir noch normal" ist, die nicht
	// "warte den Durchlauf ab" lautet: es gibt keine mehr.
	// Ausgenommen wird nichts.
	dispbottom "  Ausgenommen: nichts - MVPs, Minibosse, Adds, "
	           + "Wildlife und Instanzen inbegriffen";

	// Siehe Kopf: das ist eine Aussage ueber mob_db.yml.
	// Wer zu gross fuer sichtbare Lebenspunkte ist, siehe
	// S_Wecken. Bei diesen Monstern bleibt die Leiste, wie sie
	// war - sie stecken nur ein Zehntel des Schadens ein.
	dispbottom "  Ueber DamageTaken statt Lebenspunkte: " + $@enr_dtweg;
	dispbottom "  An der Angriffsgrenze (" + .atk_deckel + ") gekappt: "
	           + $@enr_deckel
	           + (($@enr_deckel > 0)
	              ? "  <-- die kommen nicht auf Faktor " + .faktor : "");

	dispbottom "  Saetze gerade: Basis-EXP " + getbattleflag("base_exp_rate")
	           + ", Job-EXP " + getbattleflag("job_exp_rate")
	           + ", Beute " + getbattleflag("item_rate_common");
	dispbottom "  Gemerkte Saetze: " + getarraysize($safa_enr_alt)
	           + ((getarraysize($safa_enr_alt) > 0 && $safa_enr_bis == 0)
	              ? "  <-- STEHT NOCH, obwohl aus. Bitte melden." : "");
	end;


// ------------------------------------------------------------
//  Das Ereignis anwerfen
// ------------------------------------------------------------
S_Start:
	.@min = getarg(0);

	// Reihenfolge: erst merken, dann hochsetzen. Umgekehrt
	// merkte man sich den bereits verzehnfachten Wert und
	// das Zuruecksetzen liefe ins Leere - beim zweiten
	// Ereignis waere der Server dann dauerhaft auf 100fach.
	callsub S_RatenMerken;
	callsub S_RatenSetzen, .faktor;

	// Der Endzeitpunkt ist DAUERHAFT, damit er einen Neustart
	// ueberlebt. gettimetick(2) ist die Unixzeit, nicht die
	// Laufzeit des Servers - genau deshalb taugt sie dafuer.
	$safa_enr_bis = gettimetick(2) + .@min * 60;
	$@enr_gewarnt = 0;

	deletearray $@enr_liste[0];
	$@enr_pos        = 0;
	$@enr_wach       = 0;
	$@enr_ruhig      = 0;
	$@enr_deckel     = 0;
	$@enr_takte      = 0;
	$@enr_p_wach     = 0;
	$@enr_p_gesehen  = 0;
	$@enr_stand_wach = 0;
	$@enr_stand_alle = 0;
	$@enr_dtweg      = 0;
	$@enr_modus      = 1;

	announce "The Wild - Enraged. For the next " + .@min
	         + " minutes every creature out there hits " + .faktor
	         + " times harder and takes " + .faktor
	         + " times longer to fall. So does what it leaves behind.",
	         bc_all, "0xFF3300";
	return;


// ------------------------------------------------------------
//  Das Ereignis abblasen
// ------------------------------------------------------------
//  Die Saetze gehen SOFORT zurueck, die Monster nicht - die
//  brauchen wieder ihre Haeppchen. Diese Reihenfolge ist
//  Absicht: ein Spieler, der in der Abklingphase noch ein
//  wuetendes Monster legt, bekommt normale Beute. Andersherum -
//  normale Monster bei zehnfacher Beute - waere die Einladung,
//  genau in diesem Fenster zu farmen.
S_Ende:
	callsub S_RatenZurueck;
	$safa_enr_bis = 0;

	deletearray $@enr_liste[0];
	$@enr_pos   = 0;
	$@enr_ruhig = 0;
	$@enr_modus = 2;

	announce "The Wild settles. The creatures are catching their breath.",
	         bc_all, "0x888888";
	return;


// ------------------------------------------------------------
//  Der Takt
// ------------------------------------------------------------
//  Ein Takt macht genau drei Dinge, in dieser Reihenfolge:
//  auf Ablauf pruefen, warnen, ein Haeppchen abarbeiten.
//
//  Der Timer laeuft immer, auch wenn nichts los ist. Ein
//  Vergleich zweier Zahlen je Sekunde kostet nichts, und ein
//  Timer, den man startet und stoppt, ist eine Stelle mehr,
//  an der er nach einem @reloadscript nicht wieder anlaeuft.
OnTimer1000:
	setnpctimer 0;
	$@enr_takte++;

	if ($@enr_modus == 0) end;

	if ($@enr_modus == 1) {
		.@rest = $safa_enr_bis - gettimetick(2);

		if (.@rest <= 0) {
			callsub S_Ende;
			end;
		}

		// Einmal warnen, nicht jede Sekunde. $@enr_gewarnt
		// merkt sich, was schon raus ist.
		if (.@rest <= 300 && $@enr_gewarnt < 1) {
			$@enr_gewarnt = 1;
			announce "The Wild - Enraged: five minutes left.",
			         bc_all, "0xFF9900";
		}

		callsub S_Haeppchen, 1;
		end;
	}

	// Modus 2: beruhigen. Ist ein ganzer Durchlauf vorbei, ohne
	// dass noch ein einziges Monster zurueckzusetzen war, steht
	// keines mehr - dann ist das Ereignis wirklich vorbei.
	//
	// Der zweite Fall - gar keine Monster in der Liste - kommt
	// auf einem laufenden Server nicht vor, faengt aber den
	// Zustand ab, in dem der Takt sonst fuer immer jede Sekunde
	// erfolglos alle Monster einsammelte.
	callsub S_Haeppchen, 0;
	if (getarraysize($@enr_liste) == 0
	    || ($@enr_pos >= getarraysize($@enr_liste) && $@enr_ruhig == 0)) {
		$@enr_modus = 0;
		deletearray $@enr_liste[0];
		$@enr_pos = 0;
	}
	end;


// ------------------------------------------------------------
//  Ein Haeppchen der Monsterliste
// ------------------------------------------------------------
//  getarg(0): 1 = wecken, 0 = beruhigen.
//
//  WARUM DIE LISTE ZWISCHENGESPEICHERT WIRD
//
//  getunits(BL_MOB) laeuft ueber alle Monster des Servers.
//  Das je Takt zu tun, nur um danach ein Fuenftel davon
//  anzufassen, waere die teuerste Zeile der Datei. Also
//  einmal holen, ueber mehrere Takte abarbeiten, dann neu
//  holen. Das Neuholen ist gleichzeitig die Stelle, an der
//  nachgewachsene Monster dazukommen.
S_Haeppchen:
	.@wecken = getarg(0);

	if ($@enr_pos >= getarraysize($@enr_liste)) {
		// --- Ein Durchlauf ist zu Ende: Stand veroeffentlichen --
		//  Das hier ist die Zahl, die im Status stehen muss.
		//
		//  Vorher stand dort der Cursor - "2000 von 5000" - und
		//  das las sich wie "nur 2000 Monster sind wuetend". Es
		//  hiess aber nur "das Haeppchen dieser Sekunde ist bis
		//  Position 2000 gekommen"; eine Sekunde spaeter stand da
		//  4000, dann 5000, dann wieder 2000.
		//
		//  Der Cursor beantwortet die Frage nicht, die man beim
		//  Nachsehen hat. Die lautet: sind es ALLE? Also wird
		//  waehrend des Durchlaufs mitgezaehlt, wie viele die
		//  Marke tragen und wie viele es ueberhaupt gibt, und am
		//  Ende des Durchlaufs wird beides festgehalten.
		if (getarraysize($@enr_liste) > 0) {
			$@enr_stand_wach   = $@enr_p_wach;
			$@enr_stand_alle   = $@enr_p_gesehen;
			$@enr_stand_zeit   = gettimetick(2);
		}

		deletearray $@enr_liste[0];
		$@enr_pos      = 0;
		$@enr_ruhig    = 0;
		$@enr_p_wach   = 0;
		$@enr_p_gesehen = 0;
		.@gesamt = getunits(BL_MOB, $@enr_liste[0]);
		if (.@gesamt == 0) return;
	}

	.@bis = $@enr_pos + .haeppchen;
	if (.@bis > getarraysize($@enr_liste))
		.@bis = getarraysize($@enr_liste);

	// Ohne freeloop bricht rAthena bei 2048 Durchlaeufen ab -
	// mit einer Meldung im Log, die man leicht uebersieht,
	// weil das Skript danach einfach weiterlaeuft.
	freeloop(1);
	for (.@i = $@enr_pos; .@i < .@bis; .@i++) {
		.@gid = $@enr_liste[.@i];
		if (!.@gid) continue;
		if (!unitexists(.@gid)) continue;

		// HIER STEHT KEINE AUSNAHME MEHR.
		//
		// Frueher fielen an dieser Stelle drei Gruppen heraus:
		// Diener und Beschwoerungen (UMOB_MASTERAID), fremd
		// benannte Monster und solche mit veraenderten
		// Angriffswerten. Das war die vorsichtige Fassung - sie
		// fasste nur an, was sich garantiert exakt
		// zuruecksetzen liess.
		//
		// Jetzt gilt: alles, was lebt, wird wuetend. MVPs,
		// Minibosse, Adds, Wildlife-Rudel, Instanzmonster, die
		// Pflanzen eines Alchemisten. Was das kostet, steht
		// unten bei S_Beruhigen - es ist bezahlt, nicht
		// uebersehen.
		.@name$ = getunitname(.@gid);

		// Traegt es schon die Marke? Geprueft wird NUR der
		// Vorsatz, nicht der ganze Name - und das ist wichtig:
		// md->name ist char[24], "Enraged " belegt davon acht.
		// Bei einem langen Monsternamen schneidet die Engine
		// hinten ab. Ein Vergleich auf den vollen Namen
		// "Enraged " + Datenbankname schluege dann fehl, das
		// Monster wuerde am Ende nie beruhigt und behielte
		// seine zehnfachen Werte, bis es stirbt. Der Vorsatz
		// steht vorne und wird nie abgeschnitten.
		.@markiert = (substr(.@name$, 0, .plen - 1) == .praefix$);

		// Gezaehlt wird JEDES lebende Monster, das der Durchlauf
		// sieht - und getrennt davon, wie viele die Marke tragen.
		// Nach dem Wecken gleich unten wird nachgezaehlt, damit
		// die Zahl den Stand NACH diesem Haeppchen zeigt und
		// nicht den davor.
		$@enr_p_gesehen++;
		if (.@markiert) $@enr_p_wach++;

		// --- Ab hier wird es teuer, deshalb erst ab hier ------
		//  getunitdata fuellt ein Feld mit rund sechzig
		//  Eintraegen. Das je Monster und Sekunde zu tun, nur um
		//  gleich darauf am Vorsatz abzubiegen, waere bei
		//  viertausend Monstern viermal so viel Arbeit wie
		//  noetig - und zwar dauerhaft, denn nach dem ersten
		//  Takt sind ohnehin alle wach.
		//
		//  Der Vorsatz oben entscheidet ohne einen einzigen
		//  Datenbankzugriff. Erst wer wirklich angefasst wird,
		//  kostet etwas.
		if (.@wecken) {
			// Die einzige verbliebene Bedingung, und sie ist
			// keine Ausnahme, sondern die Sperre gegen die
			// zweite Anwendung: wer die Marke schon traegt, ist
			// durch. Ohne sie stiege der Angriff bei jedem
			// Durchlauf weiter, bis die uint16-Grenze alles
			// einebnet.
			if (.@markiert) continue;
		} else {
			if (!.@markiert) continue;
		}

		// FELD, NICHT EINZELWERT. getunitdata gibt nichts
		// zurueck, es FUELLT ein Feld:
		//     getunitdata <GID>,<feldname>;
		// Ein "getunitdata(.@gid, UMOB_CLASS)" uebergibt die
		// Konstante dort, wo eine Feldvariable stehen muss.
		// .@klasse blieb daraufhin unbrauchbar, getmonsterinfo
		// lieferte "null", und die Pruefung darunter hat JEDES
		// Monster uebersprungen - das Ereignis lief, zeigte
		// steigende Takte und wertete null Monster um.
		getunitdata .@gid, .@d;
		.@klasse = .@d[UMOB_CLASS];
		.@db$    = getmonsterinfo(.@klasse, MOB_NAME);

		if (.@wecken) {
			// Kein Ausschluss, sondern eine Unmoeglichkeit: ohne
			// Datenbankeintrag gibt es keinen Namen, auf den am
			// Ende zurueckgesetzt werden koennte.
			if (.@db$ == "null") continue;
			callsub S_Wecken, .@gid, .@klasse, .@db$,
			        .@d[UMOB_DAMAGETAKEN], .@d[UMOB_ATKMIN],
			        .@d[UMOB_ATKMAX], .@d[UMOB_MAXHP];
			$@enr_p_wach++;   // ab jetzt traegt es die Marke
		} else {
			// Beim Beruhigen ist der Datenbankname nur Notnagel.
			// Fehlt er, bleibt immer noch der abgeschnittene
			// Vorsatz - besser als das Monster wuetend stehen
			// zu lassen.
			if (.@db$ == "null") .@db$ = .@name$;
			callsub S_Beruhigen, .@gid, .@klasse, .@db$, .@name$,
			        .@d[UMOB_ATKMIN], .@d[UMOB_ATKMAX],
			        .@d[UMOB_HP], .@d[UMOB_MAXHP];
			$@enr_p_wach--;   // Marke ist ab jetzt weg
		}
	}
	freeloop(0);

	$@enr_pos = .@bis;
	return;


// ------------------------------------------------------------
//  Ein Monster wuetend machen
// ------------------------------------------------------------
S_Wecken:
	.@gid     = getarg(0);
	.@klasse  = getarg(1);
	.@db$     = getarg(2);
	// Durchgereicht statt hier noch einmal gelesen - S_Haeppchen
	// hat das Feld ohnehin schon geholt.
	.@dt_alt  = getarg(3);
	.@a1_alt  = getarg(4);
	.@a2_alt  = getarg(5);
	.@hp_alt  = getarg(6);   // MaxHP, wie es jetzt dasteht

	// --- Zaehigkeit: zwei Wege, und der erste ist der sichtbare
	//
	//  LEBENSPUNKTE, WO SIE PASSEN
	//
	//  Ein Poring mit 50 Lebenspunkten soll 500 haben, und zwar
	//  auf der Leiste. DamageTaken taete dasselbe, aber
	//  unsichtbar - die Leiste faellt langsamer, die Zahl bleibt.
	//  Das fuehlt sich nicht nach einem Ereignis an, sondern nach
	//  einem zaehen Nachmittag.
	//
	//  DAMAGETAKEN, WO SIE NICHT PASSEN
	//
	//  setunitdata liest seinen Wert als int32 (script.cpp:
	//  "int32 type = script_getnum(st, 3), value = 0"), Ende bei
	//  2.147.483.647. In mob_db.yml stehen Lebenspunkte als
	//  uint32 und duerfen bis 4.294.967.295 gehen.
	//
	//  Der Elephantmagon hat 4,2 Milliarden. Mal zehn waere das
	//  keine Verzehnfachung, sondern ein Ueberlauf - und heraus
	//  kaeme eine HALBIERUNG. Das staerkste Monster des Servers
	//  waere ausgerechnet im Haerte-Ereignis das schwaechste,
	//  ohne Warnung und ohne Logeintrag.
	//
	//  151 der 2662 Monsterarten liegen ueber der Grenze - der
	//  Elephantmagon und die Trainingspuppen mit je 2
	//  Milliarden. Fuer die bleibt es beim unsichtbaren Weg: sie
	//  sind ohnehin so gross, dass die Leiste dort nichts mehr
	//  erzaehlt.
	//
	//  DER ZWISCHENSPEICHER IST GLEICHZEITIG DIE WEGMARKE
	//
	//  $@enr_dt[.@klasse] wird NUR auf dem DamageTaken-Weg
	//  gesetzt. S_Beruhigen liest daran ab, welcher Weg gegangen
	//  wurde, und muss die Entscheidung nicht neu treffen - sie
	//  waere dort auch gar nicht mehr zu treffen, weil die
	//  Lebenspunkte inzwischen zehnmal so hoch stehen.
	//
	//  Die Ablage ist um eins verschoben, weil ein leeres
	//  Feldelement in rAthena 0 liefert und 0 ein gueltiges
	//  DamageTaken waere. So heisst 0 verlaesslich "nicht auf
	//  diesem Weg".
	if (.@hp_alt > 0 && .@hp_alt <= .hp_grenze) {
		// status_set_maxhp HEILT die Differenz von selbst
		// (status.cpp: heal = maxhp - status->max_hp). Ein
		// zusaetzliches setunitdata UMOB_HP danach waere eine
		// zweite Heilung - genau der Griff, der den Biomboss
		// einmal unsterblich gemacht hat. Also nur MAXHP.
		setunitdata .@gid, UMOB_MAXHP, .@hp_alt * .faktor;
	} else {
		if ($@enr_dt[.@klasse] == 0)
			$@enr_dt[.@klasse] = .@dt_alt + 1;

		.@dt = ($@enr_dt[.@klasse] - 1) / .faktor;
		if (.@dt < 1) .@dt = 1;   // darunter kennt die Engine nichts
		setunitdata .@gid, UMOB_DAMAGETAKEN, .@dt;
		$@enr_dtweg++;
	}

	// --- Schaden ---------------------------------------------
	//  Aus dem AKTUELLEN Wert des Monsters gerechnet, nicht aus
	//  dem Datenbankwert.
	//
	//  Das war vorher andersherum, und die Umstellung gehoert zu
	//  "keine Ausnahmen": ein Wildlife-Anfuehrer traegt einen
	//  Angriffswert, den safa_wildlife am Leben des Spielers
	//  ausgerechnet hat - der liegt weit ueber allem, was in
	//  mob_db steht. Vom Datenbankwert aus verzehnfacht haette
	//  dieses Skript ihn mitten im Ereignis SCHWAECHER gemacht.
	//
	//  Die Sperre gegen doppelte Anwendung ist damit allein die
	//  Marke im Namen. Sie reicht: der Vorsatz steht vorne und
	//  wird von der Namenslaenge nie abgeschnitten.
	.@atk1 = .@a1_alt * .faktor;
	.@atk2 = .@a2_alt * .faktor;

	if (.@atk1 > .atk_deckel || .@atk2 > .atk_deckel)
		$@enr_deckel++;
	if (.@atk1 > .atk_deckel) .@atk1 = .atk_deckel;
	if (.@atk2 > .atk_deckel) .@atk2 = .atk_deckel;

	setunitdata .@gid, UMOB_ATKMIN, .@atk1;
	setunitdata .@gid, UMOB_ATKMAX, .@atk2;

	// --- Der Name ist die Marke ------------------------------
	//  Zuletzt, nicht zuerst. Bricht eine der Zeilen darueber
	//  ab, traegt das Monster sonst schon die Marke, ohne die
	//  Werte zu haben - und wird nie wieder angefasst, weil die
	//  Sperre oben greift.
	setunitname .@gid, .praefix$ + .@db$;
	$@enr_wach++;
	return;


// ------------------------------------------------------------
//  Ein Monster wieder beruhigen
// ------------------------------------------------------------
//  Spiegelbildlich, und deshalb erst der Name zum Schluss:
//  bleibt hier etwas liegen, wird dasselbe Monster im naechsten
//  Takt noch einmal versucht.
//
//  ZURUECKGERECHNET, NICHT NACHGESCHLAGEN
//
//  Jeder der drei Werte wird umgekehrt, wie er gesetzt wurde:
//  Angriff geteilt statt mal, Vorsatz abgeschnitten statt neu
//  gesetzt, DamageTaken aus dem Zwischenspeicher der Art.
//
//  Die erste Fassung holte Name und Angriff aus mob_db. Das war
//  fuer gewoehnliche Monster richtig und fuer alle anderen
//  falsch: ein Wildlife-Rudel, das ein Ereignis ueberlebte,
//  stand hinterher auf Datenbankwerten und hiess wieder
//  "Poring" - es war also SCHWAECHER als vor dem Ereignis, und
//  seine Herkunft war nicht mehr zu sehen.
//
//  Die Datenbank ist jetzt nur noch der Notnagel fuer die
//  beiden Faelle, in denen der Ausgangswert wirklich verloren
//  ist: der an der uint16-Grenze gekappte Angriff und der auf
//  23 Zeichen abgeschnittene Name. Beide sind unten benannt.
S_Beruhigen:
	.@gid        = getarg(0);
	.@klasse     = getarg(1);
	.@db$        = getarg(2);
	.@name_jetzt$ = getarg(3);   // wie es JETZT heisst, mit Vorsatz
	.@a1_jetzt   = getarg(4);
	.@a2_jetzt   = getarg(5);
	.@hp_jetzt   = getarg(6);
	.@max_jetzt  = getarg(7);

	// --- Zaehigkeit: denselben Weg zurueck ---------------------
	//  Welcher Weg gegangen wurde, steht im Zwischenspeicher:
	//  gesetzt heisst DamageTaken, leer heisst Lebenspunkte.
	//  Siehe S_Wecken.
	if ($@enr_dt[.@klasse] > 0) {
		setunitdata .@gid, UMOB_DAMAGETAKEN, $@enr_dt[.@klasse] - 1;
	} else {
		// DIE REIHENFOLGE IST HIER LEBENSWICHTIG - woertlich.
		//
		// status_set_maxhp zieht die Differenz beim Absenken
		// wieder ab (status.cpp: status_zap(bl, -heal, 0)), und
		// status_damage deckelt den Abzug auf die AKTUELLEN
		// Lebenspunkte und toetet dann. Ein Monster, das gerade
		// halb heruntergeschlagen ist, faellt beim Absenken also
		// einfach um: 250 von 500 minus 450 ist unter null.
		//
		// Deshalb erst auffuellen, dann absenken, dann den
		// Anteil wiederherstellen. Weil die Obergrenze durch
		// .faktor geteilt wird, ist der Anteil schlicht der
		// durch .faktor geteilte Stand - keine Bruchrechnung,
		// kein Ueberlauf.
		.@ziel = .@hp_jetzt / .faktor;
		if (.@ziel < 1) .@ziel = 1;
		setunitdata .@gid, UMOB_HP,    .@max_jetzt;
		setunitdata .@gid, UMOB_MAXHP, .@max_jetzt / .faktor;
		setunitdata .@gid, UMOB_HP,    .@ziel;
	}

	// --- Angriff: geteilt, nicht nachgeschlagen ---------------
	//  S_Wecken hat mal .faktor gerechnet, also ist das Teilen
	//  die exakte Umkehrung - fuer JEDES Monster, auch fuer die
	//  von anderen Skripten aufgewerteten. Der Umweg ueber
	//  getmonsterinfo hatte ein Wildlife-Rudel hinterher auf
	//  mob_db-Werte gesetzt und damit schwaecher gemacht, als es
	//  vor dem Ereignis war.
	//
	//  Nur wo die uint16-Grenze zugeschlagen hat, ist der
	//  Ausgangswert nicht mehr im Wert enthalten. Dort bleibt die
	//  Datenbank die beste verfuegbare Auskunft. Erkennbar ist
	//  der Fall daran, dass der Wert GENAU auf der Grenze steht.
	setunitdata .@gid, UMOB_ATKMIN,
	    (.@a1_jetzt >= .atk_deckel) ? getmonsterinfo(.@klasse, MOB_ATKMIN)
	                                : .@a1_jetzt / .faktor;
	setunitdata .@gid, UMOB_ATKMAX,
	    (.@a2_jetzt >= .atk_deckel) ? getmonsterinfo(.@klasse, MOB_ATKMAX)
	                                : .@a2_jetzt / .faktor;

	// --- Name: Vorsatz abschneiden ---------------------------
	//  Auch hier die Umkehrung statt der Datenbank, und aus zwei
	//  Gruenden:
	//
	//  Erstens behaelt ein Wildlife-Rudel so seinen Namen - aus
	//  "Enraged Wildlife Poring" wird wieder "Wildlife Poring"
	//  und nicht "Poring".
	//
	//  Zweitens ist der Datenbankname bei den 14 Arten aus
	//  mob_avail.yml gar nicht der richtige: getunitdata liefert
	//  ueber UMOB_CLASS die SICHTKLASSE (script.cpp:19111,
	//  md->vd->look[LOOK_BASE]), nicht die Monsternummer. Wo
	//  beide auseinandergehen, haette getmonsterinfo den Namen
	//  des Sprite-Monsters geliefert - und den haette das
	//  Monster dann dauerhaft behalten.
	//
	//  Die Ausnahme ist der abgeschnittene Name: md->name ist
	//  char[24], laenger als 23 Zeichen wird nichts. Steht der
	//  Name genau auf 23, ist er mit hoher Wahrscheinlichkeit
	//  gekuerzt und sein Ende unwiederbringlich - dann ist die
	//  Datenbank wieder die beste Auskunft.
	if (getstrlen(.@name_jetzt$) >= 23)
		setunitname .@gid, .@db$;
	else
		setunitname .@gid, substr(.@name_jetzt$, .plen,
		                          getstrlen(.@name_jetzt$) - 1);
	$@enr_ruhig++;
	return;


// ------------------------------------------------------------
//  Erfahrung und Beute
// ------------------------------------------------------------
//  Das sind Servereinstellungen, keine Monsterwerte - sie
//  wirken auf alles, was in diesem Zeitraum faellt, auch auf
//  die Monster, die der Durchlauf noch nicht erreicht hat.
//  Genau so ist es gemeint: das Ereignis ist ein Zeitraum,
//  kein Ort.
//
//  KARTEN UND TRUHEN SIND NICHT DABEI
//
//  item_rate_card und item_rate_treasure fehlen in der Liste
//  unten mit Absicht. Der Server steht bei 2500 - zehnfach
//  waeren 25000, und item_drop_card_max deckelt bei 10000,
//  also 100 Prozent. Jedes Monster liesse in einer Stunde
//  seine Karte fallen. Was danach an Karten im Umlauf ist,
//  bekommt man nicht mehr eingesammelt; die Erfahrung dieser
//  Stunde bleibt dagegen auf den Konten, wo sie hingehoert.
//
//  Wer es trotzdem will, haengt die beiden Namen unten an. Die
//  Mechanik traegt sie mit - es ist eine Entscheidung ueber
//  die Wirtschaft, keine ueber das Skript.
S_RatenMerken:
	// Nur wenn wirklich nichts gemerkt ist. Sonst ueberschriebe
	// ein zweites Anwerfen (oder ein OnInit nach @reloadscript)
	// die echten Ausgangswerte mit den verzehnfachten.
	if (getarraysize($safa_enr_alt) > 0) return;
	for (.@i = 0; .@i < getarraysize(.rate$); .@i++)
		$safa_enr_alt[.@i] = getbattleflag(.rate$[.@i]);
	return;

S_RatenSetzen:
	.@f = getarg(0);
	for (.@i = 0; .@i < getarraysize(.rate$); .@i++)
		setbattleflag .rate$[.@i], $safa_enr_alt[.@i] * .@f;
	return;

S_RatenZurueck:
	if (getarraysize($safa_enr_alt) == 0) return;
	for (.@i = 0; .@i < getarraysize(.rate$); .@i++)
		setbattleflag .rate$[.@i], $safa_enr_alt[.@i];
	deletearray $safa_enr_alt[0];
	return;


// ------------------------------------------------------------
//  Einstellungen
// ------------------------------------------------------------
OnInit:
	bindatcmd "enraged", "safa_enraged::OnGM", 99, 99;

	// --- Was sich lohnt zu drehen ----------------------------
	.faktor      = 10;    // Zaehigkeit, Schaden, Erfahrung, Beute
	.std_dauer   = 60;    // Minuten, wenn @enraged an ohne Zahl kommt
	.max_dauer   = 720;   // Obergrenze, damit kein Tippfehler zwoelf
	                      // Tage Ereignis daraus macht
	.praefix$    = "Enraged ";
	.plen        = getstrlen(.praefix$);   // einmal, nicht je Monster

	// --- Der Durchsatz ---------------------------------------
	//  .haeppchen ist die einzige Zahl, die den Server belasten
	//  kann. Sie sagt, wie viele Monster je Sekunde angesehen
	//  werden - angesehen, nicht umgewertet; das meiste faellt
	//  gleich an der Namenspruefung weg.
	//
	//  Der Wert stand auf 2000 und war damit zu vorsichtig
	//  gewaehlt: er stammte aus der Annahme, hier stuenden rund
	//  20000 Monster. Tatsaechlich sind es gut 5000, und bei 2000
	//  je Takt brauchte ein voller Durchlauf drei Sekunden.
	//
	//  Das war nicht falsch, aber es sah falsch aus. "@enraged
	//  status" zeigte den Cursor - "2000 von 5000" -, und das
	//  liest sich wie "nur 2000 Monster sind wuetend". Ein Wert
	//  ueber dem Bestand macht aus dem Durchlauf einen einzigen
	//  Takt: der Stand im Status ist danach immer vollstaendig,
	//  und ein nachgewachsenes Monster ist hoechstens eine
	//  Sekunde lang normal.
	//
	//  Teuer ist nicht das Ansehen, sondern das Umwerten: jedes
	//  setunitdata auf den Angriff zieht ein status_calc_bl nach
	//  sich. Das faellt genau EINMAL an, im ersten Takt nach dem
	//  Start - danach sind alle wach und die Namenspruefung wirft
	//  sie sofort wieder raus.
	//
	//  Waechst der Server, waechst diese Zahl mit. Ob sie zu hoch
	//  steht, sagt das map-Log und nicht das Spiel.
	.haeppchen   = 8000;

	// --- Die harte Grenze der Engine -------------------------
	//  status_data.rhw.atk ist uint16. Kein Stellwert, sondern
	//  eine Tatsache - hier nur benannt, damit sie im Skript
	//  einen Namen hat statt einer nackten Zahl.
	.atk_deckel  = 65535;

	// --- Ab wo Lebenspunkte nicht mehr gehen -----------------
	//  Alles bis hier bekommt sichtbare Lebenspunkte mal
	//  .faktor. Darueber wuerde der Wert beim Schreiben durch
	//  setunitdata ueber die int32-Grenze laufen (2.147.483.647)
	//  und dabei ins Gegenteil kippen - siehe S_Wecken. Fuer
	//  diese 151 Arten bleibt es beim unsichtbaren Weg ueber
	//  DamageTaken.
	//
	//  200 Millionen statt der rechnerischen 214.748.364: der
	//  Abstand ist Absicht. Ein Monster, dessen Lebenspunkte ein
	//  anderes Skript schon angehoben hat, soll nicht knapp
	//  darueber landen.
	.hp_grenze   = 200000000;

	// --- Was mitwaechst --------------------------------------
	//  Zur Auslassung von Karten und Truhen siehe S_RatenMerken.
	setarray .rate$[0],
	    "base_exp_rate",
	    "job_exp_rate",
	    "mvp_exp_rate",
	    "item_rate_common",  "item_rate_common_boss",  "item_rate_common_mvp",
	    "item_rate_heal",    "item_rate_heal_boss",    "item_rate_heal_mvp",
	    "item_rate_use",     "item_rate_use_boss",     "item_rate_use_mvp",
	    "item_rate_equip",   "item_rate_equip_boss",   "item_rate_equip_mvp",
	    "item_rate_mvp";

	// --- Aufraeumen nach einem Absturz -----------------------
	//  Siehe Kopf. Der Fall, den das abfaengt: der Server geht
	//  waehrend des Ereignisses hart aus. Die Monster sind nach
	//  dem Hochfahren von selbst wieder normal - die Saetze
	//  nicht, die stehen in der Konfiguration des laufenden
	//  Prozesses und waeren beim naechsten Start wieder
	//  zehnfach.
	//
	//  ACHTUNG, DER UNTERSCHIED ZU @reloadscript: dabei bleiben
	//  $@-Variablen und Monster stehen, $-Variablen ohnehin.
	//  Deshalb wird hier NICHT stumpf zurueckgesetzt, sondern
	//  nach der Uhr entschieden.
	if ($safa_enr_bis > 0) {
		if (gettimetick(2) >= $safa_enr_bis) {
			// Zeit war abgelaufen, waehrend nichts lief.
			callsub S_RatenZurueck;
			$safa_enr_bis = 0;
			$@enr_modus   = 2;   // falls doch Monster stehen
		} else {
			// Laeuft noch. Die Saetze koennen nach einem
			// Neustart auf den Ausgangswerten stehen - dann
			// muessen sie wieder hoch. Zweimal Hochsetzen kann
			// nicht passieren, weil immer von $safa_enr_alt
			// aus gerechnet wird und nie vom aktuellen Wert.
			callsub S_RatenSetzen, .faktor;
			$@enr_modus = 1;
		}
	} else if (getarraysize($safa_enr_alt) > 0) {
		// Kein Ereignis, aber gemerkte Saetze: das ist genau
		// der Absturz zwischen S_RatenMerken und S_Start.
		callsub S_RatenZurueck;
	}

	initnpctimer;
