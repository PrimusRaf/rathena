// ============================================================
//  SafaRO - GM-Werkbank
// ============================================================
//
//  Setzt alle fuenf Zufallsoptionen eines getragenen Stuecks auf
//  einen festen Wert und wertet auf Wunsch bis +30 auf.
//
//  Bedient werden ZWANZIG Plaetze: die zehn normalen, sechs
//  Shadow und vier Costume. Warum nicht einundzwanzig - EQI_AMMO
//  fehlt mit Absicht - steht weiter unten bei der Liste.
//
//  WOZU
//
//  Fuer Testgegenstaende gibt es keinen GM-Befehl. Die Liste in
//  atcommand.cpp kennt @item, @refine, @refineui und @itembound -
//  nichts davon setzt Zufallsoptionen. Der einzige Weg dorthin ist
//  setrandomoption, und das ist ein Skriptbefehl.
//
//  Ueber den Master Artisan kaeme man auch hin, aber dessen Werte
//  sind zufaellig aus einer Spanne. Bis fuenfmal die 100 faellt,
//  verzaubert man lange. Hier steht, was man eintippt.
//
//  GRENZEN, ALLE NACHGESCHLAGEN
//
//    MAX_ITEM_RDM_OPT  5      common/mmo.hpp:113
//    MAX_REFINE        30     map/status.hpp:37, von uns gesetzt
//    Wertebereich      int16  s_item_randomoption in mmo.hpp
//
//  Mehr als fuenf Optionen gibt es also nicht, und ueber 32767
//  laeuft der Wert ueber.
//
//  NUR FUER GM
//
//  Gruppenstufe 99. Der NPC steht in valkyrie, wo ohnehin kein
//  Spieler hinkommt - die Pruefung ist der zweite Riegel, nicht
//  der einzige.
// ============================================================

// Koordinate von Raffael, im Spiel abgelesen. Geprueft: begehbar,
// und der naechste NPC steht drei Felder weiter auf 48,86.
valkyrie,48,83,4	script	GM Workbench#safa	4_M_ALCHE_A,{

	if (getgroupid() < .stufe) {
		mes "[GM Workbench]";
		mes "This bench is not for you.";
		close;
	}

	mes "[GM Workbench]";
	mes "What are we doing?";
	next;

	if (select("Work on one piece:^0000FFSocket everything I am wearing^000000") == 2) {
		callsub S_AlleSockel;
		close;
	}

	mes "[GM Workbench]";
	mes "Which piece?";
	next;

	// ACHTUNG: feste Anzahl, NICHT getarraysize.
	//
	// EQI_ACC_L hat den Wert 0 (pc.hpp:75), und rAthena entfernt eine
	// Array-Zelle, sobald sie auf 0 gesetzt wird. Steht die Null am
	// Ende, meldet getarraysize eine zu kleine Zahl und der linke
	// Ring fehlt in der Liste. Genau dieser Fehler hat beim Master
	// Artisan monatelang gesteckt.
	//
	// ------------------------------------------------------------
	//  ZWANZIG PLAETZE, NICHT ZEHN
	// ------------------------------------------------------------
	//  Dazugekommen sind die sechs Shadow- und die vier
	//  Costume-Plaetze. Beide tragen Zufallsoptionen wie normale
	//  Ausruestung - nachgesehen in status.cpp:4147: die Schleife,
	//  die Optionen auswertet, laeuft ueber ALLE EQI von 0 bis
	//  EQI_MAX und ueberspringt genau eine, naemlich EQI_AMMO.
	//
	//  Deshalb steht EQI_AMMO auch hier nicht in der Liste. Man
	//  koennte einem Pfeil Optionen aufschreiben, sie wuerden nur
	//  nie gelesen - eine Zeile, die aussieht, als taete sie etwas.
	//
	//  Die Reihenfolge ist Ausruestung, Shadow, Costume, damit die
	//  drei Gruppen im Menue beieinander stehen.
	setarray .@platz[0], EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW,
	                     EQI_ARMOR, EQI_HAND_R, EQI_HAND_L,
	                     EQI_GARMENT, EQI_SHOES, EQI_ACC_R, EQI_ACC_L,
	                     EQI_SHADOW_ARMOR, EQI_SHADOW_WEAPON,
	                     EQI_SHADOW_SHIELD, EQI_SHADOW_SHOES,
	                     EQI_SHADOW_ACC_R, EQI_SHADOW_ACC_L,
	                     EQI_COSTUME_HEAD_TOP, EQI_COSTUME_HEAD_MID,
	                     EQI_COSTUME_HEAD_LOW, EQI_COSTUME_GARMENT;

	// Die Kennzeichnung ist noetig, nicht huebsch: getequipname
	// liefert den Gegenstandsnamen, und ein Costume-Hut kann
	// genauso heissen wie der echte darunter. Ohne Vorsatz stuenden
	// zwei gleiche Zeilen im Menue.
	setarray .@kat$[0], "", "", "", "", "", "", "", "", "", "",
	                    "^0000FF[S]^000000 ", "^0000FF[S]^000000 ",
	                    "^0000FF[S]^000000 ", "^0000FF[S]^000000 ",
	                    "^0000FF[S]^000000 ", "^0000FF[S]^000000 ",
	                    "^AA00AA[C]^000000 ", "^AA00AA[C]^000000 ",
	                    "^AA00AA[C]^000000 ", "^AA00AA[C]^000000 ";

	.@n = 0;
	.@liste$ = "";
	for (.@i = 0; .@i < 20; .@i++) {
		if (getequipisequiped(.@platz[.@i])) {
			.@wahl[.@n] = .@platz[.@i];
			// Ob es ein Costume ist, wird HIER gemerkt und nicht
			// spaeter aus der EQI-Nummer zurueckgerechnet: die
			// Costume-Nummern sind 10 bis 13, die Shadow-Nummern 15
			// bis 20, und eine solche Spanne im Kopf zu behalten ist
			// genau die Sorte Wissen, die beim naechsten rAthena-
			// Update falsch wird.
			.@kost[.@n] = (.@i >= 16);
			.@liste$ = .@liste$ + .@kat$[.@i]
			         + getequipname(.@platz[.@i]) + ":";
			.@n++;
		}
	}
	if (.@n == 0) {
		mes "[GM Workbench]";
		mes "You are wearing nothing.";
		close;
	}

	.@k = select(.@liste$ + "Never mind") - 1;
	if (.@k >= .@n)
		close;
	.@teil = .@wahl[.@k];
	.@istkostuem = .@kost[.@k];

	mes "[GM Workbench]";
	mes "^0000FF" + getequipname(.@teil) + "^000000";

	// Warnung statt Sperre. Beide Eingriffe FUNKTIONIEREN auf einem
	// Costume - setrandomoption schreibt, und die Optionen werden
	// auch gelesen (status.cpp:4147). Nur beim Aufwerten ist es
	// eine Zahl ohne Wirkung: successrefitem prueft nicht, ob ein
	// Stueck ueberhaupt aufwertbar ist (script.cpp, BUILDIN
	// successrefitem - es setzt refine und fertig), und ein Costume
	// hat keine Werte, die eine Aufwertung erhoehen koennte.
	//
	// Wer hier trotzdem +30 setzen will, soll das koennen. Er soll
	// es nur nicht fuer einen Test halten, der etwas beweist.
	if (.@istkostuem) {
		mes " ";
		mes "^777777Costume slot. Options work and are read";
		mes "like anywhere else. Refine writes a number";
		mes "that nothing reads - there are no stats on";
		mes "a costume for it to raise.^000000";
	}
	next;

	switch (select("Set all five options:Refine to +30:Both:Never mind")) {
	case 1: .@was = 1; break;
	case 2: .@was = 2; break;
	case 3: .@was = 3; break;
	case 4: close;
	}

	if (.@was & 1) {
		.@einzeln = select("All five the same:Five different ones") - 1;
		.@zeile$ = "";

		// Ein Durchlauf bei "alle gleich", fuenf bei "verschieden".
		.@bis = .@einzeln ? 5 : 1;

		for (.@i = 0; .@i < .@bis; .@i++) {

			// KEIN next vor input.
			//
			// next leert das Fenster, und der Eingabekasten steht
			// dann vor leerem Grund - genau deshalb war die Liste
			// beim Tippen nicht mehr zu sehen. Ohne next bleibt der
			// Text stehen, waehrend der Kasten darueber liegt.
			mes "[GM Workbench]";
			if (.@einzeln)
				mes "Slot ^0000FF" + (.@i + 1) + "^000000 of 5 - which option?";
			else
				mes "Which option for all five slots?";
			mes " ";
			for (.@z = 0; .@z < 5; .@z++)
				mes .hilfe$[.@z];
			mes " ";
			if (.@einzeln)
				mes "^7777770 leaves this slot empty.^000000";
			input .@id, (.@einzeln ? 0 : 1), 255;

			// Id 0 laesst einen Platz leer. Dafuer haben wir in
			// db/import/item_randomopt_db.yml eigens einen Eintrag
			// VAR_NOTHING angelegt: setrandomoption lehnt jede
			// unbekannte Id ab, und die amtliche Datenbank beginnt
			// erst bei 1.
			if (.@id == 0) {
				setrandomoption(.@teil, .@i, 0, 0, 0);
				.@zeile$ = .@zeile$ + (.@i + 1) + ": empty  ";
				next;
				continue;
			}

			// Zweite Frage, getrennt von der ersten - und die
			// gewaehlte Id steht dabei vor Augen.
			mes " ";
			mes "Option ^0000FF" + .@id + "^000000 - which value?";
			mes "^777777Stored as int16, so 32767 is the ceiling.^000000";
			input .@wert, 1, 32767;

			if (.@einzeln) {
				setrandomoption(.@teil, .@i, .@id, .@wert, 0);
				.@zeile$ = .@zeile$ + (.@i + 1) + ": " + .@id
				         + "/" + .@wert + "  ";
			} else {
				for (.@z = 0; .@z < 5; .@z++)
					setrandomoption(.@teil, .@z, .@id, .@wert, 0);
				.@zeile$ = "id " + .@id + ", value " + .@wert + " on all five";
			}
			next;
		}

		// setrandomoption schreibt in den Gegenstand, rechnet die
		// Werte aber nicht selbst neu. Ohne das merkt der Charakter
		// die Aenderung erst beim naechsten Ab- und Anlegen.
		recalculatestat;
	}

	if (.@was & 2) {
		// @refine rechnet relativ, dieser Befehl absolut. Von einem
		// bereits aufgewerteten Stueck aus waere ein "+30" sonst zu
		// viel - deshalb die Differenz.
		.@ist = getequiprefinerycnt(.@teil);
		if (.@ist < 30)
			successrefitem .@teil, 30 - .@ist;
	}

	mes "[GM Workbench]";
	mes "Done.";
	mes " ";
	mes "^0000FF" + getequipname(.@teil) + "^000000";
	mes "Refine: ^0000FF+" + getequiprefinerycnt(.@teil) + "^000000";
	if (.@was & 1)
		mes "Options: ^0000FF" + .@zeile$ + "^000000";
	close;

// ============================================================
//  Alles sockeln
// ============================================================
//  Tauscht jedes getragene Stueck gegen dieselbe Ausfuehrung mit
//  mehr Slots und zieht es wieder an.
//
//  WARUM TAUSCHEN UND NICHT AUFBOHREN
//
//  Die Slotzahl haengt an der Item-ID, nicht am einzelnen
//  Exemplar - der Client liest sie aus seiner eigenen Tabelle.
//  Ein Slot laesst sich deshalb nicht aufschalten. Es gibt nur
//  den Tausch gegen die weitere Fassung, sofern die Schmiede sie
//  je gemacht haben. Ausfuehrlich steht das im Kopf von
//  npc/custom/safa_schmied.txt.
//
//  WOHER DIE PAARE KOMMEN
//
//  Aus genau derselben Tabelle wie beim Meisterhandwerker -
//  6.640 Paare, erzeugt von tools/schmied/bauen.py. Sie wird
//  NICHT kopiert, sondern ueber getvariableofnpc gelesen. Eine
//  zweite Kopie waere eine zweite Wahrheit, und beim naechsten
//  Lauf des Generators waere eine davon veraltet.
//
//  WARUM inarray UND KEINE SCHLEIFE
//
//  Bei 6.640 Eintraegen bricht rAthena eine Suchschleife ab:
//  check_gotocount liegt bei 2.048 (script.cpp:4420), im Log
//  steht dann "infinity loop" und der Spieler sieht gar nichts.
//  inarray sucht in C und verbraucht keinen einzigen Sprung.
//
//  Dass inarray hier mit einer FREMDEN Tabelle arbeitet, geht,
//  weil es seinen ersten Parameter als Referenz nimmt
//  (BUILDIN_DEF(inarray,"rv")) und daraus reference_getref liest -
//  und getvariableofnpc setzt genau diesen Verweis auf den
//  Geltungsbereich des anderen NPCs.
//
//  WAS UEBERSPRUNGEN WIRD
//
//    - Stuecke ohne weitere Fassung. Rund 97 Prozent der
//      Ausruestung hat keine.
//    - Stuecke mit Karten oder Optionen. Beide ueberleben den
//      Tausch NICHT, und ein Knopf, der stillschweigend eine
//      Karte verschluckt, ist auf einer GM-Werkbank genauso
//      falsch wie sonst.
//
//  Nur die zehn normalen Plaetze. In Kostuem und Schatten passt
//  ohnehin keine Karte (pc.cpp:5632), ein Slot dort waere ein
//  Loch ohne Zweck.
//
//  MEHRFACH JE STUECK
//
//  inarray liefert den ERSTEN Treffer, und ein Stueck kann
//  mehrere Paare haben - 0 auf 1 und 0 auf 4 stehen beide in der
//  Tabelle. Deshalb wird je Stueck bis zu vier Mal getauscht,
//  solange es noch weiter geht. Die Schranke ist da, damit ein
//  fehlerhaftes Paar, das auf sich selbst zeigt, keine
//  Endlosschleife wird.
// ============================================================
S_AlleSockel:

	setarray .@platz[0], EQI_HEAD_TOP, EQI_HEAD_MID, EQI_HEAD_LOW,
	                     EQI_ARMOR, EQI_HAND_R, EQI_HAND_L,
	                     EQI_GARMENT, EQI_SHOES, EQI_ACC_R, EQI_ACC_L;

	.@getauscht = 0;
	.@ohne = 0;
	.@belegt = 0;
	.@bericht$ = "";

	// Feste Zehn, kein getarraysize - EQI_ACC_L ist 0 und faellt aus
	// dem Array. Siehe oben.
	for (.@i = 0; .@i < 10; .@i++) {

		if (!getequipisequiped(.@platz[.@i]))
			continue;

		.@teil = .@platz[.@i];

		// Karten oder Optionen? Dann Finger weg, und zwar bevor
		// irgendetwas angefasst wird.
		if (getequipcardcnt(.@teil) > 0) {
			.@belegt++;
			continue;
		}
		.@opt = 0;
		for (.@j = 0; .@j < 5; .@j++)
			if (getequiprandomoption(.@teil, .@j, ROA_ID) > 0)
				.@opt++;
		if (.@opt > 0) {
			.@belegt++;
			continue;
		}

		.@vorher$ = getequipname(.@teil);
		.@stufen = 0;

		// Bis zu vier Stufen hoch, siehe MEHRFACH JE STUECK.
		for (.@s = 0; .@s < 4; .@s++) {

			.@id = getequipid(.@teil);
			if (.@id <= 0)
				break;

			.@idx = inarray(getvariableofnpc(.von[0], "Master Artisan#safa"), .@id);
			if (.@idx < 0)
				break;

			.@neu = getvariableofnpc(.nach[.@idx], "Master Artisan#safa");
			if (.@neu <= 0 || .@neu == .@id)
				break;

			.@ref = getequiprefinerycnt(.@teil);

			// Erst vom Koerper nehmen, dann tauschen. Sonst behaelt
			// der Client bis zum naechsten Statuslauf die alten Boni
			// im Fenster.
			unequip .@teil;
			delitem2 .@id, 1, 1, .@ref, 0, 0, 0, 0, 0;
			getitem2 .@neu, 1, 1, .@ref, 0, 0, 0, 0, 0;

			// Wieder anziehen - anders als beim Meisterhandwerker,
			// der die Stuecke bewusst in den Beutel legt. Hier soll
			// EIN Klick genuegen, sonst ist es kein Knopf, sondern
			// eine Aufgabenliste.
			equip .@neu;
			.@stufen++;
		}

		if (.@stufen == 0) {
			.@ohne++;
			continue;
		}

		// Je Stueck eine eigene Zeile, gesammelt in einem Array und
		// nicht in einer Zeichenkette: zehn Namen hintereinander in
		// EIN mes waeren eine Zeile, die das Fenster quer verlaesst.
		.@bericht$[.@getauscht] = "  " + .@vorher$ + " ^777777->^000000 ^00AA00"
		                        + getequipname(.@teil) + "^000000";
		.@getauscht++;
	}

	mes "[GM Workbench]";
	if (.@getauscht == 0 && .@ohne == 0 && .@belegt == 0) {
		mes "You are wearing nothing.";
		next;
		return;
	}

	mes "^00AA00" + .@getauscht + " traded^000000, ^777777" + .@ohne
	  + " had no wider version^000000";

	// Klammern, obwohl es nach einer Zeile aussieht - zwei mes ohne
	// sie gehoerten nicht beide zum if.
	if (.@belegt > 0) {
		mes "^FF6666" + .@belegt + " skipped^000000 - cards or enchants";
		mes "^FF6666would have been lost in the trade.^000000";
	}

	if (.@getauscht > 0) {
		mes " ";
		for (.@i = 0; .@i < .@getauscht; .@i++)
			mes .@bericht$[.@i];
	}
	next;

	// recalculatestat, weil equip zwar neu anlegt, die Boni der
	// frischen Slots aber erst beim naechsten Statuslauf im Fenster
	// stehen. Dasselbe Nachziehen wie bei den Optionen weiter oben.
	recalculatestat;
	return;


OnInit:
	.stufe = 99;   // ab dieser Gruppenstufe sicht- und bedienbar
	bindatcmd "emo", "GM Workbench#safa::OnEmote", 99, 99;

	// Die Optionsliste steht einmal hier statt dreimal im Ablauf.
	// Es sind dieselben zwoelf, die der Master Artisan vergibt -
	// siehe tools/schmied/bauen.py, OPTIONEN.
	setarray .hilfe$[0],
		"  ^0000FF17^000000 ATK          ^0000FF14^000000 MATK in percent",
		"  ^0000FF 1^000000 Max HP       ^0000FF 2^000000 Max SP",
		"  ^0000FF 3^000000 STR   ^0000FF4^000000 AGI   ^0000FF5^000000 VIT",
		"  ^0000FF 6^000000 INT   ^0000FF7^000000 DEX   ^0000FF8^000000 LUK",
		"  ^0000FF15^000000 ASPD         ^0000FF24^000000 Critical";
	end;

// ------------------------------------------------------------
//  Unsichtbar fuer alle ausser GM
// ------------------------------------------------------------
//  cloakonnpcself verbirgt den NPC fuer GENAU DIESEN Spieler.
//  Der Zustand steht in sd->cloaked_npc (npc.cpp:1007), also je
//  Sitzung und je Charakter - nicht am NPC selbst. Deshalb muss
//  es bei jeder Anmeldung neu gesetzt werden.
//
//  disablenpc waere falsch: das schaltet ihn fuer ALLE ab, auch
//  fuer uns.
//
//  Voreinstellung ist sichtbar - die Liste beginnt leer. Wer
//  kein GM ist, bekommt ihn hier weggeblendet; GMs lassen wir in
//  Ruhe und sehen ihn dadurch.
//
//  Mehrere NPCs duerfen OnPCLoginEvent haben, rAthena ruft sie
//  alle auf. safa_clan.txt benutzt es ebenfalls.
OnPCLoginEvent:
	if (getgroupid() < .stufe)
		cloakonnpcself "GM Workbench#safa";
	end;


// ------------------------------------------------------------
//  @emo - Gegentest fuer die Emote-Frage
// ------------------------------------------------------------
//  WOZU
//
//  Im Spiel zeigt /gawi keine Emote. Dafuer gibt es genau zwei
//  moegliche Ursachen, und sie liegen an entgegengesetzten
//  Enden:
//
//    1. Der Client SENDET nichts, was der Server versteht.
//       Seit PACKETVER 20230705 registriert rAthena das Paket
//       CZ_REQ_EMOTION_EXPANSION (0x0BE9) - und haengt es an
//       clif_parse_dull, eine Attrappe, die nichts tut
//       (clif_packetdb.hpp:2030). Die Struktur dazu ist nicht
//       einmal ausgewertet: "uint8 unknown[4]"
//       (packets.hpp:2033). Unser Client ist 20250716.
//
//    2. Der Client kann Emotes gar nicht ZEICHNEN, etwa weil im
//       Patchstand etwas an den Effekt-Sprites fehlt.
//
//  Dieser Befehl trennt die beiden. Er geht NICHT ueber ein
//  Client-Paket, sondern laesst den Server die Emote von sich
//  aus senden.
//
//    Emote erscheint  -> zeichnen kann er, es liegt am Paket
//                        (Ursache 1, braucht C++ und Neubau)
//    Emote erscheint nicht -> Client-/Patchproblem (Ursache 2)
//
//  Nummern stehen als ET_ in src/map/script_constants.hpp.
//  /gawi ist ET_SCISSOR = 10, /bawi ist ET_ROCK = 11.
//
//    @emo        macht ET_SCISSOR, also dasselbe wie /gawi
//    @emo 23     eine bestimmte Nummer
OnEmote:
	if (getgroupid() < .stufe) {
		dispbottom "[Emote] Nicht fuer dich.";
		end;
	}

	.@n = 10;
	if (.@atcmd_numparameters > 0)
		.@n = atoi(.@atcmd_parameters$[0]);

	// ET_MAX liegt bei rund 90; alles darueber wuerde der Client
	// verwerfen, ohne etwas zu melden.
	if (.@n < 0 || .@n > 89) {
		dispbottom "[Emote] Nummer 0 bis 89.";
		end;
	}

	emotion .@n, getcharid(3);
	dispbottom "[Emote] Nummer " + .@n + " vom Server gesendet. "
	           + "Siehst du sie ueber deinem Kopf?";
	end;

}
