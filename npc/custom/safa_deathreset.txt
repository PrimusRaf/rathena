// ============================================================
//  SafaRO - Die Schwellenwaechterin
// ------------------------------------------------------------
//  Setzt den Todes-Debuff zurueck. Der Debuff selbst steht in
//  safa_systems.txt: death_count zaehlt die Tode, die Strafe
//  ist death_count * 5 Prozent auf alle Kampfwerte, gedeckelt
//  bei 90.
//
//  Drei Stufen, in dieser Reihenfolge:
//    1  Material    50 bis 200 Stueck, beim Annehmen ausgewuerfelt
//    2  Jagd        300 Alarm, Quest 70251
//    3  Bezahlung   400 Destiny Coins
//
//  Beliebig wiederholbar - man stirbt ja weiter.
//
//  STANDORT: VEINS, NICHT MEHR PRONTERA
//
//  Mit Episode 22 wird Veins die Hauptstadt. Die Waechterin
//  zieht deshalb auf veins 206,54 - den Platz, auf dem bis
//  hierher ihre kuerzere Schwester stand. Die geht ihrerseits
//  weiter nach wolfvill, siehe unten.
//
//  Die Prontera-Achse verliert damit ihren Eintrag bei
//  146,182. Die Zelle bleibt frei; wer dort spaeter etwas
//  hinstellt, hat freie Bahn. Der Kopfkommentar in
//  safa_dienste.txt ist entsprechend nachgezogen.
//
//  veins 206,54 ist Typ 0, also trockener begehbarer Boden -
//  geprueft mit tools/karte/zelle_pruefen.py gegen
//  db/map_cache.dat. Der Job-Quest-Ausloeser bei 206,56 hat
//  Reichweite 1,1 und deckt die Zeilen 55 bis 57 ab; auf 54
//  steht ihr nichts im Weg.
// ============================================================

veins,206,54,4	script	Threshold Warden#safa	4_F_SISTER,{

	// Kein Debuff, nichts zu tun - aber die Quest laeuft womoeglich
	// noch aus einem frueheren Anlauf. Dann trotzdem weitermachen
	// lassen, sonst haengt der Fortschritt fest.
	if (death_count <= 0 && dq_stufe == 0) {
		mes "[Threshold Warden]";
		mes "You carry nothing yet. No deaths, no debt.";
		mes "Come back when the cycle has taken";
		mes "something from you.";
		close;
	}

	// "marks" und nicht mehr "deaths in all": seit der Rangliste
	// gibt es zwei Zahlen, und sie sind nicht dieselbe.
	// death_count sind die Male, die noch auf einem liegen -
	// bezahlbar. tode_gesamt ist die Lebensbilanz und bleibt, auch
	// nachdem hier bezahlt wurde. Wer das nicht erfaehrt, haelt
	// seinen Rang auf der Rangliste hinterher fuer einen Fehler.
	mes "[Threshold Warden]";
	mes "Every death leaves a mark. Yours weighs";
	mes "^FF0000" + (death_count * 5 > 90 ? 90 : death_count * 5) + "%^000000 on everything you do -";
	mes "^FF0000" + death_count + "^000000 " + (death_count == 1 ? "mark" : "marks") + " you still carry.";
	if (tode_gesamt > death_count)
		mes "^777777The cycle counts " + tode_gesamt + " in all. That number it keeps.^000000";
	next;

	// Jeder Zweig endet mit close, kehrt also nie zurueck. Das break
	// steht trotzdem da: rAthena faellt ohne es in den naechsten Fall
	// durch, und wer hier spaeter ein return einbaut, wuerde das
	// sonst erst im Spiel merken.
	switch (dq_stufe) {
		case 0: callsub S_Annehmen; break;
		case 1: callsub S_Material; break;
		case 2: callsub S_Jagd;     break;
		case 3: callsub S_Bezahlen; break;
	}
	end;


// ------------------------------------------------------------
//  Stufe 0 - annehmen
// ------------------------------------------------------------
S_Annehmen:
	mes "[Threshold Warden]";
	mes "The cycle can be broken. Not begged out";
	mes "of - broken. Three things, in order.";
	next;
	mes "[Threshold Warden]";
	mes "First a material, and how much of it is";
	mes "not for me to decide. Then three hundred";
	mes "Alarms. Then a price in coin.";
	next;

	if (select("Begin:Not now") == 2) {
		mes "[Threshold Warden]";
		mes "The marks will wait. They always do.";
		close;
	}

	mes "[Threshold Warden]";
	mes "Which will you bring?";
	next;

	.@w = select("^0000FF" + getitemname(.mat[0]) + "^000000:^0000FF" + getitemname(.mat[1]) + "^000000:Let me think");
	if (.@w > 2) {
		mes "[Threshold Warden]";
		mes "Think, then. The cycle is patient.";
		close;
	}

	set dq_item, .mat[.@w - 1];
	// Einmal auswuerfeln und merken. rand(a,b) schliesst beide
	// Grenzen ein - 50 und 200 sind also moeglich.
	set dq_menge, rand(.min_menge, .max_menge);
	set dq_stufe, 1;

	mes "[Threshold Warden]";
	mes "^0000FF" + dq_menge + " " + getitemname(dq_item) + "^000000.";
	mes "That is what the cycle asks of you.";
	next;
	mes "[Threshold Warden]";
	mes "Not a number I chose. Come back when you";
	mes "have them.";
	close;


// ------------------------------------------------------------
//  Stufe 1 - Material
// ------------------------------------------------------------
S_Material:
	.@habe = countitem(dq_item);

	mes "[Threshold Warden]";
	mes "^0000FF" + getitemname(dq_item) + "^000000: " +
	    (.@habe >= dq_menge ? "^00AA00" : "^FF0000") + .@habe + "^000000 of " + dq_menge + ".";

	if (.@habe < dq_menge) {
		next;
		mes "[Threshold Warden]";
		mes "Not yet.";
		close;
	}

	next;
	if (select("Hand them over:Keep them for now") == 2) {
		mes "[Threshold Warden]";
		mes "As you like.";
		close;
	}

	// Erst pruefen, dann nehmen. Zwischen der Abfrage oben und hier
	// kann nichts passieren - der Spieler steht im Dialog - aber die
	// Reihenfolge kostet nichts und schuetzt vor spaeteren Umbauten.
	if (countitem(dq_item) < dq_menge) {
		mes "[Threshold Warden]";
		mes "They are gone from your hands.";
		close;
	}
	delitem dq_item, dq_menge;
	set dq_stufe, 2;

	// Vor dem Eintragen aufraeumen: quest_add weist eine bereits
	// vorhandene Quest ab (quest.cpp Zeile 599). Beim zweiten
	// Durchlauf bliebe sonst der alte, erledigte Stand stehen und
	// Stufe 2 waere sofort geschafft.
	if (isbegin_quest(.quest) > 0)
		erasequest .quest;
	setquest .quest;

	mes "[Threshold Warden]";
	mes "Taken. Now the second thing.";
	next;
	mes "[Threshold Warden]";
	mes "^0000FF300 Alarms^000000. They keep time for the";
	mes "cycle. Stop enough of them and it loses";
	mes "count of you.";
	next;
	mes "[Threshold Warden]";
	mes "Your quest log will keep score.";
	close;


// ------------------------------------------------------------
//  Stufe 2 - Jagd
// ------------------------------------------------------------
S_Jagd:
	// checkquest mit HUNTING: 2 heisst, alle Ziele sind erledigt.
	// -1 waere "nicht im Log" - das kann vorkommen, wenn jemand die
	// Quest ueber einen Reset verloren hat. Dann neu eintragen,
	// statt den Spieler haengen zu lassen.
	if (checkquest(.quest, HUNTING) == -1) {
		setquest .quest;
		mes "[Threshold Warden]";
		mes "Your log had lost the count. It is set";
		mes "again - from the beginning, I am afraid.";
		close;
	}

	if (checkquest(.quest, HUNTING) != 2) {
		mes "[Threshold Warden]";
		mes "The Alarms still keep time. Check your";
		mes "quest log - it counts better than I do.";
		close;
	}

	mes "[Threshold Warden]";
	mes "Three hundred. The ticking has stopped.";
	next;
	// Ganz entfernen statt auf "erledigt" setzen: die Quest gehoert
	// nur zu Stufe 2, und der naechste Durchlauf legt sie neu an.
	erasequest .quest;
	set dq_stufe, 3;

	mes "[Threshold Warden]";
	mes "One thing remains, and it is the only one";
	mes "the cycle understands: ^0000FF" + .preis + " Destiny";
	mes "Coins^000000.";
	next;
	mes "[Threshold Warden]";
	mes "Bend fate in your favour. It bends for";
	mes "coin like everything else.";
	close;


// ------------------------------------------------------------
//  Stufe 3 - bezahlen und zuruecksetzen
// ------------------------------------------------------------
S_Bezahlen:
	.@habe = countitem(.coin);

	mes "[Threshold Warden]";
	mes "^0000FF" + .preis + " Destiny Coins^000000. You carry " +
	    (.@habe >= .preis ? "^00AA00" : "^FF0000") + .@habe + "^000000.";

	if (.@habe < .preis) {
		next;
		mes "[Threshold Warden]";
		mes "Come back richer.";
		close;
	}

	next;
	if (select("Pay:Later") == 2) {
		mes "[Threshold Warden]";
		mes "Later, then.";
		close;
	}

	if (countitem(.coin) < .preis) {
		mes "[Threshold Warden]";
		mes "The coins are gone from your hands.";
		close;
	}
	delitem .coin, .preis;

	.@vorher = death_count;
	set death_count, 0;

	// F_SafaRefresh setzt Geschenk und Debuff neu. Ohne diesen
	// Aufruf bliebe die Strafe bis zum naechsten Login oder bis zur
	// stuendlichen Auffrischung bestehen - der Spieler haette bezahlt
	// und wuerde nichts merken.
	callfunc "F_SafaRefresh";

	// Alles zuruecksetzen, damit der naechste Durchlauf sauber
	// beginnt und neu ausgewuerfelt wird.
	set dq_stufe, 0;
	set dq_item, 0;
	set dq_menge, 0;

	specialeffect2 EF_ANGEL2;
	mes "[Threshold Warden]";
	mes "^00AA00" + .@vorher + " " + (.@vorher == 1 ? "mark" : "marks") + " gone.^000000";
	next;
	mes "[Threshold Warden]";
	mes "You are whole again. The cycle has not";
	mes "forgiven you - it has merely lost your";
	mes "name for a while.";
	next;
	mes "[Threshold Warden]";
	mes "It will find you. Come back when it does.";
	close;


OnInit:
	.coin = 30001;                    // Destiny Coin
	.preis = 400;
	setarray .mat[0], 1001084, 1001085;   // Blaze Stone, Cool Liquid
	.min_menge = 50;
	.max_menge = 200;
	.quest = 70251;                   // 300 Alarm, siehe db/import/quest_db.yml
	end;
}


// ============================================================
//  Dieselbe Aufgabe ohne Umweg - jetzt hinter einer Losung
// ------------------------------------------------------------
//  Die Waechterin in Veins verlangt drei Stufen: Material,
//  Jagd, Bezahlung. Das ist als Beschaeftigung gedacht - aber wer
//  nur weiterspielen will, soll nicht erst eine Questreihe
//  abarbeiten muessen, um seine eigenen Werte zurueckzubekommen.
//
//  WAS SICH MIT EPISODE 22 AENDERT
//
//  Bisher nahm diese Fassung 1.000 Zeny und stand mitten in der
//  Stadt. Beides war zu bequem: 1.000 Zeny sind auf diesem Server
//  kein Preis, sondern eine Formalitaet, und wer sie zahlen wollte,
//  brauchte dafuer drei Schritte zu gehen.
//
//  Jetzt kostet sie gar nichts mehr - und ist trotzdem teurer.
//  Sie steht in wolfvill, dem Doerfchen der Grauen Woelfe, und
//  bedient nur, wer die Losung kennt. Die Losung erfaehrt man
//  ausschliesslich innerhalb der Episode.
//
//  Damit ist der kurze Weg kein Geldproblem mehr, sondern ein
//  Wissensproblem. Zeny hat auf einem 25x-Server jeder; die
//  Losung hat nur, wer gespielt hat. Und weil Wissen sich unter
//  Spielern verbreitet, ist das kein Riegel fuer immer, sondern
//  ein Vorsprung fuer die Ersten - genau das, was eine Episode
//  am Anfang braucht.
//
//  WARUM SIE DIE LOSUNG NUR EINMAL HOEREN WILL
//
//  Wer sie richtig gesagt hat, bekommt ep22_losung gesetzt und
//  wird danach ohne Abfrage bedient. Man stirbt auf diesem Server
//  oft; die Losung bei jedem Besuch neu zu tippen waere Reibung
//  ohne Spielwert. Soll sie doch jedes Mal gefragt werden, faellt
//  genau eine Zeile weg - die Pruefung auf ep22_losung ganz unten.
//
//  WARUM GROSS- UND KLEINSCHREIBUNG EGAL IST
//
//  Die Losung wird ueber strtolower verglichen. Ein Passwort, das
//  an einer Umschalttaste scheitert, bestraft nicht Unwissen,
//  sondern Tippen - und der Spieler sieht nicht, woran es lag.
//
//  Warum kein duplicate(): das Original ist eine Zustandsmaschine
//  ueber dq_stufe. Ein Duplikat waere derselbe Ablauf an einem
//  zweiten Ort, nicht der kuerzere. Gemeinsam ist beiden nur der
//  Kern - death_count auf 0 und F_SafaRefresh - und der sind zwei
//  Zeilen.
//
//  dq_stufe wird hier bewusst NICHT angefasst. Wer in Veins
//  mitten in der Questreihe steckt, behaelt seinen Fortschritt;
//  die dortige Eingangspruefung laesst ihn mit dq_stufe > 0
//  ohnehin weitermachen. Ein Zuruecksetzen von hier aus wuerde ihm
//  nur die halbe Kette wegnehmen.
//
//  Standort: wolfvill 190,124. Die Koordinate kommt von Raffael
//  und ist mit tools/karte/zelle_pruefen.py gegen
//  db/map_cache.dat geprueft: begehbar.
//
//  Zum Zelltyp, damit sich niemand daran stoert: die Zelle ist
//  Typ 3, also "Wasser, begehbar". Das ist auf dieser Karte der
//  Normalfall und kein Fehlgriff - wolfvill hat 23.474 Zellen vom
//  Typ 3 und die naechste trockene Typ-0-Zelle liegt 50 Felder
//  entfernt. Das gesamte begehbare Dorf ist so ausgezeichnet.
// ============================================================

wolfvill,190,124,4	script	Threshold Warden#safa_wolf	4_F_SISTER,{

	if (death_count <= 0) {
		mes "[Threshold Warden]";
		mes "Nothing clings to you. No deaths, no debt.";
		mes "Keep it that way as long as you can.";
		close;
	}

	.@pen = death_count * 5;
	if (.@pen > 90)
		.@pen = 90;

	mes "[Threshold Warden]";
	mes "^FF0000" + death_count + "^000000 "
	    + (death_count == 1 ? "mark" : "marks") + " on you.";
	mes "That is ^FF0000" + .@pen + "%^000000 off everything you do -";
	mes "strike, magic, life, spirit, armour, dodge and aim.";
	next;

	mes "[Threshold Warden]";
	mes "My sister in Veins will ask you for errands.";
	mes "She has the patience for it. I do not.";
	mes " ";
	mes "I take no coin. I have no use for it out";
	mes "here.";
	next;

	// Wer die Losung schon einmal gesagt hat, sagt sie nicht
	// wieder. Siehe Kopfkommentar: diese eine Bedingung ist der
	// Unterschied zwischen "einmal lernen" und "jedes Mal tippen".
	if (!ep22_losung) {

		mes "[Threshold Warden]";
		mes "What I take is a ^FFCC00word^000000. One word, and";
		mes "you only hear it out there - from the ones";
		mes "who go into the tall grass and come back.";
		next;

		if (select("Give the word:Not now") == 2) {
			mes "[Threshold Warden]";
			mes "Then carry them a while longer.";
			close;
		}

		mes "[Threshold Warden]";
		mes "Quietly.";
		next;

		// Der Rueckgabewert von input ist hier ohne Belang: bei
		// Zeichenketten meldet er nur Laengenverstoesse, und eine
		// Laenge geben wir nicht vor. Die leere Eingabe faellt
		// ohnehin durch den Vergleich unten.
		input .@wort$;

		// strtolower auf beiden Seiten: siehe Kopfkommentar. Der
		// Vergleich toleriert die Umschalttaste, sonst nichts -
		// Leerzeichen und Tippfehler gehen weiterhin durch.
		if (strtolower(.@wort$) != strtolower(.losung$)) {
			mes "[Threshold Warden]";
			mes "^FF0000That is not it.^000000";
			next;
			mes "[Threshold Warden]";
			mes "I am not going to tell you whether you";
			mes "were close. Go back out and listen.";
			close;
		}

		// Permanente Charaktervariable - ueberlebt Logout und
		// Serverneustart. Absichtlich nicht account-weit: die
		// Losung gehoert dem, der sie sich verdient hat.
		set ep22_losung, 1;

		mes "[Threshold Warden]";
		mes "^00AA00That is it.^000000";
		next;
		mes "[Threshold Warden]";
		mes "You will not have to say it to me again.";
		mes "Say it to no one who has not earned it.";
		next;

	} else {

		// Die Rueckfrage fuer die Wissenden. Ohne sie waere das
		// blosse Ansprechen der Waechterin bereits die Handlung -
		// wer nur nachsehen will, wieviele Male er traegt, haette
		// sie beim Zumachen des Fensters schon verloren. Der
		// Unwissende bekommt seine Rueckfrage weiter oben, im
		// "Give the word", und braucht diese hier nicht doppelt.
		mes "[Threshold Warden]";
		mes "You have said the word before. It still";
		mes "holds.";
		next;

		if (select("Take them off:Not now") == 2) {
			mes "[Threshold Warden]";
			mes "Then carry them a while longer.";
			close;
		}
	}

	.@vorher = death_count;
	set death_count, 0;

	// Ohne diesen Aufruf bliebe die Strafe bis zum naechsten Login
	// stehen: der Spieler haette die Losung gesagt und wuerde nichts
	// merken. Dieselbe Falle wie beim Original, siehe oben.
	callfunc "F_SafaRefresh";

	specialeffect2 EF_ANGEL2;
	mes "[Threshold Warden]";
	mes "^00AA00" + .@vorher + " " + (.@vorher == 1 ? "mark" : "marks")
	    + " gone.^000000";
	next;
	mes "[Threshold Warden]";
	mes "No charge. The word was the charge, and";
	mes "somebody out there paid for it once.";
	mes "Go on, then - it will find you again.";
	close;

OnInit:
	// Die Losung steht genau hier und nirgends sonst. Wird sie je
	// geaendert, aendert sich das Spiel an einer Stelle - und alle,
	// die ep22_losung schon gesetzt haben, merken davon nichts.
	// Soll eine Aenderung auch die Wissenden wieder abfragen, muss
	// die Variable mitgezaehlt werden (ep22_losung == 2 und so fort).
	.losung$ = "Hunter";
	end;
}
