//===== SafaRO Script =====================================
//= The Threshold - Custom Start
//===== Beschreibung: ========================================
//= Neue Charaktere erwachen in der Halle von Valhalla. Der
//= Threshold Warden erklaert den Server, laesst den Spieler
//= sein Isekai-Geschenk waehlen - und erst diese Wahl oeffnet
//= den Weg nach Veins - der Hauptstadt seit Episode 22.
//=
//= Die Geschenke selbst (Namen, Wirkung, Beschreibung) liegen
//= in npc/custom/safa_systems.txt. Dieser NPC ist nur die
//= Buehne dafuer.
//===== Hinweise: ============================================
//= Karte: valkyrie (100x100). Startkorridor x 46-51, y 7-30.
//= Bestehende Charaktere kommen mit @warp valkyrie 48,9 her.
//= Spieltexte durchgehend englisch, ohne Umlaute.
//============================================================

valkyrie,48,14,4	script	Threshold Warden	4_F_VALKYRIE2,{

	// --- Wer sein Geschenk schon hat, wird nur noch entlassen ---
	//     Ausnahme: GMs duerfen zuruecksetzen, sonst waere Testen
	//     nach jeder Aenderung ein neuer Charakter.
	if (isekai_gift > 0) {
		mes "[Threshold Warden]";
		mes "You already carry your gift:";
		mes "^0000FF" + callfunc("F_SafaGiftName", isekai_gift) + "^000000";
		next;
		if (getgroupid() >= 99) {
			mes "[Threshold Warden]";
			mes "^777777You are not bound by the rules";
			mes "of this place.^000000";
			next;
			if (select("Leave it:^FF0000Unbind it (GM)^000000") == 2) {
				set isekai_gift, 0;
				callfunc "F_SafaRefresh";
				mes "[Threshold Warden]";
				mes "^777777The binding is undone.^000000";
				mes "Talk to me again to choose anew.";
				close;
			}
		}
		// --- Kanten nachholen ---
		//     Dieselbe Ueberlegung wie beim Startpaket eine Zeile
		//     weiter unten: wer sein Geschenk gewaehlt hat, bevor es
		//     die freie Fertigkeitswahl gab, steht sonst dauerhaft
		//     ohne Schadensbonus da - die festen Vergaben sind aus
		//     F_SafaGift heraus.
		//
		//     S_Kanten kehrt von selbst zurueck, wenn nichts zu tun
		//     ist: kein Geschenk ab Tier B, oder schon gewaehlt.
		callsub S_Kanten;

		// --- Startpaket nachholen ---
		//     Fuer alle, die ihr Geschenk gewaehlt haben, bevor es das
		//     Paket gab - und fuer jeden, bei dem die Platzpruefung
		//     beim ersten Versuch fehlgeschlagen ist. safa_startpaket
		//     ist eine Charaktervariable, es bleibt also bei einmal.
		if (safa_startpaket == 0) {
			mes "[Threshold Warden]";
			mes "You left without your provisions.";
			mes "Careless. Here.";
			next;
			callfunc "F_SafaStartpaketText";
			next;
			if (callfunc("F_SafaStartpaket") == 0) {
				mes "[Threshold Warden]";
				mes "Come back when you can carry it.";
				close;
			}
			// Kein next hier: das Fenster ist durch das next oben
			// schon leer, und F_SafaStartpaket schreibt kein mes.
			// Ein zweites next zeigte eine leere Seite zum Wegklicken.
		}

		mes "[Threshold Warden]";
		mes "There is nothing left for you here.";
		close2;
		savepoint "veins",210,123;
		warp "veins",210,123;
		end;
	}

	// --- Ankunft ---
	mes "[Threshold Warden]";
	if (safa_start_seen == 0) {
		mes "You are not quite here yet, soul.";
		mes "And not quite there either.";
		next;
		mes "[Threshold Warden]";
		mes "This is the Threshold. A place";
		mes "between what you were and what";
		mes "you are about to become.";
		next;
		set safa_start_seen, 1;
	} else {
		mes "Still here. Ask, then choose.";
		next;
	}

	// --- Hauptmenue ---
	while (1) {
		mes "[Threshold Warden]";
		mes "You may ask. But you will not leave";
		mes "this hall until you have chosen";
		mes "what you carry into the world.";
		next;

		.@main$ = "Where am I?:What is different in that world?:What commands do I have?:^0000FFChoose my gift^000000";
		if (getgroupid() >= 99)
			.@main$ = .@main$ + ":^FF0000Set test points (GM)^000000";

		switch (select(.@main$)) {

		case 1:
			mes "[Threshold Warden]";
			mes "This hall is Valhalla.";
			mes "Here gather those whose story";
			mes "has not been written yet.";
			next;
			mes "[Threshold Warden]";
			mes "Below waits ^0000FFVeins^000000. Everything";
			mes "begins there - and everything you";
			mes "do from now on is counted.";
			next;
			break;

		case 2:
			mes "[Threshold Warden]";
			mes "In the cities you will find helpers";
			mes "that do not exist elsewhere:";
			next;
			mes "[Threshold Warden]";
			mes "^0000FFWarper^000000 - takes you to any city";
			mes "and any dungeon.";
			mes "^0000FFHealer^000000 - heals you for free and";
			mes "blesses you for twenty minutes.";
			next;
			mes "[Threshold Warden]";
			mes "^0000FFJob Master^000000 - class changes without";
			mes "the old trials. He stands beside";
			mes "every Healer.";
			mes "^0000FFStylist^000000 and ^0000FFCard Remover^000000 wait";
			mes "in Veins, with all the rest.";
			next;
			mes "[Threshold Warden]";
			mes "One more thing, and listen well.";
			next;
			mes "[Threshold Warden]";
			mes "^FF0000Death is not free here.^000000";
			mes "Every time you fall, you lose 5% of";
			mes "everything you are. It does not wear";
			mes "off on its own.";
			next;
			break;

		case 3:
			mes "[Threshold Warden]";
			mes "Speak these words whenever you like:";
			next;
			mes "[Threshold Warden]";
			mes "^0000FF@go^000000 - a list of all cities.";
			mes "^0000FF@commands^000000 - everything you may do.";
			mes "^0000FF@help <word>^000000 - what a word does.";
			next;
			mes "[Threshold Warden]";
			mes "^0000FF@mobinfo^000000 and ^0000FF@iteminfo^000000 will tell";
			mes "you what a creature drops and what";
			mes "an item is worth.";
			next;
			break;

		case 4:
			// --- Tier-Auswahl ---
			.@pts = callfunc("F_SafaPoints");

			mes "[Threshold Warden]";
			mes "The gifts are sorted by weight.";
			mes "Tier ^0000FFF^000000 is open to anyone.";
			mes "The rest you must have earned.";
			next;
			mes "[Threshold Warden]";
			mes "Every soul of your line standing at";
			mes "level ^0000FF500^000000 is worth ^0000FF100 points^000000 to you.";
			next;
			mes "[Threshold Warden]";
			mes "^FF0000Standing^000000, mind you. Erase one and its";
			mes "weight leaves with it. The dead do not";
			mes "vouch for you here.";
			next;
			mes "[Threshold Warden]";
			mes "You carry ^0000FF" + .@pts + " points^000000, which opens";
			mes "everything up to tier ^0000FF" + callfunc("F_SafaTierName", callfunc("F_SafaHighestTier")) + "^000000.";
			next;

			while (1) {
				.@menu$ = "";
				freeloop(1);
				for (.@i = 0; .@i < 9; .@i++) {
					.@req = callfunc("F_SafaTierReq", .@i);
					if (.@pts >= .@req)
						.@menu$ = .@menu$ + "Tier " + callfunc("F_SafaTierName", .@i) + ":";
					else
						.@menu$ = .@menu$ + "^777777Tier " + callfunc("F_SafaTierName", .@i) + " - " + .@req + " points^000000:";
				}
				freeloop(0);
				.@menu$ = .@menu$ + "^777777Let me ask something else^000000";

				.@t = select(.@menu$) - 1;
				if (.@t == 9)
					break;

				if (.@pts < callfunc("F_SafaTierReq", .@t)) {
					mes "[Threshold Warden]";
					mes "Not yet. That tier asks for";
					mes "^FF0000" + callfunc("F_SafaTierReq", .@t) + " points^000000, and you carry ^0000FF" + .@pts + "^000000.";
					next;
					continue;
				}

				// Geschenke dieses Tiers aufbauen
				//
				// freeloop, und zwar nicht aus Bequemlichkeit.
				// rAthena uebersetzt jedes "case" eines switch in ein
				// goto (script.cpp, parse_syntax), und goto zaehlt
				// gegen check_gotocount - Abbruch bei 2048. Bei 120
				// Geschenken kostet ein Aufruf von F_SafaGiftName
				// schon bis zu 120 Spruenge; mal achtzehn Eintraege
				// in den grossen Tiers sind das ueber 2100 - die
				// Grenze reisst also nicht mehr knapp, sondern sicher.
				//
				// Genau daran ist der Waechter gestorben: Text im
				// Fenster, kein Next-Knopf, und im Protokoll
				// "run_script_main: infinity loop". Die Schleife ist
				// nachweislich endlich - sie laeuft .@n Mal - also
				// ist das Abschalten der Schleifenwache hier richtig
				// und nicht bloss bequem.
				freeloop(1);
				.@n = callfunc("F_SafaTierCount", .@t);
				.@menu2$ = "";
				for (.@i = 1; .@i <= .@n; .@i++)
					.@menu2$ = .@menu2$ + callfunc("F_SafaGiftName", callfunc("F_SafaTierGift", .@t, .@i)) + ":";
				freeloop(0);
				.@menu2$ = .@menu2$ + "^777777Another tier^000000";

				.@c = select(.@menu2$);
				if (.@c > .@n)
					continue;

				.@pick = callfunc("F_SafaTierGift", .@t, .@c);

				mes "[" + callfunc("F_SafaGiftName", .@pick) + "]";
				mes "^777777Tier " + callfunc("F_SafaTierName", .@t) + "^000000";
				mes " ";
				callfunc "F_SafaGiftDesc", .@pick;
				next;

				mes "[Threshold Warden]";
				mes "Bind ^0000FF" + callfunc("F_SafaGiftName", .@pick) + "^000000 to yourself?";
				mes "^FF0000This cannot be undone.^000000";
				next;

				if (select("Yes, bind it:Show me something else") == 1) {
					set isekai_gift, .@pick;
					callfunc "F_SafaRefresh";
					specialeffect2 EF_ANGEL2;

					mes "[Threshold Warden]";
					mes "It is done.";
					mes "^0000FF" + callfunc("F_SafaGiftName", .@pick) + "^000000 is yours.";
					next;

					// --- Die zwei Kanten ---
					//     MUSS hier stehen und nicht bei einem eigenen
					//     NPC in der Halle: dieser Ablauf endet unten
					//     mit savepoint und warp nach Veins. Wer hier
					//     durch ist, kommt nicht zurueck, und ein
					//     zweiter NPC in Valhalla staende hinter einer
					//     Tuer, die schon zugefallen ist.
					callsub S_Kanten;

					// --- Startpaket ---
					//     Erst hier, nicht beim Betreten der Halle: wer
					//     sein Geschenk noch nicht gewaehlt hat, soll den
					//     Beutel nicht schon voll haben, wenn die
					//     Platzpruefung laeuft.
					mes "[Threshold Warden]";
					mes "One more thing. Nobody walks out";
					mes "of here empty-handed.";
					next;
					callfunc "F_SafaStartpaketText";
					next;
					callfunc "F_SafaStartpaket";

					mes "[Threshold Warden]";
					mes "The hall has no more use for you.";
					mes "Go and be someone.";
					close2;
					savepoint "veins",210,123;
					warp "veins",210,123;
					end;
				}
			}
			break;

		case 5:
			// --- Nur GM: Punktestand zum Testen ueberschreiben ---
			//     Die echten Punkte kommen aus der Datenbank und lassen
			//     sich nicht setzen. Dieser Wert legt sich davor.
			mes "[Threshold Warden]";
			mes "^777777How many points should this line";
			mes "appear to carry?^000000";
			mes " ";
			mes "^7777770 puts the real count back in charge.";
			mes "Currently overridden with: " + #isekai_debug_points + "^000000";
			next;
			input .@v, 0, 1500;
			set #isekai_debug_points, .@v;
			mes "[Threshold Warden]";
			if (.@v == 0) {
				mes "^777777Override removed. The database";
				mes "decides again: " + callfunc("F_SafaPoints") + " points.^000000";
			} else {
				mes "^777777Override set to " + .@v + " points.";
				mes "Highest tier: " + callfunc("F_SafaTierName", callfunc("F_SafaHighestTier")) + "^000000";
			}
			next;
			break;
		}
	}

	// Die Schleife darueber wird nur ueber close oder warp
	// verlassen, herausfallen kann man nicht. Das end steht
	// trotzdem da: darunter beginnen die Sprungziele, und ohne
	// diese Zeile liefe ein Skript, das doch einmal herausfaellt,
	// stillschweigend in S_Kanten hinein.
	end;


// ============================================================
//  Die zwei Kanten
// ============================================================
//  Ab Tier B schaerft ein Geschenk zwei Fertigkeiten. Welche
//  zwei, waehlt der Spieler hier - einmal, dann endgueltig, wie
//  das Geschenk selbst.
//
//  WARUM HIER UND NICHT BEI EINEM EIGENEN NPC
//
//  Beide Ausgaenge dieses Skripts enden mit
//
//      savepoint "veins",210,123;
//      warp "veins",210,123;
//
//  Wer die Waechterin hinter sich hat, ist aus der Halle
//  draussen und kommt nicht zurueck. Ein zweiter NPC in
//  Valhalla - erst so gebaut - stuende hinter einer Tuer, die
//  schon zugefallen ist. Die Wahl gehoert in denselben Ablauf,
//  vor den Sprung.
//
//  Aufgerufen wird S_Kanten an ZWEI Stellen:
//
//    1. direkt nach dem Binden eines neuen Geschenks
//    2. oben im "hat schon ein Geschenk"-Zweig, fuer alle, die
//       ihr Geschenk gewaehlt haben, bevor es die Wahl gab
//
//  Beide Male kehrt es von selbst zurueck, wenn nichts zu tun
//  ist. Der Ablauf ringsum merkt also nicht, ob gewaehlt wurde.
//
//  WOHER DIE LISTE KOMMT
//
//  npc/custom/safa_skillwahl.txt, erzeugt von
//  tools/isekai/skillwahl.py. Dort steht auch, was als
//  Angriffsfertigkeit zaehlt und warum.
// ============================================================
S_Kanten:

	// ------------------------------------------------------------
	//  Drei Gruende zu schweigen - und warum sie fuer einen GM
	//  nicht still sein duerfen
	// ------------------------------------------------------------
	//  Dieser Abschnitt kehrt in drei Faellen wortlos zurueck. Fuer
	//  den Spieler ist das richtig: er soll nicht bei jedem Besuch
	//  an eine Entscheidung erinnert werden, die ihn nichts angeht
	//  oder die er laengst getroffen hat.
	//
	//  Fuer den, der es baut, ist dasselbe Schweigen aber nicht von
	//  einem Fehler zu unterscheiden - "man waehlt das Geschenk und
	//  ist draussen" sieht bei einem Tier-F-Charakter genauso aus
	//  wie bei einem kaputten Sprungziel. Deshalb sagt es einem GM,
	//  warum es nichts getan hat.
	//
	//  dispbottom und nicht mes: der Ablauf ringsum ist mitten in
	//  einem Fenster, und ein mes an dieser Stelle wuerde ihn
	//  zerreissen.
	// ------------------------------------------------------------

	if (isekai_gift == 0) {
		// Kann hier eigentlich nicht vorkommen, beide Aufrufer
		// pruefen vorher. Ein Zugriff auf Steckplatz -1 waere still
		// und falsch.
		if (getgroupid() >= 99)
			dispbottom "[Edges] No gift bound - nothing to sharpen.";
		return;
	}

	.@tier = callfunc("F_SafaGiftTier", isekai_gift);

	// Tier F bis C: F und E haben gar keinen Fertigkeitsbonus,
	// D und C behalten ihre feste Vergabe. Nichts zu waehlen.
	//
	// Das ist der Fall, der am ehesten fuer einen Fehler gehalten
	// wird: Tier B verlangt 450 Union-Punkte, und die gibt es nur
	// fuer Charaktere des Accounts auf Level 500. Ein frischer
	// Account steht bei 0 und kann ausschliesslich Tier F binden.
	// Zum Testen hilft "Set test points (GM)" im Hauptmenue.
	if (.@tier < 4) {
		if (getgroupid() >= 99)
			dispbottom "[Edges] Gift " + isekai_gift + " is tier "
			         + callfunc("F_SafaTierName", .@tier)
			         + " - free edges start at B. Points: "
			         + callfunc("F_SafaPoints") + ", B needs "
			         + callfunc("F_SafaTierReq", 4) + ".";
		return;
	}

	.@slot = (.@tier - 4) * 18 + (isekai_gift % 100) - 1;
	.@pct = callfunc("F_SafaSkillProzent", .@tier);

	// ------------------------------------------------------------
	//  Drei Fragen, in dieser Reihenfolge
	// ------------------------------------------------------------
	//    1. Klinge oder Zauber   S_Weg
	//    2. die zwei Kanten      S_KantenNur
	//    3. die feste Zauberzeit S_Cast
	//
	//  Jede prueft selbst, ob sie schon beantwortet ist, und kehrt
	//  dann wortlos zurueck. Das ist wichtiger, als es aussieht: die
	//  drei sind zu verschiedenen Zeiten dazugekommen, und ein
	//  Charakter kann Frage 2 beantwortet haben, ohne 1 und 3 je
	//  gesehen zu haben. Wer hier mit return abkuerzt, sobald eine
	//  Frage erledigt ist, sperrt genau diese Leute aus.
	// ------------------------------------------------------------
	callsub S_Weg, .@pct;

	if (isekai_skill1 == 0)
		callsub S_KantenNur, .@slot, .@pct, .@tier;
	else if (getgroupid() >= 99)
		dispbottom "[Edges] Already set: " + isekai_skill1
		         + " and " + isekai_skill2 + ".";

	callsub S_Cast, callfunc("F_SafaStatWert", .@tier);
	return;


// ------------------------------------------------------------
//  Zweite Frage: die zwei Kanten
// ------------------------------------------------------------
//  getarg(0) Steckplatz, getarg(1) Tier-Prozentsatz, getarg(2)
//  das Tier selbst fuer die Gegenprobe.
S_KantenNur:
	.@slot = getarg(0);
	.@pct = getarg(1);
	.@tier = getarg(2);

	mes "[Threshold Warden]";
	mes "One thing is still loose. A gift like";
	mes "yours does not sharpen everything - it";
	mes "sharpens ^0000FFtwo^000000 things, by ^0000FF" + .@pct + " percent^000000";
	mes "each.";
	next;

	mes "[Threshold Warden]";
	mes "Which two is yours to say.";
	mes "^FF0000Say it once. It does not come loose";
	mes "again.^000000";
	next;

	// --- Erste Kante ---
	.@i1 = callsub(S_KanteWaehlen, .@slot, -1, "first");
	if (.@i1 < 0) {
		callsub S_KantenSpaeter;
		return;
	}

	// --- Zweite Kante ---
	.@i2 = callsub(S_KanteWaehlen, .@slot, .@i1, "second");
	if (.@i2 < 0) {
		callsub S_KantenSpaeter;
		return;
	}

	mes "[Threshold Warden]";
	mes "  ^0000FF" + getvariableofnpc(.sk_name$[.@i1], "#safa_skillwahl") + "^000000";
	mes "  ^0000FF" + getvariableofnpc(.sk_name$[.@i2], "#safa_skillwahl") + "^000000";
	mes " ";
	mes "^FF0000This cannot be undone.^000000";
	next;

	if (select("Set them:Let me look again") == 2) {
		callsub S_KantenSpaeter;
		return;
	}

	// Gegenprobe vor dem Schreiben. Zwischen der Auswahl und dieser
	// Zeile liegen mehrere Klicks, und in denen passt ein GM, der
	// das Geschenk abgenommen hat - dann zeigte der Steckplatz auf
	// einen fremden Klassenbaum.
	if (isekai_gift == 0 || callfunc("F_SafaGiftTier", isekai_gift) != .@tier) {
		mes "[Threshold Warden]";
		mes "^FF6666Your gift changed while we spoke.^000000";
		mes "Ask me again.";
		next;
		return;
	}

	set isekai_skill1,  getvariableofnpc(.sk_id[.@i1],  "#safa_skillwahl");
	set isekai_skill1b, getvariableofnpc(.sk_id2[.@i1], "#safa_skillwahl");
	set isekai_skill2,  getvariableofnpc(.sk_id[.@i2],  "#safa_skillwahl");
	set isekai_skill2b, getvariableofnpc(.sk_id2[.@i2], "#safa_skillwahl");

	callfunc "F_SafaRefresh";
	specialeffect2 EF_ANGEL2;

	mes "[Threshold Warden]";
	mes "Then that is what you are.";
	next;
	return;


// ------------------------------------------------------------
//  Erste Frage: Klinge oder Zauber
// ------------------------------------------------------------
//  getarg(0) ist der Tier-Prozentsatz, nur fuer die Anzeige.
//
//  Physisch gibt bAtkRate, magisch bMatkRate - beides in voller
//  Hoehe, beides ohne Bindung an eine Waffengattung.
//
//  WAS DAMIT WEGFAELLT
//
//  Bis zum Umbau gab jedes Geschenk bWeaponDamageRate auf ein bis
//  zwei feste Waffengattungen. Das hat zweimal geschadet: es
//  ruehrte Magie gar nicht an (pc.cpp:4709, battle.cpp:2439 - nur
//  der physische Pfad), und wer seine Klasse mit einer anderen
//  Waffe spielte als der vorgesehenen, ging ebenfalls leer aus.
//  Ein Kagerou, der Ninja-Magie wirkt, bekam von einem Tier-B-
//  Geschenk ganze 25 Statuspunkte.
//
//  bAtkRate kennt weder Gattung noch Ausnahme.
//
//  WER NICHT ANTWORTET
//
//  bekommt die Vorgabe aus der Tabelle: das, was sein Geschenk
//  vorher war. Ein Zauberer bleibt Zauberer, ein Schlaeger
//  Schlaeger. Der Abbruch ist deshalb harmlos.
S_Weg:
	.@p = getarg(0);

	if (isekai_pfad > 0) {
		if (getgroupid() >= 99)
			dispbottom "[Path] Already set: "
			         + (isekai_pfad == 1 ? "physical" : "magic") + ".";
		return;
	}

	mes "[Threshold Warden]";
	mes "A gift has to know what kind of harm it";
	mes "is for. Steel, or the other thing.";
	next;

	mes "[Threshold Warden]";
	mes "Whichever you name, it gives ^0000FF" + .@p + " percent^000000.";
	mes "^777777It does not care what you are holding.^000000";
	next;

	.@w = select("^0000FFPhysical^000000 - attack power:"
	           + "^0000FFMagic^000000 - magic attack power:"
	           + "^777777Neither, not now^000000");

	if (.@w == 3) {
		mes "[Threshold Warden]";
		mes "^777777Then it keeps the shape it came in.^000000";
		next;
		return;
	}

	// Gegenprobe vor dem Schreiben, wie ueberall in diesem Ablauf.
	if (isekai_gift == 0) {
		mes "[Threshold Warden]";
		mes "^FF6666Your gift changed while we spoke.^000000";
		next;
		return;
	}

	set isekai_pfad, .@w;
	callfunc "F_SafaRefresh";
	specialeffect2 EF_ANGEL2;

	mes "[Threshold Warden]";
	mes "^0000FF" + (.@w == 1 ? "Steel" : "The other thing") + "^000000, then.";
	next;
	return;


// ------------------------------------------------------------
//  Dritte Frage: die feste Zauberzeit
// ------------------------------------------------------------
//  getarg(0) ist der volle Statuswert des Tiers.
//
//  Ja  -> bFixedCastrate,-100 und die HALBEN Statuswerte
//  Nein-> volle Statuswerte, keine Verkuerzung
//
//  WARUM DAS EINE ECHTE FRAGE IST
//
//  In Renewal besteht eine Zauberzeit aus zwei Teilen. Der
//  variable sinkt mit DEX und INT und wird null bei
//  DEX*2 + INT >= 530 (conf/battle/skill.conf, vcast_stat_scale).
//  Der feste Teil - 20 Prozent - wird von Statuswerten UEBERHAUPT
//  NICHT beruehrt. Wer alles hochzieht, kommt bei 80 Prozent an
//  und dann nicht weiter; den Rest nimmt nur bFixedCastrate weg.
//
//  Fuer einen Zauberer ist das also viel wert. Fuer jemanden, der
//  zuschlaegt, ist es nichts - und der nimmt lieber die vollen
//  Statuswerte mit. Genau deshalb steht die Frage hier und nicht
//  fest im Geschenk.
//
//  -100 und nicht weniger: bFixedCastrate stapelt nicht,
//  pc.cpp:4422 rechnet min(fixcastrate, val). Es gilt immer nur
//  der staerkste Wert, und -100 nimmt die feste Zeit ganz weg.
//
//  Wer nicht antwortet, bekommt die vollen Statuswerte und keine
//  Verkuerzung.
S_Cast:
	.@s = getarg(0);

	if (isekai_cast > 0) {
		if (getgroupid() >= 99)
			dispbottom "[Cast] Already set: "
			         + (isekai_cast == 1 ? "no fixed cast, half stats"
			                             : "full stats") + ".";
		return;
	}

	mes "[Threshold Warden]";
	mes "Last thing. Some gifts cut away the part";
	mes "of a spell that no amount of wit makes";
	mes "faster.";
	next;

	mes "[Threshold Warden]";
	mes "You can have that - or you can have the";
	mes "whole weight of what the gift carries.";
	mes "Not both.";
	mes " ";
	mes "  ^0000FFno fixed cast time^000000, and ^FF6666" + (.@s / 2) + "^000000 to all stats";
	mes "  ^0000FFfull " + .@s + "^000000 to all stats, and the wait stays";
	next;

	.@w = select("^0000FFCut the wait^000000 - half the stats:"
	           + "^0000FFKeep the stats^000000 - live with the wait:"
	           + "^777777Neither, not now^000000");

	if (.@w == 3) {
		mes "[Threshold Warden]";
		mes "^777777Then you keep the stats and the wait.^000000";
		next;
		return;
	}

	if (isekai_gift == 0) {
		mes "[Threshold Warden]";
		mes "^FF6666Your gift changed while we spoke.^000000";
		next;
		return;
	}

	set isekai_cast, .@w;
	callfunc "F_SafaRefresh";
	specialeffect2 EF_ANGEL2;

	mes "[Threshold Warden]";
	// Klammern, obwohl es nach einer Zeile aussieht: der erste Zweig
	// hat zwei mes, und ohne Klammern gehoerte das zweite nicht mehr
	// zum if - das else stuende dann hinter einer Anweisung, die der
	// Parser nicht mehr erwartet.
	if (.@w == 1) {
		mes "No more waiting, then. It costs you half";
		mes "of what the gift was carrying.";
	} else {
		mes "The full weight it is.";
	}
	next;
	return;


// ------------------------------------------------------------
//  Abbruch
// ------------------------------------------------------------
//  Wer nicht waehlen will, wird nicht gezwungen - aber er muss
//  wissen, dass er ohne Bonus hinausgeht und nur ueber diese
//  Halle wieder herankommt. Die erreicht er von aussen mit
//  @warp valkyrie 48,9; das kann ein Spieler nicht, ein GM
//  schon.
S_KantenSpaeter:
	mes "[Threshold Warden]";
	mes "^FF6666Then your gift stays blunt.^000000";
	mes "Nothing of it is lost - but nothing of";
	mes "it works either, until you name the two.";
	next;
	mes "[Threshold Warden]";
	mes "Come back to this hall when you know.";
	mes "^777777It is not on any road. You will have to";
	mes "ask someone to put you here.^000000";
	next;
	return;


// ------------------------------------------------------------
//  Eine Kante waehlen
// ------------------------------------------------------------
//  getarg(0) Steckplatz, getarg(1) schon vergebener Index
//  (-1 = keiner), getarg(2) "first" oder "second" fuer den Text.
//
//  Rueckgabe: Index in .sk_id, oder -1 bei Abbruch.
//
//  Zwei Menues statt einem: ein Shinkiro hat 33 waehlbare
//  Fertigkeiten, und ein select() mit 33 langen Eintraegen
//  sprengt das Menuepaket zum Client. Die Gruppen sind ohnehin
//  die natuerliche Ordnung - man denkt in Klassenstufen.
S_KanteWaehlen:
	.@s = getarg(0);
	.@aus = getarg(1);
	.@was$ = getarg(2);

	.@gv = getvariableofnpc(.g_von[.@s], "#safa_skillwahl");
	.@ga = getvariableofnpc(.g_anz[.@s], "#safa_skillwahl");

	while (1) {
		mes "[Threshold Warden]";
		mes "The ^0000FF" + .@was$ + "^000000 edge. Where from?";

		.@m$ = "";
		for (.@k = 0; .@k < .@ga; .@k++)
			.@m$ += getvariableofnpc(.gr_name$[.@gv + .@k], "#safa_skillwahl") + ":";
		.@m$ += "^777777Not now^000000";

		.@g = select(.@m$) - 1;
		if (.@g == .@ga)
			return -1;

		.@sv = getvariableofnpc(.gr_von[.@gv + .@g], "#safa_skillwahl");
		.@sa = getvariableofnpc(.gr_anz[.@gv + .@g], "#safa_skillwahl");

		.@m2$ = "";
		for (.@k = 0; .@k < .@sa; .@k++) {
			// Die schon vergebene Kante wird grau gezeigt statt
			// weggelassen: eine Liste, die sich zwischen erstem und
			// zweitem Durchgang verkuerzt, laesst einen an der
			// eigenen Erinnerung zweifeln.
			.@m2$ += ((.@sv + .@k) == .@aus ? "^777777" : "") +
			    getvariableofnpc(.sk_name$[.@sv + .@k], "#safa_skillwahl") +
			    ((.@sv + .@k) == .@aus ? " - taken^000000" : "") + ":";
		}
		.@m2$ += "^777777Back^000000";

		.@w = select(.@m2$) - 1;
		if (.@w == .@sa)
			continue;
		if ((.@sv + .@w) == .@aus) {
			mes "[Threshold Warden]";
			mes "^FF6666That one already holds your first edge.^000000";
			mes "Two edges on one blade is one edge.";
			next;
			continue;
		}
		return .@sv + .@w;
	}
	return -1;
}
