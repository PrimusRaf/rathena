//  WORAUF DER RAT BERUHT
//
//  Die Tabelle unten ist vorgerechnet aus:
//    npc/re/mobs/**.txt   Besatz je Karte
//    db/re/mob_db.yml     Level, Erfahrung, Angriff, Verhalten
//    db/re/map_cache.dat  ob die Karte existiert
//    npc/**/warps         Weg zur naechsten Stadt
//
//  Ein Skript kann 2.600 Monster ueber 300 Karten nicht bei
//  jedem Gespraech sortieren - deshalb vorgerechnet.
//
//  WARUM DIE LEBENSPUNKTE ENTSCHEIDEN UND NICHT DIE STATS
//
//  Naheliegend waere, STR/AGI/VIT gegen eine Erwartung fuer
//  das Level zu halten. Das misst aber nur die Bauweise, nicht
//  die Ausruestung - ein Level 80 in geschenkter Ruestung und
//  einer in nichts haben dieselben Stats.
//
//  MaxHp misst beides zugleich, denn dort schlagen Ausruestung,
//  Karten UND die Todesmale durch (bMaxHPrate in
//  safa_systems.txt). Armstrong vergleicht deshalb den
//  haertesten Angriff auf einer Karte mit den tatsaechlichen
//  Lebenspunkten. Wer sich besser ausruestet, bekommt sofort
//  einen anderen Rat.
//

prontera,163,168,4	script	Armstrong#safa	4_M_SITDOWN,{
	mes "[Armstrong]";
	mes "Mind the notebooks, friend. Forty years of them.";
	mes "Every field, every cave, and every thing that lives in one. Counted, weighed, written down.";
	next;
	mes "[Armstrong]";
	mes "I do not dig for gold. I dig for numbers.";
	mes "Ask, and I will tell you where yours grow fastest.";
	next;

L_Menue:
	switch (select("Where should I be hunting?",
	               "Show me everything for my level.",
	               "These death marks - what are they costing me?",
	               "Nothing for now.")) {
	case 1: callsub S_Rat;   break;
	case 2: callsub S_Liste; break;
	case 3: callsub S_Male;  break;
	case 4:
		mes "[Armstrong]";
		mes "Mind your step out there.";
		mes "The ground remembers everyone who did not.";
		close;
	}
	goto L_Menue;

// ------------------------------------------------------------
//  Der eigentliche Rat
// ------------------------------------------------------------
S_Rat:
	.@band = BaseLevel / 10;
	if (.@band >= 26)
		.@band = 26 - 1;

	.@hp  = MaxHp;
	.@tod = death_count;

	// Wieviel darf ein einzelner Treffer kosten? Ein Viertel laesst
	// sich noch weglaufen. Wer Todesmale traegt, hat weniger Luft -
	// und MaxHp ist durch die Male ohnehin schon kleiner, der Malus
	// wirkt also zweimal. Das ist so gewollt: gerade dann soll er
	// vorsichtiger geschickt werden.
	.@grenze = .@hp / 4;
	if (.@tod >= 3) .@grenze = .@hp / 6;
	if (.@tod >= 8) .@grenze = .@hp / 10;

	// Die Baender sind nach Erfahrung geordnet, der erste Eintrag,
	// der durchkommt, ist also der beste. Findet sich im eigenen
	// Band nichts, geht es nach unten - das ist der Fall, wenn
	// jemand sein Level ueberholt hat und ohne Ausruestung dasteht.
	// --------------------------------------------------------------
	//  Warum das in Stufen sucht und nicht in einem Durchgang
	// --------------------------------------------------------------
	//  Der erste Entwurf hatte zwei harte Regeln: nichts, was zu
	//  hart zuschlaegt, und bei Todesmalen nichts Aggressives. Dann
	//  ist er notfalls Band um Band nach unten gegangen.
	//
	//  Das Ergebnis war unbrauchbar. Oberhalb von 120 ist jede
	//  ertragreiche Karte zu hundert Prozent aggressiv - ein
	//  Spieler auf 150 mit fuenf Malen bekam die Roboterfabrik
	//  genannt, vierzig Level unter ihm. Formal regelkonform und
	//  als Rat wertlos.
	//
	//  Also: hoechstens zwei Baender nach unten, und die Aggro-
	//  Regel ist ein Wunsch, keine Bedingung. Faellt sie weg, sagt
	//  Armstrong es dazu - und schickt ihn zur Waechterin.
	.@boden = .@band - 2;
	if (.@boden < 0) .@boden = 0;

	.@wahl    = -1;   // bestes offenes Gebiet
	.@tor     = -1;   // bestes ueberhaupt, aber hinter einer Tuer
	.@unruhig = 0;    // Ruhewunsch liess sich nicht erfuellen

	for (.@streng = 0; .@streng < 2; .@streng++) {
		for (.@b = .@band; .@b >= .@boden; .@b--) {
			for (.@i = 0; .@i < 5; .@i++) {
				.@k = .@b * 5 + .@i;
				if (.karte$[.@k] == "") continue;
				if (.wucht[.@k] > .@grenze) continue;
				if (.@streng == 0 && .@tod >= 3 && .aggro[.@k] > 40)
					continue;
				if (.tor[.@k]) {
					if (.@tor < 0) .@tor = .@k;
				} else if (.@wahl < 0) {
					.@wahl = .@k;
				}
			}
			if (.@wahl >= 0) break;
		}
		if (.@wahl >= 0) break;
		if (.@streng == 0 && .@tod >= 3) .@unruhig = 1;
	}

	// Steht wirklich nur Verschlossenes zur Wahl, dann eben das.
	if (.@wahl < 0 && .@tor >= 0) {
		.@wahl = .@tor;
		.@tor  = -1;
	}

	// Immer noch nichts: jetzt doch bis ganz nach unten. Wer hier
	// landet, ist deutlich unterausgeruestet fuer sein Level.
	.@zurueck = 0;
	if (.@wahl < 0) {
		for (.@b = .@boden - 1; .@b >= 0 && .@wahl < 0; .@b--) {
			for (.@i = 0; .@i < 5; .@i++) {
				.@k = .@b * 5 + .@i;
				if (.karte$[.@k] == "") continue;
				if (.wucht[.@k] > .@grenze) continue;
				.@wahl = .@k;
				.@zurueck = 1;
				break;
			}
		}
	}

	// Und erwaehnen lohnt nur, was auch mehr bringt.
	//
	// ACHTUNG, verschachtelt und nicht mit && verkettet:
	// rAthena wertet && NICHT verkuerzt aus. In script.cpp:3839
	// steht "case C_LAND: ret = (i1 && i2);" - beide Operanden sind
	// da laengst vom Stapel geholt, also ausgerechnet. Ein
	//
	//     if (.@tor >= 0 && .exp[.@tor] <= .exp[.@wahl])
	//
	// greift deshalb auch dann auf .exp[-1] zu, wenn .@tor noch -1
	// ist. Der Server meldet "getelementofarray: index out of range
	// (-1)" und bricht das Gespraech ab - vor dem ersten mes, also
	// ohne dass der Spieler ueberhaupt Text sieht. Der NPC wirkt
	// dann schlicht kaputt.
	if (.@tor >= 0) {
		if (.@wahl >= 0) {
			if (.exp[.@tor] <= .exp[.@wahl])
				.@tor = -1;
		}
	}

	// Gar nichts gefunden. Dann das Sanfteste, was es gibt - und
	// ehrlich dazusagen, dass es eng wird.
	.@notnagel = 0;
	if (.@wahl < 0) {
		.@wahl = 0;
		for (.@i = 1; .@i < 5; .@i++)
			if (.karte$[.@i] != "" && .wucht[.@i] < .wucht[.@wahl])
				.@wahl = .@i;
		.@notnagel = 1;
	}

	mes "[Armstrong]";
	mes "Hold still. ...Mm.";
	mes " ";
	mes "Base level ^FFCC00" + BaseLevel + "^000000, ^FFCC00"
	    + callfunc("F_InsertComma", .@hp) + "^000000 points of life.";
	if (.@tod > 0)
		mes "And ^FF6666" + .@tod + " death mark"
		    + ((.@tod == 1) ? "" : "s") + "^000000. Those I can see from here.";
	next;

	mes "[Armstrong]";
	if (.@notnagel)
		mes "^FF6666Nothing out there is gentle enough for you yet.^000000 Take the quietest ground I know, and take it slowly.";
	else if (.@zurueck)
		mes "^FF6666You have outrun your own equipment.^000000 Your level says one thing, what you are wearing says another. Go back a way and fix that.";
	else if (.@unruhig)
		mes "^FF6666There is nothing quiet at your level.^000000 Everything up there comes at you on sight - and you are carrying marks.";
	else if (.@tod >= 3)
		mes "Carrying marks like those, you want ground that does not come to you.";
	else
		mes "For what you are, this is the richest ground you can hold.";
	mes " ";
	mes "^00CCFF" + .ort$[.@wahl] + "^000000";
	mes "^777777" + .karte$[.@wahl] + "^000000";
	next;

	mes "[Armstrong]";
	mes "Mostly ^FFCC00" + .mon$[.@wahl] + "^000000, level "
	    + .lvu[.@wahl] + " to " + .lvo[.@wahl] + ".";
	mes "About ^FFCC00" + .dichte[.@wahl] + "^000000 of them standing at any hour,";
	mes "^FFCC00" + callfunc("F_InsertComma", .zaeh[.@wahl]) + "^000000 points of life in each.";
	if (.aggro[.@wahl] == 0)
		mes "^88CC88Not one of them will start a fight.^000000 They wait.";
	else if (.aggro[.@wahl] >= 80)
		mes "^FF6666Nearly all of them come at you on sight.^000000";
	else
		mes "^FFAA55Roughly " + .aggro[.@wahl] + " in a hundred come at you on sight.^000000";
	next;

	// Der abgelegte Wert ist die reine Datenbankerfahrung. Was
	// ankommt, haengt an base_exp_rate. Die Zahl NICHT einbacken,
	// sondern erfragen - dann folgt Armstrong der Einstellung,
	// falls sie sich einmal aendert.
	.@rate = getbattleflag("base_exp_rate");
	.@pro  = .exp[.@wahl] * .@rate / 100;

	mes "[Armstrong]";
	mes "One of them is worth about ^FFCC00"
	    + callfunc("F_InsertComma", .@pro) + "^000000 to you.";
	if (.@pro > 0 && NextBaseExp > 0) {
		.@kills = (NextBaseExp + .@pro - 1) / .@pro;
		mes "Your next level wants ^FFCC00"
		    + callfunc("F_InsertComma", NextBaseExp) + "^000000.";
		mes " ";
		mes "That is ^FFCC00" + callfunc("F_InsertComma", .@kills)
		    + "^000000 of them. Give or take.";
	} else {
		mes "^777777There is nothing above you left to climb.^000000";
	}
	next;

	mes "[Armstrong]";
	if (.tor[.@wahl]) {
		mes "^FF6666One thing.^000000 That door does not simply stand open - there is a chain of work to be done before anyone walks in.";
		mes "If you have not done it, do it. Then the ground is yours.";
		mes " ";
	} else if (.@tor >= 0) {
		mes "There is richer ground than that - ^FFCC00" + .ort$[.@tor]
		    + "^000000, near ^FFCC00"
		    + callfunc("F_InsertComma", .exp[.@tor] * .@rate / 100)
		    + "^000000 a head.";
		mes "^777777But the way in is earned, not walked. Come back when you have opened it.^000000";
		mes " ";
	}
	if (.stadt$[.@wahl] != "") {
		mes "It lies out from ^00CCFF" + .stadt$[.@wahl] + "^000000 - "
		    + .weg[.@wahl] + " crossing"
		    + ((.weg[.@wahl] == 1) ? "" : "s") + " out.";
	}
	if (.warper[.@wahl])
		mes "The ^00CCFFWarper^000000 sets you down there directly. Save your boots.";
	else
		mes "^777777No Warper runs there. That one you walk.^000000";
	if (.@unruhig) {
		next;
		mes "[Armstrong]";
		mes "But you asked me where to hunt, and I will give you the honest answer instead:";
		mes " ";
		mes "^88CC88Have the marks taken off first.^000000 The ^88CC88Threshold Warden^000000 at ^00CCFFprontera 146, 182^000000 does it.";
		mes "Walking in there at " + (.@tod * 5 > 90 ? 90 : .@tod * 5) + "% below yourself is not bravery. It is arithmetic.";
	}
	next;
	return;

// ------------------------------------------------------------
//  Alles, was im Band steht
// ------------------------------------------------------------
S_Liste:
	.@band = BaseLevel / 10;
	if (.@band >= 26)
		.@band = 26 - 1;
	.@rate = getbattleflag("base_exp_rate");

	mes "[Armstrong]";
	mes "Everything in my books for your standing:";
	mes " ";
	for (.@i = 0; .@i < 5; .@i++) {
		.@k = .@band * 5 + .@i;
		if (.karte$[.@k] == "") continue;
		mes "^00CCFF" + .ort$[.@k] + "^000000";
		mes "  ^777777lv " + .lvu[.@k] + "-" + .lvo[.@k] + " | "
		    + callfunc("F_InsertComma", .exp[.@k] * .@rate / 100) + " exp | "
		    + .aggro[.@k] + "% hostile^000000";
	}
	next;
	mes "[Armstrong]";
	mes "The first is the richest. Whether you can stand in it is a different question - ask me that one properly and I will measure you.";
	next;
	return;

// ------------------------------------------------------------
//  Todesmale
// ------------------------------------------------------------
S_Male:
	.@tod = death_count;
	mes "[Armstrong]";
	if (.@tod == 0) {
		mes "Not one on you.";
		mes "Whatever you have been doing, keep doing it.";
		next;
		return;
	}

	// Dieselbe Rechnung wie in safa_systems.txt: fuenf Prozent je
	// Tod, bei neunzig ist Schluss. Wird sie dort geaendert, hier
	// mitaendern.
	.@pen = .@tod * 5;
	if (.@pen > 90) .@pen = 90;

	mes "^FF6666" + .@tod + "^000000 of them. Five in the hundred each, and ninety is as bad as it gets.";
	mes " ";
	mes "You are down ^FF6666" + .@pen + "%^000000 on all of it -";
	mes "strike, magic, life, spirit, armour, dodge and aim.";
	next;
	mes "[Armstrong]";
	mes "It is never the dying that costs you. It is the hour after.";
	mes " ";
	mes "The ^88CC88Threshold Warden^000000 takes them off. She stands at ^00CCFFprontera 146, 182^000000, a short walk north of here.";
	mes "Be clean before you go back out.";
	next;
	return;
