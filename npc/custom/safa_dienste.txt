//============================================================
//  Reset Master
//------------------------------------------------------------
//  50.000z je Ruecksetzung, Statuswerte und Skills getrennt
//  oder zusammen.
//
//  Bewusst OHNE Begrenzung der Anzahl. Das Standardskript
//  (npc/custom/resetnpc.txt, bei uns aus) kennt ein Lebenslimit -
//  das passt zu einem Server, auf dem ein Build eine Entscheidung
//  fuer Monate ist. Bei 25x EXP ist ein neuer Charakter an einem
//  Abend nachgezogen; ein Limit wuerde niemanden bremsen, sondern
//  nur dazu fuehren, dass man den Charakter wegwirft statt ihn
//  umzubauen.
//
//  Das Standardskript setzt die Reset Girl uebrigens auf
//  prontera,150,193 - die Zelle, auf der bis Episode 22 Ironhand
//  stand. Seit dem Umzug nach Veins ist sie frei, die Warnung
//  gilt also nicht mehr. Sie bleibt trotzdem stehen: wer je
//  ueberlegt, Ironhand zurueckzuholen, soll wissen, warum genau
//  diese Zelle keine gute Idee ist.
//============================================================
veins,222,125,4	script	Reset Master#safa	4_F_TELEPORTER,{

	// Lokale Funktionen muessen VOR ihrem ersten Aufruf angemeldet
	// werden (doc/script_commands.txt:1891). Der Name darf nur
	// Buchstaben, Ziffern und Unterstrich enthalten - ein ".@" davor
	// waere ein Syntaxfehler, kein lokaler Gueltigkeitsbereich.
	function zahlen;

	.@preis = 50000;

	mes "[Reset Master]";
	mes "A build is a guess about how you like to play.";
	mes "Guesses are allowed to be wrong.";
	next;

	mes "[Reset Master]";
	mes "^0000FF" + callfunc("F_InsertComma", .@preis) + "z^000000 per reset.";
	mes "You have " + callfunc("F_InsertComma", Zeny) + "z.";
	next;

	switch (select("Reset stats:Reset skills:Reset both:What happens to my points?:Leave")) {

	case 1:
		if (!zahlen(.@preis)) close;
		resetstatus;
		mes "[Reset Master]";
		mes "Your stat points are back in your hands.";
		mes "Spend them better this time.";
		close;

	case 2:
		if (!zahlen(.@preis)) close;
		resetskill;
		mes "[Reset Master]";
		mes "Skills unlearned. Quest skills included -";
		mes "you will need to earn those again.";
		close;

	case 3:
		if (!zahlen(.@preis * 2)) close;
		resetstatus;
		resetskill;
		mes "[Reset Master]";
		mes "Both done. You are the same level you were,";
		mes "and nothing else about you is settled.";
		close;

	case 4:
		mes "[Reset Master]";
		mes "Nothing is lost. Your base and job level stay,";
		mes "and you get back exactly the points you spent.";
		next;
		mes "[Reset Master]";
		mes "One warning: resetting skills also removes";
		mes "^0000FFquest skills^000000. Those are not refunded as points -";
		mes "you have to do the quests again.";
		close;

	case 5:
		mes "[Reset Master]";
		mes "Come back when the doubt gets loud enough.";
		close;
	}

// Preis einziehen. Die Pruefung sitzt hier und nicht oben, weil
// zwischen dem Anzeigen des Preises und der Auswahl Zeit vergeht -
// in der kann der Spieler in einem zweiten Fenster einkaufen.
function	zahlen	{
	if (Zeny < getarg(0)) {
		mes "[Reset Master]";
		mes "That is " + callfunc("F_InsertComma", getarg(0) - Zeny) + "z short.";
		return 0;
	}
	Zeny -= getarg(0);
	return 1;
}
}


//============================================================
//  Stablemaster - Verleih
//------------------------------------------------------------
//  WARUM DAS KEIN "UNIVERSAL RENTAL" IST
//
//  Die Skriptbefehle pruefen Klasse und Skill selbst, und zwar im
//  Serverkern - nicht im Skript. Aus doc/script_commands.txt:
//
//    setcart      braucht MC_PUSHCART
//    setfalcon    braucht HT_FALCON
//    setriding    braucht KN_RIDING, und "will not work at all
//                 if they aren't of a class which can ride"
//    setdragon    braucht Rune Knight + RK_DRAGONTRAINING
//    setmadogear  braucht Mechanic + NC_MADOLICENCE
//
//  Ein NPC, der einem Magier einen Peco gibt, laesst sich also
//  nicht schreiben - der Aufruf laeuft ins Leere. Deshalb zeigt
//  dieser hier nur an, wozu der Spieler berechtigt ist, und sagt
//  bei allem anderen, was fehlt. Das ist ehrlicher als eine Liste
//  mit Eintraegen, die nichts tun.
//
//  Die eine Ausnahme ist setmounting: das Cash-Reittier haengt an
//  keiner Klasse. Es ist damit das einzige wirklich universelle
//  Angebot und steht deshalb zuerst.
//
//  Der Warg faellt aus der Reihe. Fuer ihn gibt es keinen
//  setwarg-Befehl - er haengt an RA_WUGMASTERY, und dieser Skill
//  verlangt laut skill_db.yml die Wolf's_Flute (6124) mit
//  Amount: 0. Das Stueck wird also nicht verbraucht, muss aber im
//  Beutel liegen, sonst laesst sich der Skill gar nicht wirken.
//
//  Im Original bekommt man sie einmalig vom Ranger-Meister am Ende
//  der Jobquest (npc/re/jobs/3-1/ranger.txt:1743). Wer bei uns
//  ueber den Jobmaster zum Ranger wird, laeuft an dieser Stelle
//  vorbei und steht danach ohne Warg da - ohne dass irgendetwas
//  darauf hinweist, denn der Skill ist ja erlernt und sichtbar.
//
//  Deshalb gibt der Stablemaster sie kostenlos aus. Sie ist
//  NoTrade, NoSell und NoStorage, also nicht weiterreichbar; es
//  gibt keinen Grund, dafuer Geld zu verlangen.
//============================================================
veins,190,125,4	script	Stablemaster#safa	4_M_JOB_KNIGHT1,{

	mes "[Stablemaster]";
	mes "Everything here is borrowed, not sold.";
	mes "Bring it back in one piece and we stay friends.";
	next;

	.@m$ = "Universal mount";
	if (ismounting()) .@m$ = "Universal mount ^777777(dismount)^000000";

	.@c$ = "Cart";
	if (!getskilllv("MC_PUSHCART")) .@c$ = "Cart ^777777(needs Pushcart)^000000";
	else if (checkcart()) .@c$ = "Cart ^777777(return it)^000000";

	.@f$ = "Falcon";
	if (!getskilllv("HT_FALCON")) .@f$ = "Falcon ^777777(needs Falcon Mastery)^000000";
	else if (checkfalcon()) .@f$ = "Falcon ^777777(release)^000000";

	.@p$ = "Riding bird";
	if (!getskilllv("KN_RIDING")) .@p$ = "Riding bird ^777777(needs Riding)^000000";
	else if (checkriding()) .@p$ = "Riding bird ^777777(stable it)^000000";

	.@d$ = "Dragon";
	if (!getskilllv("RK_DRAGONTRAINING")) .@d$ = "Dragon ^777777(needs Dragon Training)^000000";
	else if (checkdragon()) .@d$ = "Dragon ^777777(stable it)^000000";

	.@g$ = "Mado Gear";
	if (!getskilllv("NC_MADOLICENCE")) .@g$ = "Mado Gear ^777777(needs Mado License)^000000";
	else if (checkmadogear()) .@g$ = "Mado Gear ^777777(park it)^000000";

	// Die Floete ist kein Reittier, sondern ein Gegenstand - sie
	// steht trotzdem hier, weil der Warg fuer den Spieler dasselbe
	// ist wie Falke und Peco: etwas, das mitkommt.
	.@w$ = "Wolf Flute ^777777(free)^000000";
	if (!getskilllv("RA_WUGMASTERY"))
		.@w$ = "Wolf Flute ^777777(needs Warg Mastery)^000000";
	else if (countitem(6124))
		.@w$ = "Wolf Flute ^777777(you already carry one)^000000";

	switch (select(.@m$ + ":" + .@c$ + ":" + .@f$ + ":" + .@p$ + ":" + .@d$
	               + ":" + .@g$ + ":" + .@w$
	               + ":Why can I not take everything?:Leave")) {

	case 1:
		setmounting;
		mes "[Stablemaster]";
		if (ismounting()) mes "Up you go.";
		else mes "Off you get.";
		close;

	case 2:
		if (!getskilllv("MC_PUSHCART")) { .@fehlt$ = "Pushcart"; break; }
		if (checkcart()) setcart 0;
		else setcart 1;
		mes "[Stablemaster]";
		if (checkcart()) mes "One cart. Mind the corners.";
		else mes "Cart returned.";
		close;

	case 3:
		if (!getskilllv("HT_FALCON")) { .@fehlt$ = "Falcon Mastery"; break; }
		if (checkfalcon()) setfalcon 0;
		else setfalcon 1;
		mes "[Stablemaster]";
		if (checkfalcon()) mes "She likes you. That is rare.";
		else mes "Off she goes.";
		close;

	case 4:
		if (!getskilllv("KN_RIDING")) { .@fehlt$ = "Riding"; break; }
		if (checkriding()) setriding 0;
		else setriding 1;
		mes "[Stablemaster]";
		if (checkriding()) mes "Hold the reins, not the feathers.";
		else mes "Back in the stable.";
		close;

	case 5:
		if (!getskilllv("RK_DRAGONTRAINING")) { .@fehlt$ = "Dragon Training"; break; }
		setdragon;
		mes "[Stablemaster]";
		if (checkdragon()) mes "Do not let it near the thatched roofs.";
		else mes "Stabled.";
		close;

	case 6:
		if (!getskilllv("NC_MADOLICENCE")) { .@fehlt$ = "Mado License"; break; }
		if (checkmadogear()) setmadogear 0;
		else setmadogear 1, MADO_ROBOT;
		mes "[Stablemaster]";
		if (checkmadogear()) mes "Fuel is your problem, not mine.";
		else mes "Parked.";
		close;

	case 7:
		if (!getskilllv("RA_WUGMASTERY")) { .@fehlt$ = "Warg Mastery"; break; }
		mes "[Stablemaster]";
		if (countitem(6124)) {
			mes "You have one already. They do not wear out,";
			mes "and a second whistle calls the same wolf.";
			close;
		}
		// Die Floete wiegt nichts, aber ein freier Platz muss da
		// sein. Ohne diese Pruefung scheitert getitem stillschweigend
		// und der Stablemaster sagt trotzdem "Here" - der Spieler
		// stuende dann mit leeren Haenden und ohne Hinweis da.
		if (!checkweight(6124, 1)) {
			mes "Not with that bag, you will not.";
			mes "Make room first - I am not stuffing it in there.";
			close;
		}
		mes "Ah - a Ranger without a whistle.";
		mes "Happens more often than it should.";
		next;
		getitem 6124, 1;   // Wolf's_Flute
		mes "[Stablemaster]";
		mes "Here. No charge - it is a piece of carved bone,";
		mes "not a horse.";
		next;
		mes "[Stablemaster]";
		mes "^777777Keep it in your bag. Warg Mastery will not work";
		mes "without it, though nothing on your skill bar will";
		mes "tell you so.^000000";
		close;

	case 8:
		mes "[Stablemaster]";
		mes "Not my rule. The mounts answer to training,";
		mes "and training answers to your class.";
		next;
		mes "[Stablemaster]";
		mes "The ^0000FFuniversal mount^000000 is the exception - anyone";
		mes "can ride it. Everything else needs the skill";
		mes "that goes with the animal.";
		next;
		mes "[Stablemaster]";
		mes "The ^0000FFwolf^000000 is its own case. No training I can";
		mes "give you - it answers to a whistle, and the";
		mes "whistle you simply have to be carrying.";
		close;

	case 9:
		mes "[Stablemaster]";
		mes "Mind the droppings on your way out.";
		close;
	}

	mes "[Stablemaster]";
	mes "You would need ^0000FF" + .@fehlt$ + "^000000 for that.";
	mes "Come back when you have it.";
	close;
}


//============================================================
//  Die Kafra
//------------------------------------------------------------
//  WARUM DAS SEIT EPISODE 22 KEIN DUPLIKAT MEHR IST
//
//  Bis zum Umzug stand hier duplicate(kaf_prontera). Das war in
//  Prontera richtig und in Veins ein Fehler, denn ein Duplikat
//  teilt sich den GESAMTEN Skriptkoerper des Originals - und in
//  dem steht (npc/kafras/kafras.txt:244):
//
//      savepoint "prontera",157,327,1,1;
//
//  Die Kafra haette in Veins also den Speicherpunkt nach
//  Prontera gesetzt. Der Spieler klickt "Save", bekommt die
//  Bestaetigung "in the city of Prontera" zu lesen, ueberliest
//  sie - und steht beim naechsten Tod am anderen Ende der Welt.
//  Ein Fehler, der erst auffaellt, wenn er weh tut.
//
//  Deshalb hier eine eigene Fassung. Sie ruft dieselben drei
//  amtlichen Funktionen auf wie jede Stadt-Kafra, es ist also
//  keine nachgebaute Kopie: Lager, Karren und Preise bleiben
//  automatisch richtig, wenn rAthena daran etwas aendert. Eigen
//  sind nur die drei Zeilen, die ortsgebunden sind.
//
//  WARUM MENUE 3 UND NICHT 0
//
//  0 ist das volle Menue samt Teleportdienst. Dessen Zielliste
//  in functions_kafras.txt kennt Veins gar nicht - die Stadt hat
//  ihren eigenen Teleporter, die "Kafra Teleportation" auf
//  196,133. Menue 3 ist die uebliche Fassung ohne Teleport:
//  Speichern, Lager, Karren, Auskunft.
//
//  VEINS HATTE VORHER GAR KEINE LAGER-KAFRA
//
//  Nachgesehen in npc/kafras: auf veins steht nur Teleportdienst
//  (196,133) und Zonda (198,133), dazu das Gildenlager auf
//  205,130. Ein Lager fuer den einzelnen Spieler gab es hier
//  nicht. Mit der Hauptstadt kommt es jetzt dazu.
//
//  Preise wie in Prontera: 40z Lager, 800z Karren.
//============================================================
veins,230,125,4	script	Kafra Employee#safa	4_F_KAFRA1,{
	cutin "kafra_01",2;
	callfunc "F_KafSet";
	mes "[Kafra Employee]";
	mes "Welcome to Veins.";
	mes "The Kafra Corporation";
	mes "keeps an office even at";
	mes "the edge of the desert.";
	mes "How may I help you?";
	callfunc "F_Kafra",5,3,0,40,800;
	savepoint "veins",210,123,1,1;
	callfunc "F_KafEnd",0,1,"in the city of Veins";
}
