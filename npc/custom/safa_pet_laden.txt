prontera,167,178,4	script	Pet Caretaker#safa	4_M_DOGTRAVELER,{

	mes "[Pet Caretaker]";
	mes "Every one of them eats something";
	mes "different, and every one of them is";
	mes "certain that its own food is the only";
	mes "real food. I stock all of it.";
	next;

	switch (select("Buy food:Buy an accessory:What does my pet need?:Nothing today")) {

	case 1:
		close2;
		callshop "SafaPetFood", 1;
		end;

	case 2:
		mes "[Pet Caretaker]";
		mes "Fair warning: an accessory is for the";
		mes "look. It does not add to what your pet";
		mes "already gives you.";
		mes " ";
		mes "And putting it on ^FF6666uses it up^000000 - it";
		mes "leaves your bag for good.";
		next;
		close2;
		callshop "SafaPetAcce", 1;
		end;

	case 3:
		// PETINFO_FOODID kommt aus der Tierdatenbank, ist also
		// immer richtig. Fuer den Schmuck gibt es keine solche
		// Auskunft - getpetinfo kennt AcceID nicht (script.cpp,
		// BUILDIN_FUNC(getpetinfo)). Deshalb die Tabelle unten.
		if (getpetinfo(PETINFO_ID) == 0) {
			mes "[Pet Caretaker]";
			mes "Bring the animal along and I will tell";
			mes "you what it wants.";
			close;
		}

		mes "[Pet Caretaker]";
		mes "^0000FF" + getpetinfo(PETINFO_NAME) + "^000000, is it.";
		mes " ";

		.@futter = getpetinfo(PETINFO_FOODID);
		if (.@futter > 0) {
			mes "Food: ^0000FF" + getitemname(.@futter) + "^000000";
			mes "In your bag: ^0000FF" + countitem(.@futter) + "^000000";
		} else
			mes "This one does not eat at all.";

		.@i = inarray(.tier, getpetinfo(PETINFO_CLASS));
		mes " ";
		if (.@i >= 0)
			mes "Accessory: ^0000FF" + getitemname(.schmuck[.@i]) + "^000000";
		else
			mes "There is no accessory for this one.";

		mes " ";
		mes "Intimacy: ^0000FF" + getpetinfo(PETINFO_INTIMATE) + "^000000 of 1000";
		mes "Hunger: ^0000FF" + getpetinfo(PETINFO_HUNGRY) + "^000000 of 100";
		mes " ";
		mes "^777777@petfeed lets it eat on its own,^000000";
		mes "^777777out of your bag, once it gets hungry.^000000";
		close;

	case 4:
		mes "[Pet Caretaker]";
		mes "They will still be hungry tomorrow.";
		close;
	}

OnInit:
	// Monster-Id -> Schmuck-Id, gleiche Reihenfolge in beiden
	// Feldern. Erzeugt aus db/re/pet_db.yml, 46 Eintraege.
	setarray .tier[0],
		1002, 1011, 1014, 1019, 1023, 1026, 1029, 1031, 1035, 1040,
		1042, 1049, 1052, 1056, 1057, 1063, 1077, 1101, 1107, 1109,

		1299, 1370, 1374, 1379, 1401, 1404, 1416, 1495, 1504, 1505,
		1513, 1586, 1630, 1837, 20696, 20697;
	setarray .schmuck[0],
		10013, 10002, 10017, 10010, 10009, 10008, 10006, 10013, 10002, 10035,
		10002, 10012, 10014, 10019, 10018, 10007, 10017, 10001, 10003, 10004,
		10005, 10013, 10025, 10032, 10011, 10015, 10016, 10027, 10020, 10039,
		10028, 10037, 10034, 10036, 10026, 10030, 10029, 10033, 10031, 10024,

	end;
