//  Angaben in Prozent: 100 = 1x, 2500 = 25x, 5000 = 50x

// --- Erfahrung: 25x ---
base_exp_rate: 2500
job_exp_rate: 2500
quest_exp_rate: 2500

// --- Levelabzug endet bei Stufe 275 ---
//  Ab 31 Stufen Unterschied bleiben von der Erfahrung 10 % uebrig.
//  Das hoechste normale Monster steht auf Stufe 279, ab Stufe 296 gibt
//  es also nichts mehr, was ohne Abzug zaehlt - bestraft wuerde etwas,
//  wogegen der Spieler nichts tun kann. Deshalb faellt der Abzug ab 275.
//  Betrifft nur Erfahrung; auf Beute bleibt er bestehen.
//  0 schaltet es ab. Braucht eine neu uebersetzte map-server.
exp_penalty_maxlevel: 275

// --- Mehrere Stufen je Kill ---
//  Die Vorgabe "no" laesst pro Kill hoechstens eine Stufe zu und stutzt
//  den Ueberschuss auf "naechste Stufe minus 1" (pc.cpp, Zeile 8255).
//  Bei 25-facher Erfahrung ueberschiesst schon ein Poring auf Stufe 1
//  das Sechsfache einer Stufe - der Rest verfaellt, und Boni aus
//  Ausruestung sind dort ueberhaupt nicht mehr zu bemerken, weil beide
//  Faelle auf denselben gestutzten Wert fallen.
//
//  Die Level-Up-Anzeige leidet nicht darunter: pc_checkbaselevelup
//  zaehlt in der Schleife nur die Stufe hoch, alles Sichtbare steht
//  DAHINTER und laeuft genau einmal - Heilung, Statusaktualisierung,
//  clif_misceffect (Ton und Animation) und das NPC-Ereignis.
//  Fuer die Job-Stufen gilt dasselbe.
multi_level_up: yes

// --- Drops: 25x ---
//  common = normale Items, heal = Heilitems, use = Verbrauchbares,
//  equip = Ausruestung. Alle vier, sonst faellt eine Kategorie zurueck.
item_rate_common: 2500
item_rate_heal: 2500
item_rate_use: 2500
item_rate_equip: 2500

//  ACHTUNG - "_boss" heisst NICHT MVP.
//
//  src/map/mob.cpp:6859 unterscheidet drei Faelle:
//
//      bool is_mvp  = mob->get_bosstype() == BOSSTYPE_MVP;
//      bool is_boss = mob->get_bosstype() == BOSSTYPE_MINIBOSS;
//      ...
//      rate_adjust = is_mvp  ? item_rate_common_mvp
//                  : is_boss ? item_rate_common_boss
//                            : item_rate_common;
//
//  "_boss" gilt also fuer MINIBOSSE. Fuer MVPs gibt es eine dritte,
//  eigene Reihe von Einstellungen. Fehlt sie, bleibt sie auf dem
//  rAthena-Standard 100 - also 1x, waehrend ringsum 25x laeuft.
item_rate_common_boss: 2500
item_rate_heal_boss: 2500
item_rate_use_boss: 2500
item_rate_equip_boss: 2500

//  Die dritte Reihe: MVPs. Ohne diese vier Zeilen laufen ALLE
//  MVP-Drops auf 1x, und zwar lautlos - im Spiel sieht man nur,
//  dass nichts faellt.
item_rate_common_mvp: 2500
item_rate_heal_mvp: 2500
item_rate_use_mvp: 2500
item_rate_equip_mvp: 2500

//  Und noch eine, die leicht mit den obigen verwechselt wird:
//  item_rate_mvp betrifft die MvpDrops-Liste - die Sonderbelohnung
//  fuer den staerksten Angreifer, nicht die normalen Drops des
//  Monsters. Eigener Schalter, eigener Standard, ebenfalls 100.
item_rate_mvp: 2500

// --- Karten: 50x ---
//  Basisrate einer Karte liegt bei 0,01 %. Mal 50 sind das 0,5 % -
//  immer noch selten, aber in einer Spielsitzung erreichbar.
//
//  Alle drei Stufen, aus demselben Grund wie oben. Ohne die letzte
//  Zeile fiel eine MVP-Karte weiter mit 0,01 % - nachweisbar mit
//  "@whodrops <Karten-ID>": dort stand vor der Korrektur der
//  ungeaenderte Basiswert.
item_rate_card: 5000
item_rate_card_boss: 5000
item_rate_card_mvp: 5000

// --- Destiny Coins: unverkaeuflich, aber zahlbar ---
//  Gehoert zu "Trade: NoSell" beim Destiny_Coin in
//  db/import/item_db.yml. Ohne diese Zeile schliessen sich die
//  beiden Dinge aus, die wir wollen.
//
//  Der Grund steht in npc.cpp gleich zweimal:
//
//    Zeile 2461  beim Bezahlen ueberspringt der itemshop jeden
//                Stapel, den pc_can_sell_item ablehnt
//    Zeile 2645  und derselbe Aufruf bestimmt den ANGEZEIGTEN
//                Kontostand
//
//  Mit NoSell allein zeigt der Custom Merchant deshalb null Coins
//  an und nimmt keine entgegen. Sichtbar bleibt nur das
//  Standardfeld des Client-Fensters, das "Kafra Points" heisst -
//  daher der irrefuehrende Eindruck, er wolle eine andere Waehrung.
//
//  0x2 ist genau dafuer gedacht, conf/battle/items.conf:114 sagt es
//  woertlich: "Sell restricted items are able to be sold to
//  Itemshops". Fuer normale Haendler (NPCTYPE_SHOP) gibt es diesen
//  Weg NICHT - dort greift weiterhin itemdb_cansell, die Coins
//  bleiben also aus dem Verkaufsfenster verschwunden.
//
//  Bewusst 0x2 und nicht mehr: 0x1 und 0x4 betreffen gebundene
//  Gegenstaende und haben mit unserem Fall nichts zu tun.
allow_bound_sell: 0x2

// --- Sichtfeld: 14 -> 16 ---
//  Wie weit ein Spieler sehen kann, in Feldern. rAthenas Standard
//  ist 14 (battle.cpp:8585). Wer weiter herauszoomt, sieht ab dort
//  leere Flaeche - nicht weil nichts da waere, sondern weil der
//  Server nichts mehr schickt.
//
//  WAS DAS KOSTET
//  Die Flaeche waechst quadratisch, nicht linear:
//      14 -> 29 x 29 =  841 Felder
//      16 -> 33 x 33 = 1089 Felder     + 29 %
//  Entsprechend mehr Objekte je Aktualisierung und Spieler, also
//  mehr Bandbreite und mehr Rechenzeit. Bei unserer Spielerzahl
//  ist das nichts; bei zweihundert gleichzeitig waere es eine Zahl,
//  die man im Auge behaelt.
//
//  WAS SICH SONST NOCH AENDERT
//  area_size ist nicht nur Anzeige. Mitgewachsen sind:
//    mob.cpp:2351   Monster in AREA_SIZE+4 Feldern schalten in die
//                   aktive KI - aus 18 werden 20 Felder
//    mob.cpp:1959   ein Monster verliert sein Ziel erst, wenn der
//                   Angreifer weiter als AREA_SIZE entfernt ist
//    mob.cpp:3576   Quest- und Erfolgsfortschritt fuer Gruppen
//                   zaehlt in diesem Umkreis
//
//  Die Sichtweite normaler Monster haengt NICHT daran - die steht
//  je Monster in mob_db.yml. Der AREA_SIZE-Eintrag in mob.cpp:4626
//  betrifft nur Klone aus @clone.
//
//  Zum Zurueckdrehen: Zeile loeschen, @reloadbattleconf.
area_size: 16

// --- Was @whodrops und @mobinfo anzeigen ---
//  Ab Werk (conf/battle/gm.conf: 1) rechnen beide Befehle die
//  Levelstrafe des ABFRAGENDEN Spielers ein. gm.conf sagt es
//  woertlich: "display the users' real drop rate".
//
//  Fuer einen Spieler ist das richtig - ihn interessiert seine
//  eigene Chance. Fuer die Serververwaltung ist es eine Falle: man
//  prueft eine Rate, sieht die Haelfte und sucht stundenlang einen
//  Fehler in der Konfiguration, den es nicht gibt. Genau das ist
//  uns passiert.
//
//  Auf 0 zeigen beide Befehle den Wert, der wirklich in der
//  Monsterdatenbank steht. Damit wird "@whodrops" zu einem
//  Messwerkzeug statt zu einer Auskunft - und der Unterschied
//  zwischen beiden ist prueffbar, statt geraten zu werden.
//
//  Der Code-Standard ist ohnehin 0 (battle.cpp:8711); nur die
//  mitgelieferte gm.conf setzt 1.
atcommand_mobinfo_type: 0

// --- Truhen: BEWUSST bei 1x ---
//  item_rate_treasure ist die einzige Drop-Rate, die wir nicht
//  anheben. Sie betrifft die Schatztruhen der WoE-Burgen und die
//  festen Truhen in Verliesen - Inhalte, die niemand erfarmt,
//  sondern die zu festen Zeiten anfallen. Mit 25x waere eine
//  einzige Burgnacht mehr wert als eine Woche Spielen.
//
//  Steht hier als Zeile und nicht als Auslassung, damit beim
//  naechsten Durchsehen klar ist: entschieden, nicht vergessen.
item_rate_treasure: 100
